//====================================================
// MANUAL ENTRY VISUAL LEVELS
//====================================================
void DrawManualExecutionLevels()
{
   //====================================================
   // VALIDATION
   //====================================================
   if (regime == REGIME_RANGE)
      return;   
   //====================================================
   // CLEAN
   //====================================================
   int htfBias = GetHTFBias();
   ObjectDelete(0, "MANUAL_ENTRY");
   ObjectDelete(0, "MANUAL_ENTRY 2");
   ObjectDelete(0, "MANUAL_SL");
   ObjectDelete(0, "MANUAL_TP");
   ObjectDelete(0, "MANUAL_RR_TEXT");
   int shift = shifPerspektif;
   if(!CachedBullEvent.valid || !CachedBearEvent.valid)
      return;
   //====================================================
   // MARKET DATA
   //====================================================
   double entry, entry2 = 0;
   double open = iOpen(_Symbol, LTF, shift);
   double close = iClose(_Symbol, LTF, shift);
   double high = iHigh(_Symbol, LTF, shift);
   double low = iLow(_Symbol, LTF, shift);
   double atr =
       GetATR(2);
   if (atr <= 0)
      return;
   //====================================================
   // PRICE LEVELS
   //====================================================
   double sl = 0.0;
   double tp = 0.0;
   double riskDistance =
       atr *
       DefaultSL_ATR_Multiplier;
   double rewardDistance =
       atr *
       DefaultTP_ATR_Multiplier;
   //====================================================
   // BUY SETUP
   //====================================================
   if (regime == REGIME_TREND_BULL && ConfirmBullishCandle(shift) && htfBias == 1 && CachedBullEvent.valid)
   {
      entry = high - (high - low) * 0.62;
      entry2 = high - (high - low) * 0.79;
      sl =
          entry -
          riskDistance;
      tp =
          entry +
          rewardDistance;
   }
   //====================================================
   // SELL SETUP
   //====================================================
   if (regime == REGIME_TREND_BEAR && ConfirmBearishCandle(shift) && htfBias == -1 && CachedBearEvent.valid)
   {
      entry = low + (high - low) * 0.62;
      entry2 = low + (high - low) * 0.79;
      sl =
          entry +
          riskDistance;
      tp =
          entry -
          rewardDistance;
   }
   //====================================================
   // NORMALIZE
   //====================================================
   entry =
       NormalizeDouble(
           entry,
           _Digits);
   entry2 =
       NormalizeDouble(
           entry2,
           _Digits);
   sl =
       NormalizeDouble(
           sl,
           _Digits);
   tp =
       NormalizeDouble(
           tp,
           _Digits);
   //====================================================
   // RR
   //====================================================
   double risk =
       MathAbs(
           entry - sl);
   double reward =
       MathAbs(
           tp - entry);
   double rr =
       0.0;
   if (risk > 0)
   {
      rr =
          reward / risk;
   }
   //====================================================
   // ENTRY LINE
   //====================================================
   DrawLine(
       "MANUAL_ENTRY",
       entry,
       clrWhite);
   DrawLine(
       "MANUAL_ENTRY 2",
       entry2,
       clrWhite);
   //====================================================
   // STOP LOSS
   //====================================================
   DrawLine(
       "MANUAL_SL",
       sl,
       clrRed);
   //====================================================
   // TAKE PROFIT
   //====================================================
   DrawLine(
       "MANUAL_TP",
       tp,
       clrBlueViolet);
   //====================================================
   // VISUAL BOX
   //====================================================
   datetime t1 =
       TimeCurrent();
   datetime t2 =
       t1 +
       (PeriodSeconds(LTF) * 15);
   //----------------------------------------------------
   // RISK BOX
   //----------------------------------------------------
   ObjectCreate(
       0,
       "MANUAL_RISK_BOX",
       OBJ_RECTANGLE,
       0,
       t1,
       entry,
       t2,
       sl);
   ObjectSetInteger(
       0,
       "MANUAL_RISK_BOX",
       OBJPROP_FILL,
       true);
   ObjectSetInteger(
       0,
       "MANUAL_RISK_BOX",
       OBJPROP_BACK,
       true);
   ObjectSetInteger(
       0,
       "MANUAL_RISK_BOX",
       OBJPROP_COLOR,
       ColorToPRGB(
           clrRed,
           40));
   //----------------------------------------------------
   // REWARD BOX
   //----------------------------------------------------
   ObjectCreate(
       0,
       "MANUAL_REWARD_BOX",
       OBJ_RECTANGLE,
       0,
       t1,
       entry,
       t2,
       tp);
   ObjectSetInteger(
       0,
       "MANUAL_REWARD_BOX",
       OBJPROP_FILL,
       true);
   ObjectSetInteger(
       0,
       "MANUAL_REWARD_BOX",
       OBJPROP_BACK,
       true);
   ObjectSetInteger(
       0,
       "MANUAL_REWARD_BOX",
       OBJPROP_COLOR,
       ColorToPRGB(
           clrLime,
           40));
}

int CountTouches(double levelPrice,
                 double tolerance,
                 int start,
                 int end,
                 bool isHigh)
{
   int count = 0;
   double strength = 0;
   int bars = ArraySize(ratesLTF);
   if (start < 0)
      start = 0;
   if (end >= bars)
      end = bars - 1;
   bool inCluster = false;
   for (int i = start; i <= end; i++)
   {
      double high = ratesLTF[i].high;
      double low = ratesLTF[i].low;
      double open = ratesLTF[i].open;
      double close = ratesLTF[i].close;
      double price = isHigh ? high : low;
      bool touched = MathAbs(price - levelPrice) <= tolerance;
      if (touched)
      {
         if (!inCluster)
         {
            count++;
            inCluster = true;
            //============================
            // 🔥 WICK STRENGTH
            //============================
            double wick = 0;
            if (isHigh)
               wick = high - MathMax(open, close);
            else
               wick = MathMin(open, close) - low;
            if (wick > tolerance * 0.5)
               strength += 1.5; // strong rejection
            else
               strength += 1.0; // normal touch
         }
      }
      else
      {
         inCluster = false;
      }
   }
   // 🔥 return weighted result
   return (int)(count + strength);
}

int DetectExternalLiquidityTF(LiquidityLevel &levels[], bool forBuy, ENUM_TIMEFRAMES tf)
{
   ArrayResize(levels, 0);
   MqlRates rates[];
   int copied = CopyRates(_Symbol, tf, 0, 250, rates);
   if (copied < 100)
      return 0;
   //========================================
   // 🔥 TF-SPECIFIC ATR (IMPORTANT FIX)
   //========================================
   double atr = 0;
   {
      int h = iATR(_Symbol, tf, 14);
      double buf[];
      ArraySetAsSeries(buf, true);
      if (h != INVALID_HANDLE && CopyBuffer(h, 0, 0, 1, buf) >= 1)
         atr = buf[0];
      IndicatorRelease(h);
   }
   if (atr <= 0)
      return 0;
   double tolerance = atr * 0.25;
   //========================================
   // 🔥 STRUCTURE ZONE (MORE REALISTIC)
   //========================================
   double structureHigh = -DBL_MAX;
   double structureLow = DBL_MAX;
   int structureLookback = MathMin(150, copied - 20);
   for (int i = 20; i < structureLookback; i++)
   {
      structureHigh = MathMax(structureHigh, rates[i].high);
      structureLow = MathMin(structureLow, rates[i].low);
   }
   //========================================
   // 🔥 LIQUIDITY DETECTION
   //========================================
   for (int i = 30; i < structureLookback - 10; i++)
   {
      double baseHigh = rates[i].high;
      double baseLow = rates[i].low;
      int touchesHigh = 0;
      int touchesLow = 0;
      //========================================
      // 🔥 TOUCH LOGIC (MORE STRICT)
      //========================================
      for (int k = i - 1; k >= i - 12 && k >= 0; k--)
      {
         if (MathAbs(rates[k].high - baseHigh) < tolerance &&
             rates[k].close < rates[k].high) // rejection filter
            touchesHigh++;
         if (MathAbs(rates[k].low - baseLow) < tolerance &&
             rates[k].close > rates[k].low)
            touchesLow++;
      }
      //========================================
      // 🔥 EXTERNAL LIQUIDITY (BUY SIDE)
      //========================================
      if (forBuy)
      {
         if (baseHigh > structureHigh && touchesHigh >= 2)
         {
            LiquidityLevel lvl;
            lvl.price = baseHigh;
            // 🔥 REAL SWEEP (not single candle)
            lvl.swept =
                (rates[0].high > baseHigh &&
                 rates[0].close < baseHigh); // sweep + rejection
            if (!lvl.swept)
            {
               int sz = ArraySize(levels);
               ArrayResize(levels, sz + 1);
               levels[sz] = lvl;
            }
         }
      }
      //========================================
      // 🔥 EXTERNAL LIQUIDITY (SELL SIDE)
      //========================================
      else
      {
         if (baseLow < structureLow && touchesLow >= 2)
         {
            LiquidityLevel lvl;
            lvl.price = baseLow;
            lvl.swept =
                (rates[0].low < baseLow &&
                 rates[0].close > baseLow);
            if (!lvl.swept)
            {
               int sz = ArraySize(levels);
               ArrayResize(levels, sz + 1);
               levels[sz] = lvl;
            }
         }
      }
   }
   return ArraySize(levels);
}

//+------------------------------------------------------------------+
int DetectLiquidityClusters(LiquidityLevel &levels[], bool isHigh, double atr)
{
   ArrayResize(levels, 0);
   int bars = ArraySize(ratesLTF);
   if (bars < 100)
      return 0;
   double tolerance = atr * 0.15;
   double mergeDist = atr * 0.25;
   for (int i = 5; i < bars - 10; i++)
   {
      double base = isHigh ? ratesLTF[i].high : ratesLTF[i].low;
      int touches = 0;
      for (int k = i + 1; k <= i + 10; k++)
      {
         double price = isHigh ? ratesLTF[k].high : ratesLTF[k].low;
         if (MathAbs(price - base) < tolerance)
            touches++;
      }
      if (touches >= 2) // EQH/EQL minimal
      {
         LiquidityLevel lvl;
         lvl.price = base;
         lvl.touches = touches;
         lvl.isHigh = isHigh;
         lvl.age = i;
         lvl.swept = false;
         //==================================
         // 🔥 MERGE CLUSTER
         //==================================
         bool merged = false;
         for (int j = 0; j < ArraySize(levels); j++)
         {
            if (levels[j].isHigh == lvl.isHigh &&
                MathAbs(levels[j].price - lvl.price) < mergeDist)
            {
               // ambil yang lebih kuat
               if (lvl.touches > levels[j].touches)
                  levels[j] = lvl;
               merged = true;
               break;
            }
         }
         if (!merged)
         {
            int sz = ArraySize(levels);
            ArrayResize(levels, sz + 1);
            levels[sz] = lvl;
         }
      }
   }
   return ArraySize(levels);
}

void ClassifyLiquidity(LiquidityLevel &levels[])
{
   int n = ArraySize(levels);
   if (n == 0)
      return;
   // sort by strength DESC
   for (int i = 0; i < n - 1; i++)
   {
      for (int j = i + 1; j < n; j++)
      {
         if (levels[j].strength > levels[i].strength)
         {
            LiquidityLevel tmp = levels[i];
            levels[i] = levels[j];
            levels[j] = tmp;
         }
      }
   }
   for (int i = 0; i < n; i++)
   {
      if (i == 0)
         levels[i].type = LIQ_EXTERNAL;
      else if (i <= 2)
         levels[i].type = LIQ_MID;
      else
         levels[i].type = LIQ_INTERNAL;
   }
}

//+------------------------------------------------------------------+
double ScoreLiquidity(LiquidityLevel &lvl, double entry, double sl, double atr, bool isBuy)
{
   double dist = MathAbs(lvl.price - entry);
   double rr = dist / MathAbs(entry - sl);
   if (rr < 1.2 || rr > 6.0)
      return -1;
   double score = 0;
   //==================================
   // 🔥 CORE FACTORS
   //==================================
   score += lvl.touches * 2.0;          // cluster strength
   score += rr * 1.5;                   // reward
   score += (lvl.isExternal ? 3.0 : 1); // external priority
   score += (1.0 / (lvl.age + 1));      // freshness
                                        //==================================
                                        // 🔥 DIRECTION FILTER
                                        //==================================
   if (isBuy && lvl.price <= entry)
      return -1;
   if (!isBuy && lvl.price >= entry)
      return -1;
   return score;
}

//+------------------------------------------------------------------+
double GetLiquidityMapTP(bool isBuy, double entry, double sl)
{
   double atr = GetATR(1);
   if (atr <= 0)
      return 0;
   double risk = MathAbs(entry - sl);
   if (risk <= 0)
      return 0;
   LiquidityLevel highs[], lows[];
   int countHigh = DetectLiquidityClusters(highs, true, atr);
   int countLow = DetectLiquidityClusters(lows, false, atr);
   if (countHigh == 0 && countLow == 0)
   {
      // fallback RR
      double rr = 2.0;
      double tp = isBuy
                      ? entry + risk * rr
                      : entry - risk * rr;
      if (DEBUG_SMC)
         Print("TP fallback (no liquidity)");
      return NormalizeDouble(tp, _Digits);
   }
   // classify
   if (countHigh > 0)
      ClassifyLiquidity(highs);
   if (countLow > 0)
      ClassifyLiquidity(lows);
   double bestTP = 0;
   double bestScore = -DBL_MAX;
   //========================================
   // 🔥 SELECT POOL MANUALLY (NO REFERENCE BUG)
   //========================================
   if (isBuy)
   {
      for (int i = 0; i < countHigh; i++)
      {
         LiquidityLevel lvl = highs[i];
         if (lvl.swept)
            continue;
         if (lvl.price <= entry)
            continue;
         double score = ScoreLiquidity(lvl, entry, sl, atr, true);
         if (score > bestScore)
         {
            bestScore = score;
            bestTP = lvl.price;
         }
      }
   }
   else
   {
      for (int i = 0; i < countLow; i++)
      {
         LiquidityLevel lvl = lows[i];
         if (lvl.swept)
            continue;
         if (lvl.price >= entry)
            continue;
         double score = ScoreLiquidity(lvl, entry, sl, atr, false);
         if (score > bestScore)
         {
            bestScore = score;
            bestTP = lvl.price;
         }
      }
   }
   //========================================
   // 🔥 FINAL FALLBACK (IF ALL FILTERED OUT)
   //========================================
   if (bestTP == 0)
   {
      double rr = 2.0;
      bestTP = isBuy
                   ? entry + risk * rr
                   : entry - risk * rr;
      if (DEBUG_SMC)
         Print("TP fallback (filtered out)");
   }
   else
   {
      if (DEBUG_SMC)
         Print("TP selected from liquidity: ", bestTP);
   }
   return NormalizeDouble(bestTP, _Digits);
}

TPLadder GetSmartTP(bool isBuy, double entry, double sl)
{
   TPLadder tp;
   tp.tp1 = 0;
   tp.tp2 = 0;
   tp.tp3 = 0;
   double atr = GetATR(1);
   if (atr <= 0)
      return tp;
   double risk = MathAbs(entry - sl);
   if (risk <= 0)
      return tp;
   //==================================================
   // 🔥 1. INTERNAL LIQUIDITY (LTF)
   //==================================================
   LiquidityLevel internal[];
   DetectLiquidityLevels(internal, isBuy);
   double bestInternal = 0;
   double bestScoreInt = -DBL_MAX;
   for (int i = 0; i < ArraySize(internal); i++)
   {
      LiquidityLevel lvl = internal[i];
      if (lvl.swept)
         continue;
      if (isBuy && lvl.price <= entry)
         continue;
      if (!isBuy && lvl.price >= entry)
         continue;
      double dist = MathAbs(lvl.price - entry);
      double rr = dist / risk;
      if (dist < atr * 0.5 || rr < 0.8)
         continue;
      double score =
          lvl.touches * 2.0 +
          (1.0 / (lvl.age + 1)) +
          rr;
      if (score > bestScoreInt)
      {
         bestScoreInt = score;
         bestInternal = lvl.price;
      }
   }
   tp.tp1 = bestInternal;
   //==================================================
   // 🔥 2. EXTERNAL LIQUIDITY (MTF)
   //==================================================
   ENUM_TIMEFRAMES tfList[] = {PERIOD_M15, PERIOD_H1, PERIOD_H4};
   double bestExternal = 0;
   double bestScoreExt = -DBL_MAX;
   for (int t = 0; t < ArraySize(tfList); t++)
   {
      LiquidityLevel levels[];
      int count = DetectExternalLiquidityTF(levels, isBuy, tfList[t]);
      for (int i = 0; i < count; i++)
      {
         LiquidityLevel lvl = levels[i];
         if (lvl.swept)
            continue;
         if (isBuy && lvl.price <= entry)
            continue;
         if (!isBuy && lvl.price >= entry)
            continue;
         double dist = MathAbs(lvl.price - entry);
         double rr = dist / risk;
         if (dist < atr * 1.0 || rr < 1.5)
            continue;
         double tfWeight = 1.0;
         if (tfList[t] == PERIOD_H1)
            tfWeight = 1.5;
         if (tfList[t] == PERIOD_H4)
            tfWeight = 2.0;
         double score =
             lvl.touches * 2.0 +
             rr * 1.5 +
             tfWeight * 2.0 +
             (1.0 / (lvl.age + 1));
         if (score > bestScoreExt)
         {
            bestScoreExt = score;
            bestExternal = lvl.price;
         }
      }
   }
   tp.tp2 = bestExternal;
   //==================================================
   // 🔥 3. HTF LIQUIDITY (D1 / W1)
   //==================================================
   ENUM_TIMEFRAMES htfList[] = {PERIOD_D1, PERIOD_W1};
   double bestHTF = 0;
   double bestScoreHTF = -DBL_MAX;
   for (int t = 0; t < ArraySize(htfList); t++)
   {
      LiquidityLevel levels[];
      int count = DetectExternalLiquidityTF(levels, isBuy, htfList[t]);
      for (int i = 0; i < count; i++)
      {
         LiquidityLevel lvl = levels[i];
         if (lvl.swept)
            continue;
         if (isBuy && lvl.price <= entry)
            continue;
         if (!isBuy && lvl.price >= entry)
            continue;
         double dist = MathAbs(lvl.price - entry);
         double rr = dist / risk;
         if (rr < 2.5)
            continue;
         double tfWeight = (htfList[t] == PERIOD_W1) ? 3.0 : 2.5;
         double score =
             lvl.touches * 2.0 +
             rr * 2.0 +
             tfWeight * 2.0;
         if (score > bestScoreHTF)
         {
            bestScoreHTF = score;
            bestHTF = lvl.price;
         }
      }
   }
   tp.tp3 = bestHTF;
   //==================================================
   // 🔥 FALLBACK SAFETY
   //==================================================
   if (tp.tp1 == 0)
      tp.tp1 = isBuy ? entry + risk * 1.0 : entry - risk * 1.0;
   if (tp.tp2 == 0)
      tp.tp2 = isBuy ? entry + risk * 2.0 : entry - risk * 2.0;
   if (tp.tp3 == 0)
      tp.tp3 = isBuy ? entry + risk * 3.0 : entry - risk * 3.0;
   //==================================================
   // 🔥 DEBUG
   //==================================================
   if (DEBUG_SMC)
   {
      Print("TP LADDER:");
      Print("TP1=", tp.tp1);
      Print("TP2=", tp.tp2);
      Print("TP3=", tp.tp3);
   }
   return tp;
}
//================ SAFE =================//
int DetectLiquidityLevels(LiquidityLevel &levels[], bool forBuy, int maxLevels = 10)
{
   ArrayResize(levels, 0);
   int bars = ArraySize(ratesHTF);
   if (bars < 100)
      return 0;
   double atr = GetATR_HTF(1);
   if (atr <= 0)
      return 0;
   double tolerance = atr * 0.2;
   //==================================================
   // STEP 1: FIND RECENT STRUCTURE
   //==================================================
   double structureHigh = -DBL_MAX;
   double structureLow = DBL_MAX;
   for (int i = 5; i < 20; i++)
   {
      if (ratesHTF[i].high > structureHigh)
         structureHigh = ratesHTF[i].high;
      if (ratesHTF[i].low < structureLow)
         structureLow = ratesHTF[i].low;
   }
   //==================================================
   // STEP 2: SCAN EXTERNAL LIQUIDITY
   //==================================================
   for (int i = 20; i < 150; i++)
   {
      double baseHigh = ratesHTF[i].high;
      double baseLow = ratesHTF[i].low;
      int touchesHigh = 0;
      int touchesLow = 0;
      for (int k = i - 1; k >= i - 10 && k >= 0; k--)
      {
         if (MathAbs(ratesHTF[k].high - baseHigh) < tolerance)
            touchesHigh++;
         if (MathAbs(ratesHTF[k].low - baseLow) < tolerance)
            touchesLow++;
      }
      //========================
      // BUY TARGET → EQH ABOVE STRUCTURE
      //========================
      if (forBuy)
      {
         if (baseHigh <= structureHigh)
            continue;
         if (touchesHigh >= 2)
         {
            LiquidityLevel lvl;
            lvl.price = baseHigh;
            lvl.isHigh = true;
            lvl.touches = touchesHigh;
            lvl.age = i;
            // belum di sweep
            lvl.swept = (ratesHTF[0].high > baseHigh);
            if (!lvl.swept)
            {
               int sz = ArraySize(levels);
               if (sz < maxLevels)
               {
                  ArrayResize(levels, sz + 1);
                  levels[sz] = lvl;
               }
            }
         }
      }
      //========================
      // SELL TARGET → EQL BELOW STRUCTURE
      //========================
      if (!forBuy)
      {
         if (baseLow >= structureLow)
            continue;
         if (touchesLow >= 2)
         {
            LiquidityLevel lvl;
            lvl.price = baseLow;
            lvl.isHigh = false;
            lvl.touches = touchesLow;
            lvl.age = i;
            lvl.swept = (ratesHTF[0].low < baseLow);
            if (!lvl.swept)
            {
               int sz = ArraySize(levels);
               if (sz < maxLevels)
               {
                  ArrayResize(levels, sz + 1);
                  levels[sz] = lvl;
               }
            }
         }
      }
   }
   //==================================================
   // SORT → NEAREST VALID EXTERNAL
   //==================================================
   for (int i = 0; i < ArraySize(levels) - 1; i++)
   {
      for (int j = i + 1; j < ArraySize(levels); j++)
      {
         if (forBuy)
         {
            if (levels[j].price < levels[i].price)
            {
               LiquidityLevel tmp = levels[i];
               levels[i] = levels[j];
               levels[j] = tmp;
            }
         }
         else
         {
            if (levels[j].price > levels[i].price)
            {
               LiquidityLevel tmp = levels[i];
               levels[i] = levels[j];
               levels[j] = tmp;
            }
         }
      }
   }
   return ArraySize(levels);
}

double GetNearestLiquidityTarget(bool isBuy, LiquidityLevel &levels[])
{
   double price = SymbolInfoDouble(_Symbol, SYMBOL_BID);
   double bestTarget = 0;
   double minDist = DBL_MAX;
   for (int i = 0; i < ArraySize(levels); i++)
   {
      if (levels[i].swept)
         continue;
      // BUY → cari high di atas
      if (isBuy && levels[i].isHigh && levels[i].price > price)
      {
         double dist = levels[i].price - price;
         if (dist < minDist)
         {
            minDist = dist;
            bestTarget = levels[i].price;
         }
      }
      // SELL → cari low di bawah
      if (!isBuy && !levels[i].isHigh && levels[i].price < price)
      {
         double dist = price - levels[i].price;
         if (dist < minDist)
         {
            minDist = dist;
            bestTarget = levels[i].price;
         }
      }
   }
   return bestTarget;
}

   {

         found++;
      }
      else if (found == 1)
      {
         tp2 = levels[i].price;
         return true;
      }
   }
   return (found > 0);
}

bool GetTwoLiquidityTargets(bool isBuy,
                            LiquidityLevel &levels[],
                            double &tp1,
                            double &tp2)
{
   double price = SymbolInfoDouble(_Symbol, SYMBOL_BID);
   double best1 = 0, best2 = 0;
   double dist1 = DBL_MAX, dist2 = DBL_MAX;
   for (int i = 0; i < ArraySize(levels); i++)
   {
      if (levels[i].swept)
         continue;
      if (isBuy && levels[i].isHigh && levels[i].price > price)
      {
         double dist = levels[i].price - price;
         if (dist < dist1)
         {
            dist2 = dist1;
            best2 = best1;
            dist1 = dist;
            best1 = levels[i].price;
         }
         else if (dist < dist2)
         {
            dist2 = dist;
            best2 = levels[i].price;
         }
      }
      if (!isBuy && !levels[i].isHigh && levels[i].price < price)
      {
         double dist = price - levels[i].price;
         if (dist < dist1)
         {
            dist2 = dist1;
            best2 = best1;
            dist1 = dist;
            best1 = levels[i].price;
         }
         else if (dist < dist2)
         {
            dist2 = dist;
            best2 = levels[i].price;
         }
      }
   }
   tp1 = best1;
   tp2 = best2;
   return (tp1 != 0);
}

bool IsFVGBroken(double low,
                 double high,
                 int shift,
                 double atr,
                 bool bullish)
{
   double size = high - low;
   if (size <= _Point || atr <= 0)
      return true;
   int bars = ArraySize(ratesLTF);
   int maxLook = 50;
   double maxFill = 0;
   for (int i = shift - 1; i >= 0 && i >= shift - maxLook && i < bars; i--)
   {
      double h = ratesLTF[i].high;
      double l = ratesLTF[i].low;
      double c = ratesLTF[i].close;
      double overlap = MathMin(h, high) - MathMax(l, low);
      if (overlap <= 0)
         continue;
      double fillRatio = overlap / size;
      if (fillRatio > maxFill)
         maxFill = fillRatio;
      //============================
      // FULL FILL → BROKEN
      //============================
      if (fillRatio > 0.95)
      {
         if (bullish)
         {
            // [FIX] close-based break, threshold selaras dgn IsFVGMitigated (0.3 ATR)
            if (c < low - atr * 0.3)
               return true;
         }
         else
         {
            if (c > high + atr * 0.3)
               return true;
         }
      }
   }
   //============================
   // PARTIAL FILL → WEAK
   // [FIX] selaraskan 0.85 (sama dengan IsFVGMitigated)
   //============================
   if (maxFill > 0.85)
      return true;
   return false;
}

bool IsOBBroken(double obLow,
                double obHigh,
                int obShift,
                double atr,
                bool bullish)
{
   int bars = ArraySize(ratesLTF);
   if (obShift <= 0 || obShift >= bars || atr <= 0)
      return false;
   int maxLook = 50;
   for (int i = obShift - 1; i >= 0 && i >= obShift - maxLook; i--)
   {
      double close = ratesLTF[i].close;
      double open = ratesLTF[i].open;
      double high = ratesLTF[i].high;
      double low = ratesLTF[i].low;
      double range = high - low;
      bool strongCandle = (range > atr * IMPULSE_MIN_ATR);
      //============================
      // BULLISH OB (break ke bawah)
      //============================
      if (bullish)
      {
         // [FIX] buffer 0.1 → 0.3 ATR, konsisten dengan IsOBMitigated
         if (close < obLow - atr * 0.3)
         {
            if (strongCandle && close < open)
               return true;
         }
      }
      //============================
      // BEARISH OB (break ke atas)
      //============================
      else
      {
         if (close > obHigh + atr * 0.3)
         {
            if (strongCandle && close > open)
               return true;
         }
      }
   }
   return false;
}

//+------------------------------------------------------------------+
//|                                                                  |
//+------------------------------------------------------------------+
bool IsChoppyMarket(int shift, double atr)
{
   int lookback = 10;
   double avgRange = 0;
   for (int i = shift; i < shift + lookback; i++)
   {
      double range = ratesLTF[i].high - ratesLTF[i].low;
      avgRange += range;
   }
   avgRange /= lookback;
   if (avgRange <= 0)
      return true;
   // Kalau candle range rata-rata jauh lebih kecil dari ATR → market choppy/compress
   double ratio = avgRange / atr;
   if (ratio < 0.6) // candle kecil-kecil relatif ATR → choppy
      return true;
   return false;
}

double NormalizeLot(double lot)
{
   double minLot = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_MIN);
   double maxLot = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_MAX);
   double stepLot = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_STEP);
   lot = MathMax(minLot, MathMin(maxLot, lot));
   lot = MathFloor(lot / stepLot) * stepLot;
   return NormalizeDouble(lot, 2);
}

//====================================================
// 🔥 CALCULATE RAW LOT BASED ON RISK
//====================================================
double CalculateLotRaw(double entry, double sl, double riskMoney)
{
   double tickValue = SymbolInfoDouble(_Symbol, SYMBOL_TRADE_TICK_VALUE);
   double stopPoints = MathAbs(entry - sl) / _Point;
   if (stopPoints <= 0 || tickValue <= 0)
      return 0;
   return riskMoney / (stopPoints * tickValue);
}

//====================================================
// 🔥 BUILD ADAPTIVE ENTRY LAYERS
//====================================================
int BuildEntryLayers(bool isBuy,
                     double zoneLow,
                     double zoneHigh,
                     double sl,
                     EntryLayer &layers[])
{
   ArrayResize(layers, 3);
   double balance = AccountInfoDouble(ACCOUNT_BALANCE);
   double riskMoney = balance * 0.01; // 1% risk
   double range = zoneHigh - zoneLow;
   //========================================
   // 🔥 ENTRY PRICES
   //========================================
   double e1, e2, e3;
   if (isBuy)
   {
      e1 = zoneLow + range * 0.3;
      e2 = zoneLow + range * 0.5;
      e3 = zoneLow + range * 0.7;
   }
   else
   {
      e1 = zoneHigh - range * 0.3;
      e2 = zoneHigh - range * 0.5;
      e3 = zoneHigh - range * 0.7;
   }
   //========================================
   // 🔥 DISTANCE TO SL
   //========================================
   double d1 = MathAbs(e1 - sl);
   double d2 = MathAbs(e2 - sl);
   double d3 = MathAbs(e3 - sl);
   if (d1 <= 0 || d2 <= 0 || d3 <= 0)
      return 0;
   //========================================
   // 🔥 INVERSE WEIGHTING
   //========================================
   double w1 = 1.0 / d1;
   double w2 = 1.0 / d2;
   double w3 = 1.0 / d3;
   double totalW = w1 + w2 + w3;
   w1 /= totalW;
   w2 /= totalW;
   w3 /= totalW;
   //========================================
   // 🔥 RAW LOT
   //========================================
   double lot1 = CalculateLotRaw(e1, sl, riskMoney) * w1;
   double lot2 = CalculateLotRaw(e2, sl, riskMoney) * w2;
   double lot3 = CalculateLotRaw(e3, sl, riskMoney) * w3;
   //========================================
   // 🔥 NORMALIZE
   //========================================
   layers[0].price = e1;
   layers[0].lot = NormalizeLot(lot1);
   layers[1].price = e2;
   layers[1].lot = NormalizeLot(lot2);
   layers[2].price = e3;
   layers[2].lot = NormalizeLot(lot3);
   //========================================
   // 🔥 VOLATILITY FILTER (SMART ACTIVATION)
   //========================================
   double atrFast = GetATR(1);
   double atrSlow = GetATR(10);
   double impulse = 0;
   for (int i = 1; i <= IMPULSE_CANDLES; i++)
      impulse += MathAbs(ratesLTF[i].close - ratesLTF[i].open);
   bool strongImpulse = impulse > atrFast * IMPULSE_MIN_ATR;
   bool highVol = atrFast > atrSlow * 1.2;
   bool lowVol = atrFast < atrSlow * 0.8;
   //========================================
   // 🔥 LAYER ACTIVATION LOGIC
   //========================================
   if (highVol && strongImpulse)
   {
      // shallow entry only
      layers[0].active = true;
      layers[1].active = true;
      layers[2].active = false;
   }
   else if (lowVol)
   {
      // deep entries
      layers[0].active = false;
      layers[1].active = true;
      layers[2].active = true;
   }
   else
   {
      // hybrid
      layers[0].active = true;
      layers[1].active = true;
      layers[2].active = true;
   }
   return 3;
}

//====================================================
// 🔥 DEPLOY ORDERS (LIMIT BASED)
//====================================================
void DeployEntries(bool isBuy, EntryLayer &layers[], double sl, double tp)
{
   for (int i = 0; i < ArraySize(layers); i++)
   {
      if (!layers[i].active)
         continue;
      double price = layers[i].price;
      double lot = layers[i].lot;
      if (lot <= 0)
         continue;
      //==============================
      // BUY LIMIT
      //==============================
      if (isBuy)
      {
         if (price < SymbolInfoDouble(_Symbol, SYMBOL_ASK))
         {
            trade.BuyLimit(lot, price, _Symbol, sl, tp);
         }
      }
      else
      {
         if (price > SymbolInfoDouble(_Symbol, SYMBOL_BID))
         {
            trade.SellLimit(lot, price, _Symbol, sl, tp);
         }
      }
      if (DEBUG_SMC)
      {
         Print("ENTRY DEPLOYED | ",
               isBuy ? "BUY" : "SELL",
               " price=", price,
               " lot=", lot);
      }
   }
}

//| IS SWING HIGH                                                    |
//+------------------------------------------------------------------+
// IsSwingHigh: pivot high yang valid
// strict=true → butuh 3 candle setiap sisi (HTF-grade)
// strict=false → 2 candle (LTF lebih granular)
bool IsSwingHigh(MqlRates &r[], int i, bool strict = false)
{
   int n = strict ? 3 : 2;
   if (i < n || i + n >= ArraySize(r))
      return false;
   for (int k = 1; k <= n; k++)
   {
      // sisi kiri (lebih baru, indeks lebih kecil): harus STRICTLY lebih rendah
      if (r[i].high <= r[i - k].high)
         return false;
      // sisi kanan (lebih tua, indeks lebih besar): harus STRICTLY lebih rendah
      if (r[i].high <= r[i + k].high)
         return false;
   }
   return true;
}

// IsSwingLow: pivot low yang valid
bool IsSwingLow(MqlRates &r[], int i, bool strict = false)
{
   int n = strict ? 3 : 2;
   if (i < n || i + n >= ArraySize(r))
      return false;
   for (int k = 1; k <= n; k++)
   {
      // sisi kiri (lebih baru): harus STRICTLY lebih tinggi
      if (r[i].low >= r[i - k].low)
         return false;
      // sisi kanan (lebih tua): harus STRICTLY lebih tinggi
      if (r[i].low >= r[i + k].low)
         return false;
   }
   return true;
}

//+------------------------------------------------------------------+
//| GET PROTECTED SWINGS                                             |
//| INSTITUTIONAL STRUCTURE                                          |
//+------------------------------------------------------------------+

//+------------------------------------------------------------------+
//| GET LAST VALID SWINGS                                            |
//| USED FOR PREMIUM / DISCOUNT                                      |
//| USED FOR HTF RANGE                                               |
//+------------------------------------------------------------------+
bool GetLastSwings(
    MqlRates &r[],
    int bars,
    SwingPoint &lastHigh,
    SwingPoint &lastLow)
{
   //==================================================
   // RESET
   //==================================================
   lastHigh.index = -1;
   lastHigh.price = 0;
   lastLow.index = -1;
   lastLow.price = 0;
   //==================================================
   // VALIDATION
   //==================================================
   if (bars < 50)
      return false;
   //==================================================
   // ATR FILTER
   //==================================================
   double atr = GetATR_HTF(0);
   if (atr <= 0)
      return false;
   //==================================================
   // SEARCH LOOKBACK
   //==================================================
   int maxLookback =
       MathMin(
           80,
           bars - 10);
   //==================================================
   // FIND SWINGS
   //==================================================
   for (int i = 5; i < maxLookback; i++)
   {
      //--------------------------------------------------
      // SWING HIGH
      //--------------------------------------------------
      if (
          lastHigh.index == -1 &&
          IsSwingHigh(r, i))
      {
         double leftRange =
             MathAbs(
                 r[i].high -
                 r[i + 1].high);
         double rightRange =
             MathAbs(
                 r[i].high -
                 r[i - 1].high);
         //--------------------------------------------------
         // FILTER WEAK SWING
         //--------------------------------------------------
         if (
             leftRange > atr * 0.15 ||
             rightRange > atr * 0.15)
         {
            lastHigh.index = i;
            lastHigh.price = r[i].high;
         }
      }
      //--------------------------------------------------
      // SWING LOW
      //--------------------------------------------------
      if (
          lastLow.index == -1 &&
          IsSwingLow(r, i))
      {
         double leftRange =
             MathAbs(
                 r[i].low -
                 r[i + 1].low);
         double rightRange =
             MathAbs(
                 r[i].low -
                 r[i - 1].low);
         //--------------------------------------------------
         // FILTER WEAK SWING
         //--------------------------------------------------
         if (
             leftRange > atr * 0.15 ||
             rightRange > atr * 0.15)
         {
            lastLow.index = i;
            lastLow.price = r[i].low;
         }
      }
      //--------------------------------------------------
      // BOTH FOUND
      //--------------------------------------------------
      if (
          lastHigh.index != -1 &&
          lastLow.index != -1)
      {
         break;
      }
   }
   //==================================================
   // VALIDATION
   //==================================================
   if (
       lastHigh.index == -1 ||
       lastLow.index == -1)
   {
      return false;
   }
   //==================================================
   // RANGE VALIDATION
   //==================================================
   double range =
       MathAbs(
           lastHigh.price -
           lastLow.price);
   if (range < atr * 1.5)
      return false;
   //==================================================
   // INVALID STRUCTURE
   //==================================================
   if (lastHigh.price <= lastLow.price)
      return false;
   //==================================================
   // SUCCESS
   //==================================================
   return true;
}
//+------------------------------------------------------------------+
//| GET EXTERNAL STRUCTURE                                           |
//| MULTI PAIRS / INSTITUTIONAL FLOW                                 |
//+------------------------------------------------------------------+
bool GetExternalStructure(MarketStructure &ms)
{
   //====================================================
   // RESET
   //====================================================
   ZeroMemory(ms);
   //====================================================
   // VALIDATION
   //====================================================
   int bars = ArraySize(ratesHTF);
   if (bars < 150)
      return false;
   double atr = GetATR_HTF(0);
   if (atr <= 0)
      return false;
   //====================================================
   // SWING COLLECTION
   // Kumpulkan hingga 5 swing high dan low yang valid.
   // Array as-series: index 0 = paling baru, lebih besar = lebih lama.
   // Swing valid = memenuhi threshold signifikansi ATR-based.
   //====================================================
   double highs[10];
   int highIdx[10];
   double lows[10];
   int lowIdx[10];
   int hCount = 0;
   int lCount = 0;
   double minSwingSize = atr * 0.20; // [FIX] naikkan dari 0.15 → 0.20 untuk HTF grade
   for (int i = 4; i < 120 && i + 3 < bars; i++)
   {
      //----------------------------------------------------
      // SWING HIGH: gunakan strict=true untuk HTF
      //----------------------------------------------------
      if (IsSwingHigh(ratesHTF, i, true))
      {
         double leftDiff = ratesHTF[i].high - ratesHTF[i + 1].high;
         double rightDiff = ratesHTF[i].high - ratesHTF[i - 1].high;
         // [FIX] AND bukan OR — kedua sisi harus signifikan
         if (leftDiff > minSwingSize && rightDiff > minSwingSize)
         {
            if (hCount < 10)
            {
               highs[hCount] = ratesHTF[i].high;
               highIdx[hCount] = i;
               hCount++;
            }
         }
      }
      //----------------------------------------------------
      // SWING LOW: gunakan strict=true untuk HTF
      //----------------------------------------------------
      if (IsSwingLow(ratesHTF, i, true))
      {
         double leftDiff = ratesHTF[i + 1].low - ratesHTF[i].low;
         double rightDiff = ratesHTF[i - 1].low - ratesHTF[i].low;
         // [FIX] AND bukan OR
         if (leftDiff > minSwingSize && rightDiff > minSwingSize)
         {
            if (lCount < 10)
            {
               lows[lCount] = ratesHTF[i].low;
               lowIdx[lCount] = i;
               lCount++;
            }
         }
      }
      if (hCount >= 5 && lCount >= 5)
         break;
   }
   if (hCount < 2 || lCount < 2)
      return false;
   //====================================================
   // LAST STRUCTURE
   // high1/low1 = paling baru (index terkecil)
   // high2/low2 = sebelumnya
   //====================================================
   ms.high1 = highs[0];
   ms.high1_idx = highIdx[0];
   ms.high2 = highs[1];
   ms.high2_idx = highIdx[1];
   ms.low1 = lows[0];
   ms.low1_idx = lowIdx[0];
   ms.low2 = lows[1];
   ms.low2_idx = lowIdx[1];
   // Swing ke-3 untuk konfirmasi trend multi-swing
   double high3 = (hCount >= 3) ? highs[2] : 0;
   double low3 = (lCount >= 3) ? lows[2] : 0;
   //====================================================
   // STRUCTURE STATE (perbandingan swing berurutan)
   // HH: high terbaru > high sebelumnya → bullish push
   // HL: low terbaru > low sebelumnya   → bullish hold
   // LH: high terbaru < high sebelumnya → bearish push
   // LL: low terbaru < low sebelumnya   → bearish hold
   //====================================================
   ms.HH = (ms.high1 > ms.high2);
   ms.HL = (ms.low1 > ms.low2);
   ms.LH = (ms.high1 < ms.high2);
   ms.LL = (ms.low1 < ms.low2);
   // [FIX STRUCT-3] Konfirmasi trend multi-swing (ICT grade)
   // Bullish trend terkonfirmasi: high1 > high2 > high3 (series HH) DAN low1 > low2 (HL)
   // Satu perbandingan mudah false-positive saat sideways
   bool confirmedHH = (ms.HH && high3 > 0 && ms.high2 > high3);
   bool confirmedLL = (ms.LL && low3 > 0 && ms.low2 < low3);
   bool confirmedHL = ms.HL;
   bool confirmedLH = ms.LH;
   //====================================================
   // CURRENT PRICE (dari candle yang sudah CLOSED = index 1)
   // [FIX STRUCT-4] Gunakan candle closed, bukan live (index 0)
   // Candle live belum final — sweep/BOS dari candle live bisa invalid
   // saat candle akhirnya close di tempat berbeda
   //====================================================
   double currentClose = ratesHTF[1].close;
   double currentHigh = ratesHTF[1].high;
   double currentLow = ratesHTF[1].low;
   double currentOpen = ratesHTF[1].open;
   //====================================================
   // DEALING RANGE (dari 4 swing terakhir)
   // Ini adalah range institusional: antara extreme high
   // dan extreme low dari struktur terkini.
   //====================================================
   double dealingHigh = MathMax(ms.high1, ms.high2);
   double dealingLow = MathMin(ms.low1, ms.low2);
   double dealingRange = dealingHigh - dealingLow;
   if (dealingRange < atr * 1.0)
      return false;
   double equilibrium = (dealingHigh + dealingLow) * 0.5;
   //====================================================
   // PREMIUM / DISCOUNT
   //====================================================
   ms.premium = (currentClose > equilibrium);
   ms.discount = (currentClose < equilibrium);
   //====================================================
   // FLOW (posisi harga dalam dealing range)
   // bullishFlow = harga di atas 60% dealing range → momentum naik
   // bearishFlow = harga di bawah 40% dealing range → momentum turun
   //====================================================
   bool bullishFlow = (currentClose > dealingLow + dealingRange * 0.60);
   bool bearishFlow = (currentClose < dealingHigh - dealingRange * 0.60);
   //====================================================
   // SWEEP DETECTION (HTF level)
   // [FIX STRUCT-5] sweepBuffer 0.10→0.15, wick 0.25→0.30
   // Tambah: close harus di bawah high2 dengan margin (bukan tepat di level)
   // Tambah: rejection body harus tampak (close < midpoint candle sweep)
   //====================================================
   double sweepBuffer = atr * 0.15;
   double sweepCandleMidH = (currentHigh + currentOpen) * 0.5; // midpoint candle dari sisi atas
   double sweepCandleMidL = (currentLow + currentOpen) * 0.5;  // midpoint candle dari sisi bawah
   ms.sweepHigh =
       (currentHigh > ms.high2 + sweepBuffer &&
        currentClose < ms.high2 &&                   // close di bawah level
        (currentHigh - currentClose) > atr * 0.30 && // wick signifikan
        currentClose < sweepCandleMidH);             // close di bawah midpoint candle (rejection nyata)
   ms.sweepLow =
       (currentLow < ms.low2 - sweepBuffer &&
        currentClose > ms.low2 &&                   // close di atas level
        (currentClose - currentLow) > atr * 0.30 && // wick signifikan
        currentClose > sweepCandleMidL);            // close di atas midpoint candle
                                                    //====================================================
                                                    // HTF DISPLACEMENT (3 candle terakhir HTF)
                                                    // Scan candle dengan body besar sebagai bukti
                                                    // institutional order flow. Ambil yang paling kuat
                                                    // (bukan OR semua → hindari false flag).
                                                    //====================================================
   ms.displacementBull = false;
   ms.displacementBear = false;
   double bestBullImpulse = 0;
   double bestBearImpulse = 0;
   for (int i = 1; i <= 5; i++)
   {
      double range = ratesHTF[i].high - ratesHTF[i].low;
      double body = MathAbs(ratesHTF[i].close - ratesHTF[i].open);
      if (range <= 0)
         continue;
      double ratio = body / range;
      double impulse = range * ratio;
      if (ratesHTF[i].close > ratesHTF[i].open &&
          ratio > 0.60 && range > atr * 0.60)
      {
         if (impulse > bestBullImpulse)
         {
            bestBullImpulse = impulse;
            ms.displacementBull = true;
         }
      }
      if (ratesHTF[i].close < ratesHTF[i].open &&
          ratio > 0.60 && range > atr * 0.60)
      {
         if (impulse > bestBearImpulse)
         {
            bestBearImpulse = impulse;
            ms.displacementBear = true;
         }
      }
   }
   //====================================================
   // CHoCH / BOS (HTF level)
   // [FIX STRUCT-6] buffer 0.03→0.15 ATR (0.03 hampir nol, false positive)
   // BOS (continuation): break struktur tanpa sweep sebelumnya
   // CHoCH (reversal): break struktur DAN ada sweep sebelumnya
   // [FIX] CHoCH bullish harus break high2 SETELAH sweepLow terkonfirmasi
   // [FIX] BOS harus disertai displacement candle agar tidak false-positive
   //====================================================
   double chochBuffer = atr * 0.15;
   bool rawChochBull = (currentClose > ms.high2 + chochBuffer);
   bool rawChochBear = (currentClose < ms.low2 - chochBuffer);
   // CHoCH = break struktur setelah sweep (reversal konfirmasi)
   ms.chochBull = (rawChochBull && ms.sweepLow);
   ms.chochBear = (rawChochBear && ms.sweepHigh);
   // BOS = break struktur continuation (displacement wajib sebagai filter)
   bool bosBull = (rawChochBull && !ms.sweepLow && ms.displacementBull);
   bool bosBear = (rawChochBear && !ms.sweepHigh && ms.displacementBear);
   //====================================================
   // BULLISH STRUCTURE ENGINE
   // [FIX STRUCT-7] bullTrend menggunakan confirmedHH + HL
   // (bukan hanya HH+HL yang hanya 1 perbandingan)
   //
   // Tier 1 – Confirmed Trend: 3-swing HH series + HL
   // Tier 2 – Reversal: sweepLow + displacement + CHoCH
   // Tier 3 – Soft (BOS continuation + flow + discount zone)
   //====================================================
   bool bullTrend = (confirmedHH && confirmedHL);
   bool bullReversal = (ms.sweepLow && ms.displacementBull && ms.chochBull);
   bool bullSoft = (bosBull && bullishFlow && ms.discount); // [FIX] hapus ms.HH||ms.HL agar tidak overlap
   ms.bullish = (bullTrend || bullReversal || bullSoft);
   //====================================================
   // BEARISH STRUCTURE ENGINE (mirror bullish)
   // [FIX] confirmedLL untuk bearTrend
   //====================================================
   bool bearTrend = (confirmedLL && confirmedLH);
   bool bearReversal = (ms.sweepHigh && ms.displacementBear && ms.chochBear);
   bool bearSoft = (bosBear && bearishFlow && ms.premium);
   ms.bearish = (bearTrend || bearReversal || bearSoft);
   //====================================================
   // CONFLICT RESOLUTION (scoring system)
   // Jika kedua arah aktif, gunakan score terbobot.
   // BOS dan CHoCH lebih kuat dari flow saja.
   //====================================================
   if (ms.bullish && ms.bearish)
   {
      double bullScore = 0.0;
      double bearScore = 0.0;
      // Struktur
      if (ms.HH)
         bullScore += 2.0;
      if (ms.HL)
         bullScore += 2.0;
      if (ms.LL)
         bearScore += 2.0;
      if (ms.LH)
         bearScore += 2.0;
      // Reversal konfirmasi
      if (ms.sweepLow)
         bullScore += 1.5;
      if (ms.sweepHigh)
         bearScore += 1.5;
      if (ms.displacementBull)
         bullScore += 2.0;
      if (ms.displacementBear)
         bearScore += 2.0;
      if (ms.chochBull)
         bullScore += 3.0; // strongest signal
      if (ms.chochBear)
         bearScore += 3.0;
      // Flow & zone
      if (bullishFlow)
         bullScore += 1.0;
      if (bearishFlow)
         bearScore += 1.0;
      if (ms.discount)
         bullScore += 1.0;
      if (ms.premium)
         bearScore += 1.0;
      // BOS sebagai tie-breaker
      if (bosBull)
         bullScore += 1.5;
      if (bosBear)
         bearScore += 1.5;
      if (bullScore >= bearScore)
         ms.bearish = false;
      else
         ms.bullish = false;
   }
   //====================================================
   // STRONG BIAS (konfirmasi penuh semua level)
   // [FIX STRUCT-8] Hapus zone requirement di strong bias
   // displacement tidak selalu terjadi di zona discount/premium
   // Strong = struktur terkonfirmasi + displacement + (CHoCH atau BOS)
   //====================================================
   ms.strongBullish =
       (ms.bullish && ms.displacementBull && (ms.chochBull || bosBull) && bullishFlow);
   ms.strongBearish =
       (ms.bearish && ms.displacementBear && (ms.chochBear || bosBear) && bearishFlow);
   return true;
}

//+------------------------------------------------------------------+
//| HTF BIAS ENGINE                                                  |
//+------------------------------------------------------------------+
int DetectHTFBias()
{
   static int lastBias = 0;
   static uint lastBiasBar = 0;
   uint currentBar = (uint)Bars(_Symbol, HTF);
   //====================================================
   // CACHE: recalculate only on a new HTF bar.
   // Do NOT guard with lastBias!=0 — range (0) is a valid
   // cached result and must not cause per-tick recalculation.
   //====================================================
   if (currentBar == lastBiasBar)
      return lastBias;
   MarketStructure ext;
   if (!GetExternalStructure(ext))
   {
      // Structure unavailable this bar — hold previous bias.
      // Mark bar as done so we don't retry on every tick.
      lastBiasBar = currentBar;
      return lastBias;
   }
   //====================================================
   // BIAS FROM STRUCTURE
   //====================================================
   int newBias = 0;
   if (ext.bullish && !ext.bearish)
      newBias = 1;
   else if (ext.bearish && !ext.bullish)
      newBias = -1;
   // range (conflict) → newBias stays 0, lastBias stays sticky
   if (newBias != 0)
      lastBias = newBias;
   //====================================================
   // Always mark this bar calculated — prevents per-tick
   // recalculation regardless of whether bias changed.
   //====================================================
   lastBiasBar = currentBar;
   //====================================================
   // REGIME UPDATE (once per HTF bar)
   //====================================================
   regime = DetectMarketRegime(regimeStrength);
   //====================================================
   // DEBUG LABEL
   // Context tags are gated on the resolved lastBias direction
   // to avoid mixing bullish labels into a bearish bias string.
   //====================================================
   string biasStr = "";
   if (lastBias == 1)
   {
      biasStr = ext.strongBullish ? "STRONG BULLISH" : "BULLISH";
      if (ext.sweepLow && ext.chochBull)
         biasStr += " [REVERSAL]";
      if (ext.HH && ext.HL)
         biasStr += " [TREND]";
   }
   else if (lastBias == -1)
   {
      biasStr = ext.strongBearish ? "STRONG BEARISH" : "BEARISH";
      if (ext.sweepHigh && ext.chochBear)
         biasStr += " [REVERSAL]";
      if (ext.LL && ext.LH)
         biasStr += " [TREND]";
   }
   else
   {
      if (regime == REGIME_TREND_BULL)
      {
         biasStr = "BULLISH RANGE";

         lastBias = 1; // reset bias saat range detected, agar tidak sticky ke bias sebelumnya
      }
      else if (regime == REGIME_TREND_BEAR)
      {
         biasStr = "BEARISH RANGE";
         lastBias = -1; // reset bias saat range detected
      }
      else
      {
         lastBias = 0; // reset bias saat no clear regime
         biasStr = "RANGE";
      }
   }
   DebugSMC("HTF BIAS: " + biasStr);
   return lastBias;
}
//+------------------------------------------------------------------+
//| FINAL BIAS                                                       |
//+------------------------------------------------------------------+
int GetHTFBias()
{
   if (OverRideHTFBias == NO_TRADE)
      return 0;
   if (OverRideHTFBias == BUY_TRADE)
      return 1;
   if (OverRideHTFBias == SELL_TRADE)
      return -1;
   // AUTO_TRADE: gunakan deteksi otomatis
   return DetectHTFBias();
}

   int swingH = SwingHigh(3);
   if (swingH == -1)
      return false;

   double close1 = iClose(_Symbol, LTF, 1);
   return close1 > prevHigh;
}

//+------------------------------------------------------------------+
bool BOS_Bearish()
{
   int swingL = SwingLow(3);
   if (swingL == -1)
      return false;
   double prevLow = iLow(_Symbol, LTF, swingL);
   double close1 = iClose(_Symbol, LTF, 1);
   double atr = GetATR(1);
   double breakSize = prevLow - close1;
   if (close1 < prevLow && breakSize > atr * 0.2)
      return true;
   return false;
}

       iLow(_Symbol, LTF, shift) < iLow(_Symbol, LTF, shift - 1))
      return shift;
   return -1;
}

//+------------------------------------------------------------------+
bool IsSafeToTrade()
{
   double ask = SymbolInfoDouble(_Symbol, SYMBOL_ASK);
   double bid = SymbolInfoDouble(_Symbol, SYMBOL_BID);
   double spread = (ask - bid);
   double atr = GetATR(1);
   if (atr <= 0)
      return false;
   //========================================
   // ✔ SPREAD FILTER (ADAPTIVE)
   //========================================
   if (spread > atr * 0.25)
      return false;
   //========================================
   // ✔ AUTO TIME CONVERSION (SERVER → UTC)
   //========================================
   datetime serverTime = TimeCurrent();
   datetime gmtTime = TimeGMT();
   int offsetSeconds = (int)(serverTime - gmtTime);
   int offsetHours = offsetSeconds / 3600;
   MqlDateTime t;
   TimeToStruct(serverTime, t);
   int serverHour = t.hour;
   int utcHour = serverHour - offsetHours;
   if (utcHour < 0)
      utcHour += 24;
   if (utcHour > 23)
      utcHour -= 24;
   //========================================
   // ✔ SESSION (UTC BASED)
   //========================================
   bool london = (utcHour >= 8 && utcHour <= 12);
   bool newyork = (utcHour >= 13 && utcHour <= 17);
   bool preLondon = (utcHour == 7);
   bool preNY = (utcHour == 12);
   if (!(london || newyork || preLondon || preNY))
      return false;
   //========================================
   // ✔ VOLATILITY FILTER (reuse atr)
   //========================================
   double atrSlow = GetATR(10);
   if (atrSlow > 0 && atr < atrSlow * 0.6)
      return false;
   //========================================
   // ✔ ANTI NEWS SPIKE
   //========================================
   int ltfBars = ArraySize(ratesLTF);
   if (ltfBars >= 2)
   {
      double range = ratesLTF[1].high - ratesLTF[1].low;
      if (range > atr * 2.5)
         return false;
   }
   return true;
}
//+------------------------------------------------------------------+
//| INSTITUTIONAL AUCTION VOLUME PROFILE ENGINE                      |
//| TRUTHMODE REFACTORED VERSION                                     |
//+------------------------------------------------------------------+
bool GetVolumeProfile()
{
   MqlTick ticks[];
   //====================================================
   // RESET STATES
   //====================================================
   acceptedUp = false;
   acceptedDown = false;
   aggressiveBuy = false;
   aggressiveSell = false;
   failedAuctionUp = false;
   failedAuctionDown = false;
   buyerExhaustion = false;
   sellerExhaustion = false;
   liquiditySweepBuy = false;
   liquiditySweepSell = false;
   rotationDay = false;
   trendDay = false;
   pocMigrationUp = false;
   pocMigrationDown = false;
   double currentPrice;
   //====================================================
   // STATIC MEMORY
   //====================================================
   static double prevPOC = 0.0;
   static double prevDelta = 0.0;
   //====================================================
   // SESSION
   //====================================================
   datetime sessionStart =
       TimeCurrent() - 86400;
   //====================================================
   // COPY TICKS
   //====================================================
   int copied =
       CopyTicksRange(
           _Symbol,
           ticks,
           COPY_TICKS_VOLUME,
           (long)sessionStart * 1000,
           (long)TimeCurrent() * 1000);
   if (copied < 1000)
      return false;
   //====================================================
   // ATR STEP
   //====================================================
   double atr =
       GetATR(1);
   if (atr <= 0)
      return false;
   double tickSize =
       SymbolInfoDouble(
           _Symbol,
           SYMBOL_TRADE_TICK_SIZE);
   if (tickSize <= 0)
      tickSize = _Point;
   step =
       MathMax(
           atr * 0.05,
           tickSize * 5.0);
   //====================================================
   // PROFILE RANGE
   //====================================================
   highest =
       -DBL_MAX;
   lowest =
       DBL_MAX;
   long minBucket =
       LONG_MAX;
   long maxBucket =
       LONG_MIN;
   //====================================================
   // FIRST PASS
   //====================================================
   for (int i = 0; i < copied; i++)
   {
      double price =
          ticks[i].last;
      if (price <= 0)
         price = ticks[i].bid;
      if (price <= 0)
         continue;
      if (price > highest)
         highest = price;
      if (price < lowest)
         lowest = price;
      long bucket =
          (long)MathFloor(
              price / step);
      if (bucket < minBucket)
         minBucket = bucket;
      if (bucket > maxBucket)
         maxBucket = bucket;
   }
   if (
       minBucket == LONG_MAX ||
       maxBucket == LONG_MIN)
      return false;
   //====================================================
   // BINS
   //====================================================
   bins =
       (int)(maxBucket -
             minBucket + 1);
   if (bins <= 10)
      return false;
   while (bins > 2500)
   {
      step *= 1.5;
      minBucket = LONG_MAX;
      maxBucket = LONG_MIN;
      for (int i = 0; i < copied; i++)
      {
         double price = ticks[i].last;
         if (price <= 0)
            price = ticks[i].bid;
         if (price <= 0)
            continue;
         long bucket = (long)MathFloor(price / step);
         if (bucket < minBucket)
            minBucket = bucket;
         if (bucket > maxBucket)
            maxBucket = bucket;
      }
      if (minBucket == LONG_MAX || maxBucket == LONG_MIN)
         break;
      bins = (int)(maxBucket - minBucket + 1);
   }
   //====================================================
   // ARRAYS
   //====================================================
   ArrayResize(volumeX, bins);
   static double buyVolume[];
   static double sellVolume[];
   static double deltaFlow[];
   static double timeFlow[];
   ArrayResize(buyVolume, bins);
   ArrayResize(sellVolume, bins);
   ArrayResize(deltaFlow, bins);
   ArrayResize(timeFlow, bins);
   ArrayInitialize(volumeX, 0.0);
   ArrayInitialize(buyVolume, 0.0);
   ArrayInitialize(sellVolume, 0.0);
   ArrayInitialize(deltaFlow, 0.0);
   ArrayInitialize(timeFlow, 0.0);
   //====================================================
   // DISTRIBUTION
   //====================================================
   double totalVol = 0.0;
   double localDelta = 0.0;
   int directionalChanges = 0;
   double lastPrice = 0.0;
   int lastDirection = 0; // +1=up, -1=down, 0=flat
                          //====================================================
                          // SINGLE PASS AUCTION ENGINE
                          //====================================================
   for (int i = 0; i < copied; i++)
   {
      double price =
          ticks[i].last;
      if (price <= 0)
         price = ticks[i].bid;
      if (price <= 0)
         continue;
      //----------------------------------------------------
      // VOLUME
      //----------------------------------------------------
      double vol =
          ticks[i].volume_real;
      if (vol <= 0)
         vol =
             (double)ticks[i].volume;
      if (vol <= 0)
         vol = 1.0;
      //----------------------------------------------------
      // BUCKET
      //----------------------------------------------------
      long bucket =
          (long)MathFloor(
              price / step);
      int idx =
          (int)(bucket -
                minBucket);
      if (
          idx < 0 ||
          idx >= bins)
         continue;
      //----------------------------------------------------
      // DISTRIBUTION
      //----------------------------------------------------
      volumeX[idx] += vol;
      timeFlow[idx] += 1.0;
      //----------------------------------------------------
      // TRUE AGGRESSIVE DELTA
      //----------------------------------------------------
      bool aggressiveBuyer =
          (ticks[i].flags &
           TICK_FLAG_BUY);
      bool aggressiveSeller =
          (ticks[i].flags &
           TICK_FLAG_SELL);
      if (aggressiveBuyer)
      {
         buyVolume[idx] += vol;
         deltaFlow[idx] += vol;
         localDelta += vol;
      }
      if (aggressiveSeller)
      {
         sellVolume[idx] += vol;
         deltaFlow[idx] -= vol;
         localDelta -= vol;
      }
      //----------------------------------------------------
      // ROTATION FACTOR (true direction CHANGE counter)
      //----------------------------------------------------
      if (lastPrice > 0)
      {
         int dir = (price > lastPrice) ? 1 : (price < lastPrice) ? -1
                                                                 : 0;
         if (dir != 0)
         {
            // count only when direction actually reverses
            if (lastDirection != 0 && dir != lastDirection)
               directionalChanges++;
            lastDirection = dir;
         }
      }
      lastPrice = price;
      totalVol += vol;
   }
   if (totalVol <= 0)
      return false;
   //====================================================
   // SMOOTHED POC
   //====================================================
   int pocIdx =
       0;
   double bestPOC =
       -DBL_MAX;
   for (int i = 1; i < bins - 1; i++)
   {
      double smoothVol =
          (volumeX[i - 1] * 0.50 +
           volumeX[i] * 1.00 +
           volumeX[i + 1] * 0.50);
      if (smoothVol > bestPOC)
      {
         bestPOC =
             smoothVol;
         pocIdx =
             i;
      }
   }
   poc =
       (double)(minBucket +
                pocIdx) *
       step;
   //====================================================
   // AUCTION VALUE AREA
   //====================================================
   static bool included[];
   ArrayResize(included, bins);
   ArrayInitialize(included, false);
   included[pocIdx] =
       true;
   double targetVol =
       totalVol * 0.70;
   double cumulative =
       volumeX[pocIdx];
   while (cumulative < targetVol)
   {
      int bestIdx =
          -1;
      double bestVol =
          -1;
      for (int i = 0; i < bins; i++)
      {
         if (included[i])
            continue;
         bool adjacent =
             false;
         if (i > 0 && included[i - 1])
            adjacent = true;
         if (i < bins - 1 && included[i + 1])
            adjacent = true;
         if (!adjacent)
            continue;
         double participation =
             volumeX[i];
         if (participation > bestVol)
         {
            bestVol =
                participation;
            bestIdx =
                i;
         }
      }
      if (bestIdx < 0)
         break;
      included[bestIdx] =
          true;
      cumulative +=
          volumeX[bestIdx];
   }
   //====================================================
   // VALUE AREA
   //====================================================
   int lowIdx = bins;
   int highIdx = 0;
   for (int i = 0; i < bins; i++)
   {
      if (included[i])
      {
         if (i < lowIdx)
            lowIdx = i;
         if (i > highIdx)
            highIdx = i;
      }
   }
   vah =
       (double)(minBucket +
                highIdx) *
       step;
   val =
       (double)(minBucket +
                lowIdx) *
       step;
   //====================================================
   // CURRENT PRICE
   //====================================================
   currentPrice =
       SymbolInfoDouble(
           _Symbol,
           SYMBOL_BID);
   //====================================================
   // PROFILE SKEWNESS
   //====================================================
   double upperVol =
       0.0;
   double lowerVol =
       0.0;
   for (int i = pocIdx + 1; i < bins; i++)
      upperVol += volumeX[i];
   for (int i = 0; i < pocIdx; i++)
      lowerVol += volumeX[i];
   double skewness =
       (upperVol -
        lowerVol) /
       totalVol;
   //====================================================
   // PROFILE SHAPE
   //====================================================
   profileShape =
       "D";
   if (skewness > 0.20)
      profileShape = "P";
   if (skewness < -0.20)
      profileShape = "b";
   //====================================================
   // ACCEPTANCE ENGINE
   //====================================================
   int outsideTicksUp = 0;
   int outsideTicksDown = 0;
   double outsideVolUp = 0.0;
   double outsideVolDown = 0.0;
   for (int i = 0; i < bins; i++)
   {
      double price =
          (minBucket + i) * step;
      if (price > vah)
      {
         outsideTicksUp +=
             (int)timeFlow[i];
         outsideVolUp +=
             volumeX[i];
      }
      if (price < val)
      {
         outsideTicksDown +=
             (int)timeFlow[i];
         outsideVolDown +=
             volumeX[i];
      }
   }
   double outsideRatioUp =
       outsideVolUp / totalVol;
   double outsideRatioDown =
       outsideVolDown / totalVol;
   //====================================================
   // TRUE ACCEPTANCE
   // Requires: price OUTSIDE value area + meaningful volume
   // outside + delta confirms direction + minimum tick count
   // to avoid fake breakouts on single spike.
   // outsideTicksUp/Down are the correctly computed local counts.
   //====================================================
   acceptedUp =
       (currentPrice > vah &&
        outsideRatioUp > 0.15 &&
        localDelta > 0 &&
        outsideTicksUp > 20);
   acceptedDown =
       (currentPrice < val &&
        outsideRatioDown > 0.15 &&
        localDelta < 0 &&
        outsideTicksDown > 20);
   //====================================================
   // AGGRESSION
   //====================================================
   aggressiveBuy =
       (acceptedUp &&
        localDelta > prevDelta &&
        profileShape == "P");
   aggressiveSell =
       (acceptedDown &&
        localDelta < prevDelta &&
        profileShape == "b");
   //====================================================
   // FAILED AUCTION
   //====================================================
   failedAuctionUp =
       (highest > vah &&
        currentPrice < vah &&
        outsideRatioUp < 0.08 &&
        localDelta < prevDelta);
   failedAuctionDown =
       (lowest < val &&
        currentPrice > val &&
        outsideRatioDown < 0.08 &&
        localDelta > prevDelta);
   //====================================================
   // EXHAUSTION
   //====================================================
   buyerExhaustion =
       (currentPrice > vah &&
        localDelta < prevDelta);
   sellerExhaustion =
       (currentPrice < val &&
        localDelta > prevDelta);
   //====================================================
   // ROTATION FACTOR
   //====================================================
   double rotationFactor =
       (double)directionalChanges /
       (double)copied;
   // After fix: directionalChanges = actual direction reversals
   // High reversals (>15% of ticks) = rotation/chop
   // Low reversals (<5% of ticks) = trending flow
   rotationDay =
       (rotationFactor > 0.15 &&
        profileShape == "D");
   trendDay =
       (rotationFactor < 0.05 &&
        (profileShape == "P" ||
         profileShape == "b"));
   //====================================================
   // POC MIGRATION
   //====================================================
   if (prevPOC > 0)
   {
      if (poc > prevPOC)
         pocMigrationUp = true;
      if (poc < prevPOC)
         pocMigrationDown = true;
   }
   //====================================================
   // HVN / LVN
   //====================================================
   hvn = poc;
   lvn =
       (vah + val) * 0.5;
   double avgVol =
       totalVol / bins;
   double bestLVN =
       DBL_MAX;
   for (int i = 3; i < bins - 3; i++)
   {
      bool valley =
          (volumeX[i] <
               volumeX[i - 1] &&
           volumeX[i] <
               volumeX[i + 1]);
      if (!valley)
         continue;
      bool rejection =
          (volumeX[i - 1] >
               avgVol * 1.5 &&
           volumeX[i + 1] >
               avgVol * 1.5);
      if (!rejection)
         continue;
      if (volumeX[i] < bestLVN)
      {
         bestLVN =
             volumeX[i];
         lvn =
             (double)(minBucket + i) * step;
      }
   }
   //====================================================
   // LIQUIDITY SWEEP
   //====================================================
   double PDH =
       iHigh(
           _Symbol,
           PERIOD_D1,
           1);
   double PDL =
       iLow(
           _Symbol,
           PERIOD_D1,
           1);
   // Liquidity sweep confirmed by failed auction (thin volume above/below)
   liquiditySweepSell =
       (highest > PDH &&
        currentPrice < PDH &&
        failedAuctionUp // confirms thin volume above
       );
   liquiditySweepBuy =
       (lowest < PDL &&
        currentPrice > PDL &&
        failedAuctionDown // confirms thin volume below
       );
   //====================================================
   // NORMALIZE
   //====================================================
   poc =
       NormalizeDouble(
           poc,
           _Digits);
   vah =
       NormalizeDouble(
           vah,
           _Digits);
   val =
       NormalizeDouble(
           val,
           _Digits);
   hvn =
       NormalizeDouble(
           hvn,
           _Digits);
   lvn =
       NormalizeDouble(
           lvn,
           _Digits);
   //====================================================
   // SAVE MEMORY
   //====================================================
   prevPOC = poc;
   prevDelta = localDelta;
   totalDelta = localDelta;
   //====================================================
   // FINAL VALIDATION
   //====================================================
   if (vah <= val)
      return false;
   if (poc <= 0)
      return false;
   return true;
}

bool GetOrderBlock(bool bullish, double &low, double &high)
{
   // [FIX] Perluas scan dari 3-10 ke 3-20 candle
   // [FIX] Tambah minimum body ratio (bukan doji)
   // [FIX] Konfirmasi: close ATAU open (engulfing)
   for (int i = 3; i < 20; i++)
   {
      double open = iOpen(_Symbol, LTF, i);
      double close = iClose(_Symbol, LTF, i);
      double hi = iHigh(_Symbol, LTF, i);
      double lo = iLow(_Symbol, LTF, i);
      double range = hi - lo;
      if (range <= 0)
         continue;
      double body = MathAbs(close - open);
      // Minimum body 25% dari range — filter doji
      if (body / range < 0.25)
         continue;
      if (bullish && close < open) // bearish candle before bull impulse
      {
         double nextClose = iClose(_Symbol, LTF, i - 1);
         double nextOpen = iOpen(_Symbol, LTF, i - 1);
         if (nextClose > hi || nextOpen > hi)
         {
            low = lo;
            high = hi;
            return true;
         }
      }
      if (!bullish && close > open) // bullish candle before bear impulse
      {
         double nextClose = iClose(_Symbol, LTF, i - 1);
         double nextOpen = iOpen(_Symbol, LTF, i - 1);
         if (nextClose < lo || nextOpen < lo)
         {
            low = lo;
            high = hi;
            return true;
         }
      }
   }
   return false;
}

//+------------------------------------------------------------------+
//|                                                                  |
//+------------------------------------------------------------------+
bool ConfirmBullishCandle(int shift)
{
   double open = iOpen(_Symbol, LTF, shift);
   double close = iClose(_Symbol, LTF, shift);
   double open1 = iOpen(_Symbol, LTF, shift + 1);
   double close1 = iClose(_Symbol, LTF, shift + 1);
   double high = iHigh(_Symbol, LTF, shift);
   double low = iLow(_Symbol, LTF, shift);
   if (close1 < open1)
   {
      return false;
   }
   double range = high - low;
   if (range <= _Point)
      return false;
   double body = MathAbs(close - open);
   // 1. Direction
   if (close <= open)
      return false;
   // 2. Minimum momentum (lebih fleksibel)
   if (body / range < 0.25)
      return false;
   // 3. Close position (tidak harus extreme, tapi tetap kuat)
   if ((high - close) > range * 0.5)
      return false;
   // 4. Expansion vs previous candle
   double prevRange = iHigh(_Symbol, LTF, shift + 1) - iLow(_Symbol, LTF, shift + 1);
   if (range < prevRange * 0.6)
      return false;
   return true;
}
//+------------------------------------------------------------------+
//|                                                                  |
//+------------------------------------------------------------------+
bool ConfirmBearishCandle(int shift)
{

   double open = iOpen(_Symbol, LTF, shift);
   double close = iClose(_Symbol, LTF, shift);
   double open1 = iOpen(_Symbol, LTF, shift + 1);
   double close1 = iClose(_Symbol, LTF, shift + 1);

   double high = iHigh(_Symbol, LTF, shift);
   double low = iLow(_Symbol, LTF, shift);
   if (close1 > open1)
   {
      return false;
   }
   double range = high - low;
   double prevRange = iHigh(_Symbol, LTF, shift + 1) - iLow(_Symbol, LTF, shift + 1);
   double body = MathAbs(close - open);
   if (range <= _Point)
      return false;
   if (close >= open)
      return false;
   if (body / range < 0.25)
      return false;
   if ((close - low) > range * 0.5)
      return false;
   if (range < (prevRange * 0.6))
      return false;
   return true;
}

//+------------------------------------------------------------------+
bool IsPriceInZone(bool isBuy, SMCEvent &e, double price, double atr)
{
   double low, high;
   // =============================
   // PRIORITY: FVG > OB
   // =============================
   if (e.hasFVG)
   {
      low = e.fvgBottom;
      high = e.fvgTop;
      // FVG → ketat
      if (price >= low && price <= high)
         return true;
   }
   else
   {
      low = e.obLow;
      high = e.obHigh;
      // OB → pakai buffer
      double buffer = atr * 0.2;
      if (price >= (low - buffer) &&
          price <= (high + buffer))
         return true;
   }
   return false;
}
//+------------------------------------------------------------------+
//|                                                                  |
//+------------------------------------------------------------------+
bool IsInDiscountZone(bool isBuy, double low, double high, double price)
{
   double mid = (low + high) * 0.5;
   if (isBuy)
      return (price <= mid); // buy di bawah
   else
      return (price >= mid); // sell di atas
}

double FindNextHigh(int start, int lookahead = 30)
{
   double highest = iHigh(_Symbol, LTF, start);
   for (int i = start; i < start + lookahead; i++)
   {
      double h = iHigh(_Symbol, LTF, i);
      if (h > highest)
         highest = h;
   }
   return highest;
}

//+------------------------------------------------------------------+
//|                                                                  |
//+------------------------------------------------------------------+
double FindNextLow(int start, int lookahead = 30)
{
   double lowest = iLow(_Symbol, LTF, start);
   for (int i = start; i < start + lookahead; i++)
   {
      double l = iLow(_Symbol, LTF, i);
      if (l < lowest)
         lowest = l;
   }
   return lowest;
}
//+------------------------------------------------------------------+
//|                                                                  |
//+------------------------------------------------------------------+
double ComputeBuySL(SMCEvent &e, double atr)
{
   double sl;
   if (e.hasFVG)
      sl = e.fvgBottom;
   else
      sl = MathMin(e.obLow, iLow(_Symbol, LTF, e.sweepShift));
   return sl - atr * 0.3;
}
//+------------------------------------------------------------------+
//|                                                                  |
//+------------------------------------------------------------------+
double ComputeSellSL(SMCEvent &e, double atr)
{
   double sl;
   if (e.hasFVG)
      sl = e.fvgTop;
   else
      sl = MathMax(e.obHigh, iHigh(_Symbol, LTF, e.sweepShift));
   return sl + atr * 0.3;
}

//+------------------------------------------------------------------+
//|                                                                  |
//+------------------------------------------------------------------+
double ComputeSellTP(double entry, double sl)
{
   double rrTP = entry - (sl - entry) * 2.5;
   double liqTP = FindNextLow(1, 10);
   if (liqTP >= entry || MathAbs(entry - liqTP) < (sl - entry) * 0.5)
      return rrTP;
   return MathMin(rrTP, liqTP);
}
//+------------------------------------------------------------------+
//|                                                                  |
//+------------------------------------------------------------------+
double ComputeBuyTP(double entry, double sl)
{
   double rrTP = entry + (entry - sl) * 2.5;
   double liqTP = FindNextHigh(1, 10);
   if (liqTP <= entry || MathAbs(liqTP - entry) < (entry - sl) * 0.5)
      return rrTP;
   return MathMax(rrTP, liqTP);
}

void ManageBE()
{
   if (TotalTrade() == 0)
      return;
   double atr = GetATR(1);
   // 🔥 buffer berbasis ATR (multi-pair safe)
   double beBuffer = atr * 0.1;
   for (int i = 0; i < PositionsTotal(); i++)
   {
      ulong ticket = PositionGetTicket(i);
      if (!PositionSelectByTicket(ticket))
         continue;
      if (PositionGetString(POSITION_SYMBOL) != _Symbol)
         continue;
      double entry = PositionGetDouble(POSITION_PRICE_OPEN);
      double sl = PositionGetDouble(POSITION_SL);
      double tp = PositionGetDouble(POSITION_TP);
      double price = PositionGetDouble(POSITION_PRICE_CURRENT);
      double risk = MathAbs(entry - sl);
      if (risk <= 0)
         continue;
      int type = (int)PositionGetInteger(POSITION_TYPE);
      //========================================
      // BUY
      //========================================
      if (type == POSITION_TYPE_BUY)
      {
         // sudah BE atau lebih baik
         if (sl >= entry)
            continue;
         // profit >= 1R
         if (price - entry >= risk)
         {
            double newSL = entry + beBuffer;
            if (newSL > sl)
               trade.PositionModify(ticket, newSL, tp);
         }
      }
      //========================================
      // SELL
      //========================================
      if (type == POSITION_TYPE_SELL)
      {
         if (sl <= entry && sl != 0)
            continue;
         if (entry - price >= risk)
         {
            double newSL = entry - beBuffer;
            if (newSL < sl || sl == 0)
               trade.PositionModify(ticket, newSL, tp);
         }
      }
   }
}

void ManagePendingOrders()
{
   for (int i = OrdersTotal() - 1; i >= 0; i--)
   {
      ulong ticket = OrderGetTicket(i);
      if (!OrderSelect(ticket))
         continue;
      if (OrderGetString(ORDER_SYMBOL) != _Symbol)
         continue;
      ENUM_ORDER_TYPE type = (ENUM_ORDER_TYPE)OrderGetInteger(ORDER_TYPE);
      if (type != ORDER_TYPE_BUY_LIMIT && type != ORDER_TYPE_SELL_LIMIT)
         continue;
      datetime orderTime = (datetime)OrderGetInteger(ORDER_TIME_SETUP);
      int barsPassed = iBarShift(_Symbol, _Period, orderTime);
      //========================================
      // 1. EXPIRY
      //========================================
      if (barsPassed > PendingExpiryBars)
      {
         trade.OrderDelete(ticket);
         continue;
      }
      //========================================
      // 2. INVALIDATION (STRUCTURE BREAK)
      //========================================
      if (UseInvalidation)
      {
         double atr = GetATR(1);
         double entry = OrderGetDouble(ORDER_PRICE_OPEN);
         double current = SymbolInfoDouble(_Symbol, SYMBOL_BID);
         // BUY LIMIT invalid
         if (type == ORDER_TYPE_BUY_LIMIT)
         {
            if (current < entry - atr * InvalidationATR)
            {
               trade.OrderDelete(ticket);
               continue;
            }
         }
         // SELL LIMIT invalid
         if (type == ORDER_TYPE_SELL_LIMIT)
         {
            if (current > entry + atr * InvalidationATR)
            {
               trade.OrderDelete(ticket);
               continue;
            }
         }
      }
   }
}

//| SMART EXIT ENGINE                                                |
//+------------------------------------------------------------------+
//+------------------------------------------------------------------+
//| MANAGE SMART EXIT                                                |
//| FULL REFACTOR - INSTITUTIONAL EXIT ENGINE                        |
//+------------------------------------------------------------------+
void ManageSmartExit()
{
   if (TotalTrade() == 0)
      return;
   //====================================================
   // ATR
   //====================================================
   double atr = GetATR(1);
   if (atr <= 0)
      return;
   //====================================================
   // BROKER RULES
   //====================================================
   double stopLevel =
       SymbolInfoInteger(
           _Symbol,
           SYMBOL_TRADE_STOPS_LEVEL) *
       _Point;
   double freezeLevel =
       SymbolInfoInteger(
           _Symbol,
           SYMBOL_TRADE_FREEZE_LEVEL) *
       _Point;
   double minLot =
       SymbolInfoDouble(
           _Symbol,
           SYMBOL_VOLUME_MIN);
   double lotStep =
       SymbolInfoDouble(
           _Symbol,
           SYMBOL_VOLUME_STEP);
   //====================================================
   // LOOP POSITIONS
   //====================================================
   for (int i = PositionsTotal() - 1; i >= 0; i--)
   {
      ulong ticket =
          PositionGetTicket(i);
      if (!PositionSelectByTicket(ticket))
         continue;
      if (
          PositionGetString(POSITION_SYMBOL) != _Symbol)
         continue;
      //====================================================
      // POSITION DATA
      //====================================================
      bool isBuy =
          (PositionGetInteger(POSITION_TYPE) ==
           POSITION_TYPE_BUY);
      double entry =
          PositionGetDouble(
              POSITION_PRICE_OPEN);
      double sl =
          PositionGetDouble(
              POSITION_SL);
      double tp =
          PositionGetDouble(
              POSITION_TP);
      double price =
          PositionGetDouble(
              POSITION_PRICE_CURRENT);
      double vol =
          PositionGetDouble(
              POSITION_VOLUME);
      //====================================================
      // REBUILD EVENT
      //====================================================
      // Use cached event to avoid rebuilding on every tick
      SMCEvent e = isBuy ? CachedBullEvent : CachedBearEvent;
      if (!e.valid)
         continue;
      //====================================================
      // DEALING RANGE
      //====================================================
      double dealingLow = 0;
      double dealingHigh = 0;
      LiquidityLevel extLevels[];
      if (isBuy)
      {
         dealingLow =
             e.liquidity;
         int extCount =
             DetectLiquidityLevels(
                 extLevels,
                 true);
         for (int j = 0; j < extCount; j++)
         {
            if (extLevels[j].swept)
               continue;
            if (extLevels[j].price > price)
            {
               dealingHigh =
                   extLevels[j].price;
               break;
            }
         }
      }
      else
      {
         dealingHigh =
             e.liquidity;
         int extCount =
             DetectLiquidityLevels(
                 extLevels,
                 false);
         for (int j = 0; j < extCount; j++)
         {
            if (extLevels[j].swept)
               continue;
            if (extLevels[j].price < price)
            {
               dealingLow =
                   extLevels[j].price;
               break;
            }
         }
      }
      if (
          dealingLow <= 0 ||
          dealingHigh <= 0)
         continue;
      //====================================================
      // EQUILIBRIUM
      //====================================================
      double equilibrium =
          (dealingLow +
           dealingHigh) *
          0.5;
      bool inPremium =
          (price > equilibrium);
      bool inDiscount =
          (price < equilibrium);
      //====================================================
      // DRAW DEALING RANGE
      //====================================================
      DrawLine(
          "EQ_" + (string)ticket,
          equilibrium,
          clrGold);
      DrawLine(
          "DL_" + (string)ticket,
          dealingLow,
          clrDarkSlateBlue);
      DrawLine(
          "DH_" + (string)ticket,
          dealingHigh,
          clrDarkSlateBlue);
      //====================================================
      // LIQUIDITY TARGETS
      //====================================================
      double tp1 = 0;
      double tp2 = 0;
      double tp3 = 0;
      LiquidityLevel levels[];
      int lvlCount =
          DetectLiquidityLevels(
              levels,
              isBuy);
      for (int j = 0; j < lvlCount; j++)
      {
         if (levels[j].swept)
            continue;
         double lvl =
             levels[j].price;
         double distance =
             MathAbs(
                 lvl - price);
         if (distance < atr * 0.5)
            continue;
         //------------------------------------------------
         // BUY
         //------------------------------------------------
         if (
             isBuy &&
             levels[j].isHigh &&
             lvl > price)
         {
            if (tp1 == 0)
               tp1 = lvl;
            else if (tp2 == 0)
               tp2 = lvl;
            else if (tp3 == 0)
            {
               tp3 = lvl;
               break;
            }
         }
         //------------------------------------------------
         // SELL
         //------------------------------------------------
         if (
             !isBuy &&
             !levels[j].isHigh &&
             lvl < price)
         {
            if (tp1 == 0)
               tp1 = lvl;
            else if (tp2 == 0)
               tp2 = lvl;
            else if (tp3 == 0)
            {
               tp3 = lvl;
               break;
            }
         }
      }
      //====================================================
      // DRAW TP
      //====================================================
      if (tp1 > 0)
         DrawLine(
             "TP1_" + (string)ticket,
             tp1,
             clrLime);
      if (tp2 > 0)
         DrawLine(
             "TP2_" + (string)ticket,
             tp2,
             clrGreen);
      if (tp3 > 0)
         DrawLine(
             "TP3_" + (string)ticket,
             tp3,
             clrDarkGreen);
      //====================================================
      // REACCELERATION
      //====================================================
      bool reAcceleration = false;
      double impulse = 0;
      int directional = 0;
      for (int k = 0; k < 4; k++)
      {
         double body =
             ratesLTF[k].close -
             ratesLTF[k].open;
         impulse +=
             MathAbs(body);
         if (isBuy && body > 0)
            directional++;
         if (!isBuy && body < 0)
            directional++;
      }
      bool strongImpulse =
          (impulse > atr * 1.0);
      bool directionalFlow =
          (directional >= 3);
      if (
          strongImpulse &&
          directionalFlow)
      {
         reAcceleration = true;
      }
      //====================================================
      // EXIT STAGE
      //====================================================
      int stage =
          GetExitStage(ticket);
      //====================================================
      // STRUCTURAL SWING
      //====================================================
      double structuralHL = 0;
      double structuralLH = 0;
      if (isBuy)
      {
         for (int k = 3; k < 20; k++)
         {
            bool validHL =
                (ratesLTF[k].low <
                     ratesLTF[k - 1].low &&
                 ratesLTF[k].low <
                     ratesLTF[k + 1].low);
            double left =
                MathAbs(
                    ratesLTF[k].low -
                    ratesLTF[k + 1].low);
            double right =
                MathAbs(
                    ratesLTF[k].low -
                    ratesLTF[k - 1].low);
            bool significant =
                (left > atr * 0.15 ||
                 right > atr * 0.15);
            if (validHL && significant)
            {
               structuralHL =
                   ratesLTF[k].low;
               DrawLine(
                   "TRAIL_HL_" + (string)ticket,
                   structuralHL,
                   clrDeepSkyBlue);
               break;
            }
         }
      }
      else
      {
         for (int k = 3; k < 20; k++)
         {
            bool validLH =
                (ratesLTF[k].high >
                     ratesLTF[k - 1].high &&
                 ratesLTF[k].high >
                     ratesLTF[k + 1].high);
            double left =
                MathAbs(
                    ratesLTF[k].high -
                    ratesLTF[k + 1].high);
            double right =
                MathAbs(
                    ratesLTF[k].high -
                    ratesLTF[k - 1].high);
            bool significant =
                (left > atr * 0.15 ||
                 right > atr * 0.15);
            if (validLH && significant)
            {
               structuralLH =
                   ratesLTF[k].high;
               DrawLine(
                   "TRAIL_LH_" + (string)ticket,
                   structuralLH,
                   clrTomato);
               break;
            }
         }
      }
      //====================================================
      // ADVANCED TRAILING ENGINE
      //====================================================
      double finalSL = sl;
      double finalTP = tp;
      bool needModify = false;
      //====================================================
      // STAGE 1 → BREAK EVEN
      //====================================================
      if (stage == 1)
      {
         double be =
             (isBuy
                  ? entry + atr * 0.05
                  : entry - atr * 0.05);
         bool improve =
             (isBuy
                  ? be > sl
                  : be < sl || sl == 0);
         if (improve)
         {
            finalSL =
                NormalizeDouble(
                    be,
                    _Digits);
            needModify = true;
         }
      }
      //====================================================
      // STAGE 2 → LOCK PROFIT
      //====================================================
      if (stage == 2)
      {
         double lockSL =
             (isBuy
                  ? tp1 - atr * 0.15
                  : tp1 + atr * 0.15);
         bool improve =
             (isBuy
                  ? lockSL > sl
                  : lockSL < sl || sl == 0);
         if (improve)
         {
            finalSL =
                NormalizeDouble(
                    lockSL,
                    _Digits);
            needModify = true;
         }
      }
      //====================================================
      // STAGE 3 → RUNNER MODE
      //====================================================
      if (stage >= 3)
      {
         //------------------------------------------------
         // BUY
         //------------------------------------------------
         if (isBuy && structuralHL > 0)
         {
            double runnerSL =
                structuralHL - atr * 0.12;
            bool improve =
                (runnerSL > sl);
            if (improve)
            {
               finalSL =
                   NormalizeDouble(
                       runnerSL,
                       _Digits);
               needModify = true;
            }
         }
         //------------------------------------------------
         // SELL
         //------------------------------------------------
         if (!isBuy && structuralLH > 0)
         {
            double runnerSL =
                structuralLH + atr * 0.12;
            bool improve =
                (runnerSL < sl ||
                 sl == 0);
            if (improve)
            {
               finalSL =
                   NormalizeDouble(
                       runnerSL,
                       _Digits);
               needModify = true;
            }
         }
      }
      //====================================================
      // FVG PROTECTION
      //====================================================
      if (stage >= 3)
      {
         if (isBuy && e.hasFVG)
         {
            double fvgSL =
                e.fvgBottom - atr * 0.05;
            if (fvgSL > finalSL)
            {
               finalSL =
                   NormalizeDouble(
                       fvgSL,
                       _Digits);
               needModify = true;
            }
         }
         if (!isBuy && e.hasFVG)
         {
            double fvgSL =
                e.fvgTop + atr * 0.05;
            if (
                fvgSL < finalSL ||
                finalSL == 0)
            {
               finalSL =
                   NormalizeDouble(
                       fvgSL,
                       _Digits);
               needModify = true;
            }
         }
      }
      //====================================================
      // SAFE MODIFY
      //====================================================
      bool safeDistance =
          (MathAbs(
               price - finalSL) >
           MathMax(
               freezeLevel,
               stopLevel));
      bool meaningful =
          (MathAbs(
               finalSL - sl) >
           atr * 0.08);
      //====================================================
      // FINAL MODIFY
      //====================================================
      if (
          needModify &&
          safeDistance &&
          meaningful)
      {
         bool modified =
             trade.PositionModify(
                 ticket,
                 finalSL,
                 finalTP);
         if (modified)
         {
            DrawSMCDebug(
                "TRAIL_UPDATE",
                TimeCurrent(),
                finalSL,
                clrGold,
                240);
         }
      }
      //====================================================
      // PARTIAL CLOSE TP1
      //====================================================
      double vol1 =
          MathFloor(
              (vol * 0.25) / lotStep) *
          lotStep;
      if (vol1 < minLot)
         vol1 = 0;
      bool hitTP1 = false;
      if (tp1 > 0)
      {
         hitTP1 =
             (isBuy
                  ? price >= tp1 - atr * 0.1
                  : price <= tp1 + atr * 0.1);
         if (hitTP1 && vol1 > 0)
         {
            trade.PositionClosePartial(
                ticket,
                vol1);
         }
      }
      //====================================================
      // STAGE UPDATE
      //====================================================
      if (
          tp1 > 0 &&
          stage < 1)
      {
         bool breakTP1 =
             (isBuy
                  ? ratesLTF[0].close >
                        tp1 + atr * 0.05
                  : ratesLTF[0].close <
                        tp1 - atr * 0.05);
         if (breakTP1)
         {
            SetExitStage(
                ticket,
                1);
         }
      }
      if (
          tp2 > 0 &&
          stage < 2)
      {
         bool breakTP2 =
             (isBuy
                  ? ratesLTF[0].close >
                        tp2 + atr * 0.05
                  : ratesLTF[0].close <
                        tp2 - atr * 0.05);
         if (breakTP2)
         {
            SetExitStage(
                ticket,
                2);
         }
      }
      if (
          tp3 > 0 &&
          stage < 3)
      {
         bool breakTP3 =
             (isBuy
                  ? ratesLTF[0].close >
                        tp3 + atr * 0.05
                  : ratesLTF[0].close <
                        tp3 - atr * 0.05);
         if (breakTP3)
         {
            SetExitStage(
                ticket,
                3);
         }
      }
      //====================================================
      // EMERGENCY EXIT
      //====================================================
      bool emergencyExit = false;
      if (
          isBuy &&
          structuralHL > 0)
      {
         if (
             ratesLTF[0].close <
             structuralHL - atr * 0.10)
         {
            emergencyExit = true;
         }
      }
      if (
          !isBuy &&
          structuralLH > 0)
      {
         if (
             ratesLTF[0].close >
             structuralLH + atr * 0.10)
         {
            emergencyExit = true;
         }
      }
      if (
          isBuy &&
          inPremium &&
          !reAcceleration &&
          stage >= 2)
      {
         emergencyExit = true;
      }
      if (
          !isBuy &&
          inDiscount &&
          !reAcceleration &&
          stage >= 2)
      {
         emergencyExit = true;
      }
      //====================================================
      // FINAL EXIT
      //====================================================
      if (emergencyExit)
      {
         bool closed =
             trade.PositionClose(ticket);
         if (closed)
         {
            DrawSMCDebug(
                "SMART_EXIT",
                TimeCurrent(),
                price,
                clrRed,
                251);
         }
      }
   }
}

bool LiquidityClusterLow(int lookback = 15, double tolerance = 10)
{
   int bars = ArraySize(ratesLTF);
   if (bars < lookback + 5)
      return false;
   double atrTol = GetATR(1);
   double tol = (atrTol > 0) ? atrTol * EQ_TOLERANCE_ATR : tolerance * _Point;
   double ref = ratesLTF[2].low;
   int count = 0;
   for (int i = 2; i < lookback && i < bars; i++)
   {
      if (MathAbs(ratesLTF[i].low - ref) <= tol)
         count++;
   }
   return (count >= 3);
}

//+------------------------------------------------------------------+
//|                                                                  |
//+------------------------------------------------------------------+
bool LiquidityClusterHigh(int lookback = 10, double tolerance = 10)
{
   int count = 0;
   double ref = iHigh(_Symbol, LTF, 2);
   for (int i = 2; i < lookback; i++)
   {
      if (MathAbs(iHigh(_Symbol, LTF, i) - ref) < tolerance * _Point)
         count++;
   }
   return count >= 3; // cluster valid
}

void DrawLiquidityLevelsTF(LiquidityLevel &levels[], ENUM_TIMEFRAMES tf, bool isBuy)
{
   string tfLabel = EnumToString(tf);
   for (int i = 0; i < ArraySize(levels); i++)
   {
      string name = Name("LQ_" + tfLabel + "_" + IntegerToString(i));
      color clr;
      if (levels[i].isHigh)
         clr = levels[i].swept ? clrGray : clrRed;
      else
         clr = levels[i].swept ? clrGray : clrBlue;
      ObjectCreate(0, name, OBJ_HLINE, 0, 0, levels[i].price);
      ObjectSetInteger(0, name, OBJPROP_COLOR, clr);
      ObjectSetInteger(0, name, OBJPROP_WIDTH, 2);
      string text =
          tfLabel + " | " +
          (levels[i].isHigh ? "EQH" : "EQL") +
          " | " +
          DoubleToString(levels[i].price, _Digits);
      ObjectCreate(0, name + "_txt", OBJ_TEXT, 0, TimeCurrent(), levels[i].price);
      ObjectSetString(0, name + "_txt", OBJPROP_TEXT, text);
      ObjectSetInteger(0, name + "_txt", OBJPROP_COLOR, clr);

      {
         break;
      }
      else if (iClose(_Symbol, LTF, 1) < levels[i].price && !isBuy)
      {
         break;
      }
   }
}

//====================================================
// SCALE MEMORY
//====================================================
struct ScaleMemory
{
   ulong ticket;

   double initialVolume;

   bool breakEvenDone;

   bool tp1Done;
   bool tp2Done;
};

ScaleMemory scaleMemory[100];

int scaleMemoryCount = 0;

//====================================================
// GET SCALE MEMORY
//====================================================
int GetScaleMemoryIndex(ulong ticket)
{
   for (int i = 0; i < scaleMemoryCount; i++)
   {
      if (scaleMemory[i].ticket == ticket)
         return i;
   }
   if (scaleMemoryCount >= 100)
      return 0;
   scaleMemory[scaleMemoryCount].ticket = ticket;
   scaleMemory[scaleMemoryCount].initialVolume = 0.0;
   scaleMemory[scaleMemoryCount].breakEvenDone = false;
   scaleMemory[scaleMemoryCount].tp1Done = false;
   scaleMemory[scaleMemoryCount].tp2Done = false;
   scaleMemoryCount++;
   return scaleMemoryCount - 1;
}

//====================================================
// NORMALIZE VOLUME
//====================================================
double NormalizeVolume(double volume)
{
   double minLot =
       SymbolInfoDouble(
           _Symbol,
           SYMBOL_VOLUME_MIN);
   double maxLot =
       SymbolInfoDouble(
           _Symbol,
           SYMBOL_VOLUME_MAX);
   double step =
       SymbolInfoDouble(
           _Symbol,
           SYMBOL_VOLUME_STEP);
   volume =
       MathFloor(
           volume / step) *
       step;
   volume =
       MathMax(
           volume,
           minLot);
   volume =
       MathMin(
           volume,
           maxLot);
   return NormalizeDouble(
       volume,
       2);
}

//====================================================
// SCALE EXIT ENGINE
//====================================================
void ManageScalingExit()
{
   if (!TrailingOn)
      return;
   //====================================================
   // VALIDATION
   //====================================================
   if (PositionsTotal() <= 0)
      return;
   if (ArraySize(ratesLTF) < 2)
      return;
   //====================================================
   // ATR
   //====================================================
   double atr =
       GetATR(1);
   if (atr <= 0)
      return;
   int htfbias = GetHTFBias();
   //====================================================
   // LOOP POSITIONS
   //====================================================
   for (int i = PositionsTotal() - 1; i >= 0; i--)
   {
      //====================================================
      // POSITION
      //====================================================
      ulong ticket =
          PositionGetTicket(i);
      if (ticket <= 0)
         continue;
      if (!PositionSelectByTicket(ticket))
         continue;
      //====================================================
      // SYMBOL FILTER
      //====================================================
      string symbol =
          PositionGetString(
              POSITION_SYMBOL);
      if (symbol != _Symbol)
         continue;
      //====================================================
      // MEMORY
      //====================================================
      int memIndex =
          GetScaleMemoryIndex(ticket);
      //====================================================
      // POSITION DATA
      //====================================================
      ENUM_POSITION_TYPE type =
          (ENUM_POSITION_TYPE)
              PositionGetInteger(
                  POSITION_TYPE);
      double entry =
          PositionGetDouble(
              POSITION_PRICE_OPEN);
      double sl =
          PositionGetDouble(
              POSITION_SL);
      double tp =
          PositionGetDouble(
              POSITION_TP);
      double volume =
          PositionGetDouble(
              POSITION_VOLUME);
      double price =
          PositionGetDouble(
              POSITION_PRICE_CURRENT);
      //====================================================
      // SAVE INITIAL VOLUME
      //====================================================
      if (
          scaleMemory[memIndex]
              .initialVolume <= 0)
      {
         scaleMemory[memIndex]
             .initialVolume =
             volume;
      }
      double initialVolume =
          scaleMemory[memIndex]
              .initialVolume;
      //====================================================
      // AUTO SL / TP
      //====================================================
      if (sl <= 0 || tp <= 0)
      {
         double newSL = sl;
         double newTP = tp;
         //----------------------------------------------------
         // BUY
         //----------------------------------------------------
         if (type == POSITION_TYPE_BUY)
         {
            if (sl <= 0)
            {
               newSL =
                   entry -
                   (atr *
                    DefaultSL_ATR_Multiplier);
            }
            if (tp <= 0)
            {
               newTP =
                   entry +
                   (atr *
                    DefaultTP_ATR_Multiplier);
            }
         }
         //----------------------------------------------------
         // SELL
         //----------------------------------------------------
         if (type == POSITION_TYPE_SELL)
         {
            if (sl <= 0)
            {
               newSL =
                   entry +
                   (atr *
                    DefaultSL_ATR_Multiplier);
            }
            if (tp <= 0)
            {
               newTP =
                   entry -
                   (atr *
                    DefaultTP_ATR_Multiplier);
            }
         }
         //----------------------------------------------------
         // APPLY
         //----------------------------------------------------
         trade.PositionModify(
             ticket,
             NormalizeDouble(
                 newSL,
                 _Digits),
             NormalizeDouble(
                 newTP,
                 _Digits));
         //----------------------------------------------------
         // REFRESH
         //----------------------------------------------------
         PositionSelectByTicket(ticket);
         sl =
             PositionGetDouble(
                 POSITION_SL);
         tp =
             PositionGetDouble(
                 POSITION_TP);
      }
      //====================================================
      // RISK
      //====================================================
      double risk =
          MathAbs(
              entry - sl);
      if (risk <= 0)
         continue;
      //====================================================
      // PROFIT DISTANCE
      //====================================================
      double profitDistance =
          0.0;
      //----------------------------------------------------
      // BUY
      //----------------------------------------------------
      if (type == POSITION_TYPE_BUY)
      {
         profitDistance =
             price - entry;
      }
      //----------------------------------------------------
      // SELL
      //----------------------------------------------------
      if (type == POSITION_TYPE_SELL)
      {
         profitDistance =
             entry - price;
      }
      //====================================================
      // RR
      //====================================================
      double rr =
          profitDistance / risk;
      //====================================================
      // BREAK EVEN
      //====================================================
      if (
          rr >= BreakEvenRR &&
          !scaleMemory[memIndex]
               .breakEvenDone)
      {
         double beBuffer =
             atr *
             BreakEvenBufferATR;
         double bePrice =
             entry;
         //----------------------------------------------------
         // BUY
         //----------------------------------------------------
         if (type == POSITION_TYPE_BUY)
         {
            bePrice += beBuffer;
            if (sl < bePrice)
            {
               if (
                   trade.PositionModify(
                       ticket,
                       NormalizeDouble(
                           bePrice,
                           _Digits),
                       tp))
               {
                  scaleMemory[memIndex]
                      .breakEvenDone = true;
               }
            }
         }
         //----------------------------------------------------
         // SELL
         //----------------------------------------------------
         if (type == POSITION_TYPE_SELL)
         {
            bePrice -= beBuffer;
            if (
                sl > bePrice ||
                sl == 0)
            {
               if (
                   trade.PositionModify(
                       ticket,
                       NormalizeDouble(
                           bePrice,
                           _Digits),
                       tp))
               {
                  scaleMemory[memIndex]
                      .breakEvenDone = true;
               }
            }
         }
      }
      //====================================================
      // TP1 SCALE OUT
      //====================================================
      if (
          rr >= TP1_RR &&
          !scaleMemory[memIndex]
               .tp1Done &&
          htfbias == 0)
      {
         double closeVolume =
             NormalizeVolume(
                 initialVolume *
                 TP1_ClosePercent);
         if (
             closeVolume > 0 &&
             closeVolume < volume)
         {
            if (
                trade.PositionClosePartial(
                    ticket,
                    closeVolume))
            {
               scaleMemory[memIndex]
                   .tp1Done = true;
            }
         }
      }
      //====================================================
      // TP2 SCALE OUT
      //====================================================
      if (
          rr >= TP2_RR &&
          !scaleMemory[memIndex]
               .tp2Done &&
          htfbias == 0)
      {
         double closeVolume =
             NormalizeVolume(
                 initialVolume *
                 TP2_ClosePercent);
         if (
             closeVolume > 0 &&
             closeVolume < volume)
         {
            if (
                trade.PositionClosePartial(
                    ticket,
                    closeVolume))
            {
               scaleMemory[memIndex]
                   .tp2Done = true;
            }
         }
      }
      //====================================================
      // RUNNER TRAILING
      //====================================================
      if (rr >= TP1_RR)
      {
         double trailDistance =
             atr *
             TrailATRMultiplier;
         double newSL =
             sl;
         //----------------------------------------------------
         // BUY
         //----------------------------------------------------
         if (type == POSITION_TYPE_BUY)
         {
            double atrTrail =
                price - trailDistance;
            double swingTrail =
                ratesLTF[1].low;
            newSL =
                MathMax(
                    atrTrail,
                    swingTrail);
            newSL =
                NormalizeDouble(
                    newSL,
                    _Digits);
            if (
                newSL > sl &&
                newSL < price)
            {
               trade.PositionModify(
                   ticket,
                   newSL,
                   tp);
            }
         }
         //----------------------------------------------------
         // SELL
         //----------------------------------------------------
         if (type == POSITION_TYPE_SELL)
         {
            double atrTrail =
                price + trailDistance;
            double swingTrail =
                ratesLTF[1].high;
            newSL =
                MathMin(
                    atrTrail,
                    swingTrail);
            newSL =
                NormalizeDouble(
                    newSL,
                    _Digits);
            if (
                (
                    newSL < sl ||
                    sl == 0) &&
                newSL > price)
            {
               trade.PositionModify(
                   ticket,
                   newSL,
                   tp);
            }
         }
      }
   }
}

//====================================================
// DYNAMIC TP SCALING ENGINE
// REGIME + AUCTION BASED MANAGEMENT
//====================================================
void ManageDynamicTPScaling()
{
   //====================================================
   // VALIDATION
   //====================================================
   if (TotalTrade() <= 0)
      return;
   //====================================================
   // MARKET CONTEXT
   //====================================================
   double atr =
       GetATR(1);
   if (atr <= 0)
      return;
   //====================================================
   // HTF BIAS — computed once per bar, not per position
   //====================================================
   int htfBias = GetHTFBias();
   bool rangeMode = (htfBias == 0);
   bool trendMode = (htfBias == 1 || htfBias == -1);
   //====================================================
   // LOOP POSITIONS
   // Use PositionsTotal() as bound so the raw index
   // passed to PositionGetTicket(i) is always valid.
   //====================================================
   for (int i = PositionsTotal() - 1; i >= 0; i--)
   {
      ulong ticket =
          PositionGetTicket(i);
      if (ticket == 0)
         continue;
      if (!PositionSelectByTicket(ticket))
         continue;
      //====================================================
      // SYMBOL FILTER
      //====================================================
      string symbol =
          PositionGetString(
              POSITION_SYMBOL);
      if (symbol != _Symbol)
         continue;
      //====================================================
      // POSITION DATA
      //====================================================
      int type =
          (int)PositionGetInteger(
              POSITION_TYPE);
      double entry =
          PositionGetDouble(
              POSITION_PRICE_OPEN);
      double sl =
          PositionGetDouble(
              POSITION_SL);
      double tp =
          PositionGetDouble(
              POSITION_TP);
      double currentPrice =
          PositionGetDouble(
              POSITION_PRICE_CURRENT);
      //====================================================
      // AUTO SL
      //====================================================
      if (sl <= 0)
      {
         double autoSL = 0.0;
         if (type == POSITION_TYPE_BUY)
         {
            autoSL =
                entry -
                (atr * 2.0);
         }
         else
         {
            autoSL =
                entry +
                (atr * 2.0);
         }
         autoSL =
             NormalizeDouble(
                 autoSL,
                 _Digits);
         trade.PositionModify(
             ticket,
             autoSL,
             tp);
         continue;
      }
      //====================================================
      // RISK
      //====================================================
      double risk =
          MathAbs(
              entry - sl);
      if (risk <= 0)
         continue;
      //====================================================
      // CURRENT PROFIT
      //====================================================
      double profit = 0.0;
      if (type == POSITION_TYPE_BUY)
      {
         profit =
             currentPrice - entry;
      }
      else
      {
         profit =
             entry - currentPrice;
      }
      double rr =
          profit / risk;
      //====================================================
      // TARGET RR
      //====================================================
      double targetRR = 2.0;
      //====================================================
      // RANGE MODE
      //====================================================
      if (rangeMode)
      {
         targetRR = 2.0;
         if (rotationDay)
            targetRR = 1.5;
         if (failedAuctionUp || failedAuctionDown)
            targetRR = 1.2;
      }
      //====================================================
      // TREND MODE
      //====================================================
      if (trendMode)
      {
         targetRR = 3.0;
         if (trendDay)
            targetRR = 5.0;
         if (
             pocMigrationUp ||
             pocMigrationDown)
         {
            targetRR += 1.0;
         }
         if (
             aggressiveBuy ||
             aggressiveSell)
         {
            targetRR += 1.0;
         }
      }
      //====================================================
      // EXHAUSTION REDUCTION
      //====================================================
      if (
          buyerExhaustion ||
          sellerExhaustion)
      {
         targetRR *= 0.70;
      }
      //====================================================
      // FAILED AUCTION REDUCTION
      //====================================================
      if (
          failedAuctionUp ||
          failedAuctionDown)
      {
         targetRR *= 0.60;
      }
      //====================================================
      // BUILD DYNAMIC TP
      //====================================================
      double dynamicTP = 0.0;
      // Print("Target RR", targetRR);
      if (type == POSITION_TYPE_BUY)
      {
         dynamicTP =
             entry +
             (risk * targetRR);
      }
      else
      {
         dynamicTP =
             entry -
             (risk * targetRR);
      }
      dynamicTP =
          NormalizeDouble(
              dynamicTP,
              _Digits);
      //====================================================
      // AUTO TP
      //====================================================
      bool invalidTP = false;
      if (type == POSITION_TYPE_BUY)
      {
         if (tp <= entry)
            invalidTP = true;
      }
      else
      {
         if (tp >= entry || tp <= 0)
            invalidTP = true;
      }
      if (invalidTP)
      {
         trade.PositionModify(
             ticket,
             sl,
             dynamicTP);
         continue;
      }
      //====================================================
      // TP SCALING
      //====================================================
      bool shouldExpandTP = false;
      //====================================================
      // BUY EXPANSION
      //====================================================
      if (type == POSITION_TYPE_BUY)
      {
         shouldExpandTP =
             (acceptedUp &&
              aggressiveBuy &&
              !buyerExhaustion &&
              trendDay);
      }
      //====================================================
      // SELL EXPANSION
      //====================================================
      if (type == POSITION_TYPE_SELL)
      {
         shouldExpandTP =
             (acceptedDown &&
              aggressiveSell &&
              !sellerExhaustion &&
              trendDay);
      }
      //====================================================
      // ONLY EXPAND TP
      // NEVER REDUCE WINNING TP
      //====================================================
      if (type == POSITION_TYPE_BUY)
      {
         if (
             shouldExpandTP &&
             dynamicTP > tp)
         {
            trade.PositionModify(
                ticket,
                sl,
                dynamicTP);
         }
      }
      if (type == POSITION_TYPE_SELL)
      {
         if (
             shouldExpandTP &&
             dynamicTP < tp)
         {
            trade.PositionModify(
                ticket,
                sl,
                dynamicTP);
         }
      }
      //====================================================
      // BREAK EVEN
      //====================================================
      double beRR =
          rangeMode
              ? 0.8
              : 1.5;
      if (rr >= beRR)
      {
         double newSL = entry;
         if (type == POSITION_TYPE_BUY)
         {
            newSL += atr * 0.10;
            if (newSL > sl)
            {
               trade.PositionModify(
                   ticket,
                   NormalizeDouble(
                       newSL,
                       _Digits),
                   tp);
            }
         }
         else
         {
            newSL -= atr * 0.10;
            if (sl == 0 || newSL < sl)
            {
               trade.PositionModify(
                   ticket,
                   NormalizeDouble(
                       newSL,
                       _Digits),
                   tp);
            }
         }
      }
      //====================================================
      // TRAILING STOP
      //====================================================
      if (rr >= 2.0)
      {
         double trailATR =
             rangeMode
                 ? 1.0
                 : 2.5;
         double trailDistance =
             atr * trailATR;
         //====================================================
         // BUY TRAILING
         //====================================================
         if (type == POSITION_TYPE_BUY)
         {
            double newSL =
                currentPrice - trailDistance;
            newSL =
                NormalizeDouble(
                    newSL,
                    _Digits);
            if (
                newSL > sl &&
                newSL > entry)
            {
               trade.PositionModify(
                   ticket,
                   newSL,
                   tp);
            }
         }
         //====================================================
         // SELL TRAILING
         //====================================================
         if (type == POSITION_TYPE_SELL)
         {
            double newSL =
                currentPrice + trailDistance;
            newSL =
                NormalizeDouble(
                    newSL,
                    _Digits);
            if (
                (sl == 0 || newSL < sl) &&
                newSL < entry)
            {
               trade.PositionModify(
                   ticket,
                   newSL,
                   tp);
            }
         }
      }
   }
}

//+------------------------------------------------------------------+
//|                                                                  |
//+------------------------------------------------------------------+
double CalculateLot(double entry, double sl)
{
   double riskPercent = RiskPercent;
   double balance = AccountInfoDouble(ACCOUNT_BALANCE);
   double riskMoney = balance * riskPercent / 100.0;
   //========================================
   // 🔥 SYMBOL INFO
   //========================================
   double tickValue = SymbolInfoDouble(_Symbol, SYMBOL_TRADE_TICK_VALUE);
   double tickSize = SymbolInfoDouble(_Symbol, SYMBOL_TRADE_TICK_SIZE);
   double lotStep = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_STEP);
   double minLot = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_MIN);
   double maxLot = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_MAX);
   if (tickValue <= 0 || tickSize <= 0)
      return minLot;
   //========================================
   // 🔥 STOP DISTANCE
   //========================================
   double stopDistance = MathAbs(entry - sl);
   if (stopDistance <= 0)
      return minLot;
   //========================================
   // 🔥 VALUE PER POINT
   //========================================
   double valuePerPoint = tickValue / tickSize;
   //========================================
   // 🔥 RAW LOT CALCULATION
   //========================================
   double lot = riskMoney / (stopDistance * valuePerPoint);
   //========================================
   // 🔥 SAFETY LIMIT (ANTI HUGE LOT)
   //========================================
   if (lot > maxLot)
      lot = maxLot;
   if (lot < minLot)
      lot = minLot;
   //========================================
   // 🔥 ALIGN TO LOT STEP
   //========================================
   lot = MathFloor(lot / lotStep) * lotStep;
   //========================================
   // 🔥 FINAL NORMALIZE
   //========================================
   int lotDigits = (int)MathRound(-MathLog10(lotStep));
   lot = NormalizeDouble(lot, lotDigits);
   return lot;
}
//+------------------------------------------------------------------+
//|                                                                  |
//+------------------------------------------------------------------+
void EntryEngine()
{
   int bias = GetHTFBias();
   double obLow, obHigh;
   //================ BUY =================//
   if (bias == 1)
   {
      if (!SweepLow())
         return;
      if (!CHoCH_Bullish())
         return;
      if (!DisplacementBullish())
         return;
      if (!GetOrderBlock(true, obLow, obHigh))
         return;
      // confluence: OB + discount + volume
      if (obLow > poc)
         return;
      double entry = (obLow + obHigh) / 2;
      double sl = obLow - 20 * _Point;
      double tp = entry + (entry - sl) * 3;
      double lot = CalculateLot(entry, sl);
      ExecuteTrade(lot, true, entry, sl, tp);
   }
   //================ SELL =================//
   if (bias == -1)
   {
      if (!SweepHigh())
         return;
      if (!CHoCH_Bearish())
         return;
      if (!DisplacementBearish())
         return;
      if (!GetOrderBlock(false, obLow, obHigh))
         return;
      if (obHigh < poc)
         return;
      double entry = (obLow + obHigh) / 2;
      double sl = obHigh + 20 * _Point;
      double tp = entry - (sl - entry) * 3;
      double lot = CalculateLot(entry, sl);
      ExecuteTrade(lot, false, entry, sl, tp);
   }
}

//+------------------------------------------------------------------+
//|                                                                  |
//+------------------------------------------------------------------+
bool ExecuteTrade(double lot, bool isBuy, double entry, double sl, double tp)
{
   double tolerance = GetATR(1) * 0.2;
   if (HasSimilarPending(entry, tolerance))
      return false;
   string comment = "SMC_" + IntegerToString(TimeCurrent());
   if (isBuy)
      return trade.BuyLimit(lot, entry, _Symbol, sl, tp, ORDER_TIME_GTC, 0, comment);
   else
      return trade.SellLimit(lot, entry, _Symbol, sl, tp, ORDER_TIME_GTC, 0, comment);
}

bool IsValidStops(double entry, double sl, double tp)
{
   double stopLevel = SymbolInfoInteger(_Symbol, SYMBOL_TRADE_STOPS_LEVEL) * _Point;
   if (MathAbs(entry - sl) < stopLevel)
      return false;
   if (MathAbs(entry - tp) < stopLevel)
      return false;
   return true;
}

//|                                                                  |
//+------------------------------------------------------------------+
bool IsOBMitigated(int obShift, bool bullish, double atr)
{
   double obHigh = iHigh(_Symbol, LTF, obShift);
   double obLow = iLow(_Symbol, LTF, obShift);
   double mid = (obHigh + obLow) * 0.5;
   if (atr <= 0)
      return false;
   int maxLookback = 50;
   for (int i = obShift - 1; i >= 0 && i >= obShift - maxLookback; i--)
   {
      double high = iHigh(_Symbol, LTF, i);
      double low = iLow(_Symbol, LTF, i);
      double close = iClose(_Symbol, LTF, i);
      if (bullish)
      {
         // [FIX] OB bullish ter-mitigasi jika CLOSE (bukan wick) turun di bawah midpoint
         // dengan penetrasi signifikan >= 0.3 ATR (bukan 0.1)
         // Wick boleh masuk OB - itu biasa dan sehat
         if (close < mid && (mid - close) > atr * 0.3)
            return true;
      }
      else
      {
         // [FIX] OB bearish ter-mitigasi jika CLOSE naik di atas midpoint
         if (close > mid && (close - mid) > atr * 0.3)
            return true;
      }
   }
   return false;
}
//+------------------------------------------------------------------+
//|                                                                  |
//+------------------------------------------------------------------+
bool IsFVGMitigated(double fvgLow, double fvgHigh, double atr, bool bullish)
{
   double fvgSize = fvgHigh - fvgLow;
   if (fvgSize <= _Point || atr <= 0)
      return true;
   int bars = ArraySize(ratesLTF);
   int maxLookback = 50;
   double maxFill = 0;
   for (int i = 1; i < maxLookback && i < bars; i++)
   {
      double high = ratesLTF[i].high;
      double low = ratesLTF[i].low;
      double close = ratesLTF[i].close;
      double overlap = MathMin(high, fvgHigh) - MathMax(low, fvgLow);
      if (overlap <= 0)
         continue;
      double fillRatio = overlap / fvgSize;
      if (fillRatio > maxFill)
         maxFill = fillRatio;
      //============================
      // FULL FILL → INVALID (>95%)
      //============================
      if (fillRatio > 0.95)
         return true;
      //============================
      // STRUCTURE BREAK BEYOND FVG
      // [FIX] threshold 0.2→0.3 ATR agar tidak false-positive
      // pada kondisi volatil normal
      //============================
      if (bullish)
      {
         if (close < fvgLow - atr * 0.3)
            return true;
      }
      else
      {
         if (close > fvgHigh + atr * 0.3)
            return true;
      }
   }
   //============================
   // PARTIAL FILL
   // [FIX] 0.70 → 0.85: FVG masih valid sampai 85% terisi
   // (70% terlalu agresif, FVG biasa tap dalam dan masih reaktif)
   //============================
   if (maxFill > 0.85)
      return true;
   return false;
}

void DrawOBZone(const SMCEvent &e, bool bullish)
{
   string name = Name(bullish ? "OB_BULL_" : "OB_BEAR_") + IntegerToString(e.obShift);
   if (!e.hasOB || e.obShift < 0 || e.obHigh <= e.obLow)
   {
      removeOBFVG(name);
      return;
   }
   double high = e.obHigh;
   double low = e.obLow;
   double top = MathMax(high, low);
   double bottom = MathMin(high, low);
   datetime time1 = iTime(_Symbol, LTF, e.obShift);
   datetime time2 = iTime(_Symbol, LTF, 0) + PeriodSeconds(LTF) * 30;
   color clr = bullish ? clrGreen : clrRed;
   // =============================
   // CREATE / UPDATE
   // =============================
   if (ObjectFind(0, name) < 0)
   {
      if (!ObjectCreate(0, name, OBJ_RECTANGLE, 0, time1, top, time2, bottom))
      {
         DebugSMC("OB CREATE ERROR: " + GetLastError());
         return;
      }
   }
   else
   {
      ObjectMove(0, name, 0, time1, top);
      ObjectMove(0, name, 1, time2, bottom);
   }
   // =============================
   // STYLE
   // =============================
   ObjectSetInteger(0, name, OBJPROP_COLOR, ColorToARGB(clr, 70));
   ObjectSetInteger(0, name, OBJPROP_FILL, true);
   ObjectSetInteger(0, name, OBJPROP_BACK, true);
   ObjectSetInteger(0, name, OBJPROP_WIDTH, 1);
   ObjectSetInteger(0, name, OBJPROP_SELECTABLE, false);
}

   string name = Name(bullish ? "FVG_BULL_" : "FVG_BEAR_") + IntegerToString(e.displacementShift);
   if (!e.hasFVG || e.fvgBottom == 0 || e.fvgTop == 0 ||
       e.displacementShift < 0)
   {

      return;
   }
   // biar selalu update
   ObjectDelete(0, name);
   // =============================
   // TIME
   // =============================
   datetime time1 = iTime(_Symbol, LTF, e.displacementShift + 2);
   datetime time2 = TimeCurrent(); // time1 + PeriodSeconds(LTF) * 50;
   // =============================
   // PRICE
   // =============================
   double top = MathMax(e.fvgTop, e.fvgBottom);
   double bottom = MathMin(e.fvgTop, e.fvgBottom);
   // =============================
   // CREATE
   // =============================
   if (!ObjectCreate(0, name, OBJ_RECTANGLE, 0, time1, top, time2, bottom))
   {
      DebugSMC("FVG DRAW ERROR: " + GetLastError());
      return;
   }
   color clr = bullish ? clrGreen : clrOrange;
   // =============================
   // STYLE
   // =============================
   ObjectSetInteger(0, name, OBJPROP_COLOR, ColorToPRGB(clr, 70));
   ObjectSetInteger(0, name, OBJPROP_FILL, true);
   ObjectSetInteger(0, name, OBJPROP_BACK, true);
   ObjectSetInteger(0, name, OBJPROP_WIDTH, 1);
}

void DrawSniperZone(bool bullish)
{
   double fvgLow, fvgHigh;
   if (!FVG(bullish, fvgLow, fvgHigh))
      return;
   int swingL = SwingLow(3);
   int swingH = SwingHigh(3);
   if (swingL == -1 || swingH == -1)
      return;
   double low = iLow(_Symbol, LTF, swingL);
   double high = iHigh(_Symbol, LTF, swingH);
   double oteLow, oteHigh;
   if (bullish)
   {
      oteHigh = high - (high - low) * 0.618;
      oteLow = high - (high - low) * 0.782;
   }
   else
   {
      oteLow = low + (high - low) * 0.618;
      oteHigh = low + (high - low) * 0.782;
   }
   // INTERSECTION
   double zoneLow = MathMax(fvgLow, oteLow);
   double zoneHigh = MathMin(fvgHigh, oteHigh);
   if (zoneHigh <= zoneLow)
      return;
   DrawZone(Name("SNIPER"),
            zoneLow, zoneHigh,
            iTime(_Symbol, LTF, 5),
            iTime(_Symbol, LTF, 0),
            clrGold);
}

//| BULLISH CHECKLIST (SYNC WITH NEW EVENT PIPELINE)                 |
//+------------------------------------------------------------------+
string BuildBullishChecklist(SMCEvent &e, double price)
{
   string text = "";
   //====================================================
   // PIPELINE STATE
   //====================================================
   text +=
       Check(e.liquidityShift > 0) +
       "Liquidity Pool Found\n";
   text +=
       Check(e.sweepShift > -1) +
       "Sweep Low\n";
   text +=
       Check(e.displacementShift > -1) +
       "Displacement (Bullish)\n";
   text +=
       Check(e.chochShift > -1) +
       "CHoCH Confirmed\n";
   //====================================================
   // MARKET ACCEPTANCE
   //====================================================
   bool accepted =
       (e.liquidity > 0 &&
        ratesLTF[0].close > e.liquidity);
   text +=
       Check(accepted) +
       "Market Accepted Above Liquidity\n";
   //====================================================
   // ORDER BLOCK
   //====================================================
   bool validOB =
       (e.hasOB &&
        e.obHigh > e.obLow &&
        e.obLow > 0);
   text +=
       Check(validOB) +
       "OB Valid  [" +
       DoubleToString(e.obLow, _Digits) +
       " – " +
       DoubleToString(e.obHigh, _Digits) +
       "]\n";
   //====================================================
   // FVG
   //====================================================
   bool validFVG =
       (e.hasFVG &&
        e.fvgTop > e.fvgBottom &&
        e.fvgBottom > 0);
   text +=
       Check(validFVG) +
       "FVG Valid  [" +
       DoubleToString(e.fvgBottom, _Digits) +
       " – " +
       DoubleToString(e.fvgTop, _Digits) +
       "]\n";
   //====================================================
   // EXECUTION ZONE
   //====================================================
   double low = 0;
   double high = 0;
   string zoneType = "NONE";
   if (validFVG)
   {
      low = e.fvgBottom;
      high = e.fvgTop;
      zoneType = "FVG";
   }
   else if (validOB)
   {
      low = e.obLow;
      high = e.obHigh;
      zoneType = "OB";
   }
   bool executableZone =
       (low > 0 &&
        high > low);
   text +=
       Check(executableZone) +
       "Entry Zone (" + zoneType + ")\n";
   //====================================================
   // IN ZONE
   //====================================================
   bool inZone = false;
   if (executableZone)
   {
      inZone =
          (price >= low &&
           price <= high);
   }
   text +=
       Check(inZone) +
       "Price In Zone\n";
   //====================================================
   // MICRO CONFIRMATION
   //====================================================
   bool confirm = false;
   if (inZone && e.chochShift >= 0)
   {
      confirm = ConfirmBullishCandle(1);
   }
   text +=
       Check(confirm) +
       "Candle Confirmation\n";
   //====================================================
   // RE-ACCELERATION
   //====================================================
   bool reAccel = false;
   if (e.displacementShift >= 0)
   {
      double body =
          ratesLTF[0].close -
          ratesLTF[0].open;
      reAccel =
          (body > 0 &&
           MathAbs(body) > GetATR(1) * 0.08);
   }
   text +=
       Check(reAccel) +
       "Re-Acceleration\n";
   //====================================================
   // ENTRY READY + SCORE
   //====================================================
   text +=
       Check(e.valid) +
       "Event Valid  [Score: " + IntegerToString(e.strength) + "]\n";
   text +=
       Check(e.entryReady) +
       "Entry Ready\n";
   //====================================================
   // SEND NOTIFICATION
   //====================================================
   if (accepted)
      SendDashboardNotification("BUY SETUP ACTIVE\n" + text);
   return text;
}

       Check(inZone) +
       "Price In Zone\n";
   //====================================================
   // MICRO CONFIRMATION
   //====================================================
   bool confirm = false;
   if (inZone && e.chochShift >= 0)
   {
      confirm = ConfirmBearishCandle(1);
   }
   text +=
       Check(confirm) +
       "Candle Confirmation\n";
   //====================================================
   // RE-ACCELERATION
   //====================================================
   bool reAccel = false;
   if (e.displacementShift >= 0)
   {
      double body =
          ratesLTF[0].open -
          ratesLTF[0].close;
      reAccel =
          (body > 0 &&
           MathAbs(body) > GetATR(1) * 0.08);
   }
   text +=
       Check(reAccel) +
       "Re-Acceleration\n";
   //====================================================
   // ENTRY READY + SCORE
   //====================================================
   bool ready =
       (e.valid &&
        accepted &&
        executableZone);
   text +=
       Check(e.valid) +
       "Event Valid  [Score: " + IntegerToString(e.strength) + "]\n";
   text +=
       Check(ready) +
       "Entry Ready\n";
   //====================================================
   // SEND NOTIFICATION
   //====================================================
   if (accepted)
      SendDashboardNotification("SELL SETUP ACTIVE\n" + text);
   return text;
}

//================ DISPLACEMENT =================//
bool DisplacementBullish()
{
   double body = MathAbs(iClose(_Symbol, LTF, 1) - iOpen(_Symbol, LTF, 1));
   double range = iHigh(_Symbol, LTF, 1) - iLow(_Symbol, LTF, 1);
   return (range > 0 && body / range > 0.6 && iClose(_Symbol, LTF, 1) > iOpen(_Symbol, LTF, 1));
}

//+------------------------------------------------------------------+
//|                                                                  |
//+------------------------------------------------------------------+
bool DisplacementBearish()
{
   double body = MathAbs(iClose(_Symbol, LTF, 1) - iOpen(_Symbol, LTF, 1));
   double range = iHigh(_Symbol, LTF, 1) - iLow(_Symbol, LTF, 1);
   return (range > 0 && body / range > 0.6 && iClose(_Symbol, LTF, 1) < iOpen(_Symbol, LTF, 1));
}

      return false;

   double lastClose = iClose(_Symbol, LTF, 1);

}

void DrawVP_Bars()
{
   if (!ShowVPBars)
      return;
   if (bins <= 0)
      return;
   double maxVol = 0;
   double avg = 0;
   for (int i = 0; i < bins; i++)
   {
      if (volumeX[i] > maxVol)
         maxVol = volumeX[i];
      avg += volumeX[i];
   }
   if (maxVol <= 0)
      return;
   avg /= bins;
   datetime t0 = iTime(_Symbol, LTF, 0) + PeriodSeconds(LTF) * 2;
   int baseWidth = PeriodSeconds(LTF) * 30;
   for (int i = 0; i < bins; i++)
   {
      double priceLow = lowest + i * step;
      double priceHigh = priceLow + step;
      double factor = volumeX[i] / maxVol;
      datetime tEnd = t0 + (datetime)(baseWidth * factor);
      string name = Name("VP_BAR_" + IntegerToString(i));
      //========================================
      // 🔥 REUSE OBJECT (NO CREATE SPAM)
      //========================================
      if (ObjectFind(0, name) < 0)
      {
         ObjectCreate(0, name, OBJ_RECTANGLE, 0,
                      t0, priceLow,
                      tEnd, priceHigh);
      }
      else
      {
         ObjectMove(0, name, 0, t0, priceLow);
         ObjectMove(0, name, 1, tEnd, priceHigh);
      }
      color c = clrGray;
      if (IsHVN(volumeX[i], avg))
         c = clrGreen;
      else if (IsLVN(volumeX[i], avg))
         c = clrOrange;
      ObjectSetInteger(0, name, OBJPROP_COLOR, c);
      ObjectSetInteger(0, name, OBJPROP_BACK, true);
   }
}

bool IsNearHVN(double price, double atr)
{
   if (bins <= 2)
      return false;
   double range = highest - lowest;
   if (range <= 0)
      return false;
   double step = range / bins;
   int idx = int((price - lowest) / step);
   if (idx < 1 || idx >= bins - 1)
      return false;
   double v = volumeX[idx];
   //========================
   // NORMALIZATION
   //========================
   double maxVol = 0;
   for (int i = 0; i < bins; i++)
      if (volumeX[i] > maxVol)
         maxVol = volumeX[i];
   if (maxVol <= 0)
      return false;
   double normV = v / maxVol;
   //========================
   // LOCAL PEAK + STRONG
   //========================
   bool isPeak = (v > volumeX[idx - 1] && v > volumeX[idx + 1]);
   if (isPeak && normV > 0.6) // 🔥 threshold realistis
   {
      double nodePrice = lowest + idx * step;
      if (MathAbs(price - nodePrice) < atr * 0.2)
         return true;
   }
   return false;
}

//+------------------------------------------------------------------+
bool IsNearLVN(double price, double atr)
{
   if (bins <= 2)
      return false;
   double range = highest - lowest;
   if (range <= 0)
      return false;
   double step = range / bins;
   int idx = int((price - lowest) / step);
   if (idx < 1 || idx >= bins - 1)
      return false;
   double v = volumeX[idx];
   //========================
   // NORMALIZATION
   //========================
   double maxVol = 0;
   for (int i = 0; i < bins; i++)
      if (volumeX[i] > maxVol)
         maxVol = volumeX[i];
   if (maxVol <= 0)
      return false;
   double normV = v / maxVol;
   //========================
   // LOCAL VALLEY
   //========================
   bool isValley = (v < volumeX[idx - 1] && v < volumeX[idx + 1]);
   if (isValley && normV < 0.4)
   {
      double nodePrice = lowest + idx * step;
      if (MathAbs(price - nodePrice) < atr * 0.2)
         return true;
   }
   return false;
}

//+------------------------------------------------------------------+
void DrawMultiTFExternalLiquidity(int htfBias)
{
   if (htfBias == 0)
      return;
   ENUM_TIMEFRAMES listF[] = {LTF, PERIOD_M30, PERIOD_H1, PERIOD_H4, PERIOD_D1, PERIOD_W1};
   bool isBuy = (htfBias == 1);
   double price = isBuy
                      ? SymbolInfoDouble(_Symbol, SYMBOL_BID)
                      : SymbolInfoDouble(_Symbol, SYMBOL_ASK);
   double atr = GetATR(1);
   if (atr <= 0)
      return;
   double minDist = atr * 0.5; // 🔥 avoid noise
   double bestDist = DBL_MAX;
   LiquidityLevel bestLevel;
   ENUM_TIMEFRAMES bestTF = LTF;
   bool found = false;
   //========================================
   // 🔥 SCAN ALL TF → PICK BEST
   //========================================
   for (int iTF = 0; iTF < ArraySize(listF); iTF++)
   {
      LiquidityLevel levels[];
      int count = DetectExternalLiquidityTF(levels, isBuy, listF[iTF]);
      if (count <= 0)
         continue;
      for (int i = 0; i < count; i++)
      {
         if (levels[i].swept)
            continue;
         double lvl = levels[i].price;
         double dist = MathAbs(lvl - price);
         // 🔥 FILTER DIRECTION
         if (isBuy && lvl <= price)
            continue;
         if (!isBuy && lvl >= price)
            continue;
         // 🔥 FILTER NOISE
         if (dist < minDist)
            continue;
         // 🔥 PICK CLOSEST VALID
         if (dist < bestDist)
         {
            bestDist = dist;
            bestLevel = levels[i];
            bestTF = listF[iTF];
            found = true;
         }
      }
   }
   //========================================
   // 🔥 DRAW ONLY BEST
   //========================================
   if (found)
   {
      LiquidityLevel single[];
      ArrayResize(single, 1);
      single[0] = bestLevel;
      DrawLiquidityLevelsTF(single, bestTF, isBuy);
      DebugSMC("Best Liquidity TF: " + EnumToString(bestTF));
      DebugSMC("Liquidity Price: " + DoubleToString(bestLevel.price));
   }
   else
   {
      DebugSMC("No Valid External Liquidity Found");
   }
}
//====================================================
// INSTITUTIONAL VOLUME PROFILE STRATEGY ENGINE
//====================================================
//====================================================
// VOLUME PROFILE STRATEGY ENGINE
//====================================================
void VolumeProfileStrategy()
{
   //====================================================
   // RESET
   //====================================================
   buySignal = false;
   sellSignal = false;
   balancedMarket = false;
   trendingUP = false;
   trendingDOWN = false;
   signalType = "";
   //====================================================
   // MARKET DATA
   //====================================================
   double bid =
       SymbolInfoDouble(
           _Symbol,
           SYMBOL_BID);
   double ask =
       SymbolInfoDouble(
           _Symbol,
           SYMBOL_ASK);
   double spread =
       ask - bid;
   double atr =
       GetATR(1);
   if (atr <= 0)
      return;
   //====================================================
   // CORE VALIDATION
   //====================================================
   if (vah <= val)
      return;
   if (step <= 0)
      return;
   if (spread > atr * 0.20)
      return;
   if (ArraySize(ratesLTF) < 3)
      return;
   //====================================================
   // VALUE STRUCTURE
   //====================================================
   double valueRange =
       vah - val;
   if (valueRange <= 0)
      return;
   ZeroMemory(signal);
   //====================================================
   // MARKET LOCATION
   //====================================================
   bool insideValue =
       (bid >= val &&
        bid <= vah);
   bool aboveValue =
       (bid > vah);
   bool belowValue =
       (bid < val);
   int htfBias = GetHTFBias();
   //====================================================
   //  PROFILE CONTEXT
   //====================================================
   balancedMarket =
       (profileShape == "D" &&
        !acceptedUp &&
        !acceptedDown);
   trendingUP =
       (acceptedUp &&
        pocMigrationUp &&
        aboveValue);
   trendingDOWN =
       (acceptedDown &&
        pocMigrationDown &&
        belowValue);
   //====================================================
   // CANDLE DATA
   //====================================================
   double open0 = ratesLTF[0].open;
   double close0 = ratesLTF[0].close;
   double high0 = ratesLTF[0].high;
   double low0 = ratesLTF[0].low;
   double open1 = ratesLTF[1].open;
   double close1 = ratesLTF[1].close;
   double high1 = ratesLTF[1].high;
   double low1 = ratesLTF[1].low;
   double currentPrice = ratesLTF[0].close;
   //====================================================
   // CANDLE STRUCTURE
   //====================================================
   double body0 =
       MathAbs(
           close0 - open0);
   double range0 =
       high0 - low0;
   bool bullishClose =
       (close0 > open0);
   bool bearishClose =
       (close0 < open0);
   //====================================================
   // DISPLACEMENT
   //====================================================
   bool bullishDisplacement =
       (close0 > high1 &&
        body0 > atr * 0.20);
   bool bearishDisplacement =
       (close0 < low1 &&
        body0 > atr * 0.20);
   //====================================================
   // FVG
   //====================================================
   bool bullishFVG =
       (ratesLTF[2].high <
        ratesLTF[0].low);
   bool bearishFVG =
       (ratesLTF[2].low >
        ratesLTF[0].high);
   //====================================================
   // DELTA CONTEXT
   //====================================================
   bool bullishDelta =
       (aggressiveBuy &&
        !buyerExhaustion);
   bool bearishDelta =
       (aggressiveSell &&
        !sellerExhaustion);
   //====================================================
   // DELTA DIVERGENCE
   //====================================================
   bool bearishDeltaDivergence =
       (high0 > high1 &&
        !bullishDelta);
   bool bullishDeltaDivergence =
       (low0 < low1 &&
        !bearishDelta);
   //====================================================
   // INVENTORY LOGIC
   //====================================================
   bool trappedLongs =
       (acceptedUp &&
        currentPrice < vah &&
        bearishDelta);
   bool trappedShorts =
       (acceptedDown &&
        currentPrice > val &&
        bullishDelta);
   //====================================================
   // INVENTORY CORRECTION
   //====================================================
   bool inventoryCorrectionBuy =
       (trappedShorts &&
        bullishDisplacement);
   bool inventoryCorrectionSell =
       (trappedLongs &&
        bearishDisplacement);
   //====================================================
   // FALSE BREAKOUT
   //====================================================
   bool falseBreakoutSell =
       (high0 > vah &&
        close0 < vah &&
        bearishDisplacement);
   bool falseBreakoutBuy =
       (low0 < val &&
        close0 > val &&
        bullishDisplacement);
   //====================================================
   // LIQUIDATION CASCADE
   //====================================================
   bool liquidationSell =
       (falseBreakoutSell &&
        bearishDelta &&
        buyerExhaustion);
   bool liquidationBuy =
       (falseBreakoutBuy &&
        bullishDelta &&
        sellerExhaustion);
   //====================================================
   // LVN REJECTION
   //====================================================
   bool lvnRejectBuy =
       (MathAbs(
            bid - lvn) <= step &&
        bullishDisplacement);
   bool lvnRejectSell =
       (MathAbs(
            bid - lvn) <= step &&
        bearishDisplacement);
   //====================================================
   // SINGLE PRINT
   //====================================================
   bool singlePrintBuy =
       (bullishDisplacement &&
        range0 > atr * 0.80);
   bool singlePrintSell =
       (bearishDisplacement &&
        range0 > atr * 0.80);
   //====================================================
   // OPEN DRIVE
   //====================================================
   bool openDriveBuy =
       (londonSession &&
        bullishDelta &&
        bullishDisplacement &&
        bid > vah);
   bool openDriveSell =
       (londonSession &&
        bearishDelta &&
        bearishDisplacement &&
        bid < val);
   //====================================================
   // RESPONSIVE BUY
   //====================================================
   if (balancedMarket)
   {
      bool location =
          (bid <= val + step * 2.0);
      bool rejection =
          (bullishDisplacement ||
           bullishFVG);
      bool absorption =
          (sellerExhaustion ||
           failedAuctionDown);
      if (
          location &&
          rejection &&
          absorption)
      {
         signal.buy = true;
         signal.responsive = true;
         signal.absorption = true;
         signal.priority =
             PRIORITY_STRONG;
         signal.state =
             EXEC_CONFIRM;
         signal.score += 5.0;
         signal.reasons +=
             "RESPONSIVE_BUY|";
         if (bullishFVG)
         {
            signal.score += 1.0;
            signal.reasons +=
                "FVG|";
         }
         signal.type =
             "RESPONSIVE BUY";
         signal.entry = ask;
         signal.sl =
             val - step * 4.0;
         signal.tp = poc;
      }
   }
   //====================================================
   // RESPONSIVE SELL
   //====================================================
   if (balancedMarket)
   {
      bool location =
          (bid >= vah - step * 2.0);
      bool rejection =
          (bearishDisplacement ||
           bearishFVG);
      bool absorption =
          (buyerExhaustion ||
           failedAuctionUp);
      if (
          location &&
          rejection &&
          absorption)
      {
         signal.sell = true;
         signal.responsive = true;
         signal.absorption = true;
         signal.priority =
             PRIORITY_STRONG;
         signal.state =
             EXEC_CONFIRM;
         signal.score += 5.0;
         signal.reasons +=
             "RESPONSIVE_SELL|";
         if (bearishFVG)
         {
            signal.score += 1.0;
            signal.reasons +=
                "FVG|";
         }
         signal.type =
             "RESPONSIVE SELL";
         signal.entry = bid;
         signal.sl =
             vah + step * 4.0;
         signal.tp = poc;
      }
   }
   //====================================================
   // INITIATIVE BUY
   //====================================================
   if (trendingUP)
   {
      bool continuation =
          (bullishDisplacement &&
           bullishDelta);
      bool acceptance =
          (acceptedUp &&
           pocMigrationUp);
      bool location =
          (bid >= vah - step * 2.0);
      if (
          continuation &&
          acceptance &&
          location)
      {
         signal.buy = true;
         signal.initiative = true;
         signal.priority =
             PRIORITY_INSTITUTIONAL;
         signal.state =
             EXEC_ENTRY;
         signal.score += 7.0;
         signal.reasons +=
             "INITIATIVE_BUY|";
         if (singlePrintBuy)
         {
            signal.score += 1.5;
            signal.singlePrint = true;
            signal.reasons +=
                "SINGLE_PRINT|";
         }
         if (openDriveBuy)
         {
            signal.score += 2.0;
            signal.reasons +=
                "OPEN_DRIVE|";
         }
         signal.type =
             "INITIATIVE BUY";
         signal.entry = ask;
         signal.sl =
             vah - valueRange * 0.30;
         signal.tp =
             signal.entry +
             valueRange * 2.5;
      }
   }
   //====================================================
   // INITIATIVE SELL
   //====================================================
   if (trendingDOWN)
   {
      bool continuation =
          (bearishDisplacement &&
           bearishDelta);
      bool acceptance =
          (acceptedDown &&
           pocMigrationDown);
      bool location =
          (bid <= val + step * 2.0);
      if (
          continuation &&
          acceptance &&
          location)
      {
         signal.sell = true;
         signal.initiative = true;
         signal.priority =
             PRIORITY_INSTITUTIONAL;
         signal.state =
             EXEC_ENTRY;
         signal.score += 7.0;
         signal.reasons +=
             "INITIATIVE_SELL|";
         if (singlePrintSell)
         {
            signal.score += 1.5;
            signal.singlePrint = true;
            signal.reasons +=
                "SINGLE_PRINT|";
         }
         if (openDriveSell)
         {
            signal.score += 2.0;
            signal.reasons +=
                "OPEN_DRIVE|";
         }
         signal.type =
             "INITIATIVE SELL";
         signal.entry = bid;
         signal.sl =
             val + valueRange * 0.30;
         signal.tp =
             signal.entry -
             valueRange * 2.5;
      }
   }
   //====================================================
   // FALSE BREAK SELL
   //====================================================
   if (falseBreakoutSell)
   {
      signal.sell = true;
      signal.falseBreakout = true;
      signal.priority =
          PRIORITY_INSTITUTIONAL;
      signal.score += 8.0;
      signal.reasons +=
          "FALSE_BREAK_SELL|";
      if (liquidationSell)
      {
         signal.score += 2.0;
         signal.liquidation = true;
         signal.reasons +=
             "LIQUIDATION|";
      }
      if (bearishDeltaDivergence)
      {
         signal.score += 1.5;
         signal.deltaDivergence = true;
         signal.reasons +=
             "DELTA_DIVERGENCE|";
      }
      signal.type =
          "FALSE BREAK SELL";
      signal.entry = bid;
      signal.sl =
          high0 + step * 2.0;
      signal.tp = poc;
   }
   //====================================================
   // FALSE BREAK BUY
   //====================================================
   if (falseBreakoutBuy)
   {
      signal.buy = true;
      signal.falseBreakout = true;
      signal.priority =
          PRIORITY_INSTITUTIONAL;
      signal.score += 8.0;
      signal.reasons +=
          "FALSE_BREAK_BUY|";
      if (liquidationBuy)
      {
         signal.score += 2.0;
         signal.liquidation = true;
         signal.reasons +=
             "LIQUIDATION|";
      }
      if (bullishDeltaDivergence)
      {
         signal.score += 1.5;
         signal.deltaDivergence = true;
         signal.reasons +=
             "DELTA_DIVERGENCE|";
      }
      signal.type =
          "FALSE BREAK BUY";
      signal.entry = ask;
      signal.sl =
          low0 - step * 2.0;
      signal.tp = poc;
   }
   //====================================================
   // LVN REJECTION BUY
   //====================================================
   if (lvnRejectBuy)
   {
      signal.buy = true;
      signal.priority =
          PRIORITY_STRONG;
      signal.score += 6.0;
      signal.reasons +=
          "LVN_REJECT_BUY|";
      signal.type =
          "LVN REJECTION BUY";
      signal.entry = ask;
      signal.sl =
          lvn - step * 3.0;
      signal.tp = poc;
   }
   //====================================================
   // LVN REJECTION SELL
   //====================================================
   if (lvnRejectSell)
   {
      signal.sell = true;
      signal.priority =
          PRIORITY_STRONG;
      signal.score += 6.0;
      signal.reasons +=
          "LVN_REJECT_SELL|";
      signal.type =
          "LVN REJECTION SELL";
      signal.entry = bid;
      signal.sl =
          lvn + step * 3.0;
      signal.tp = poc;
   }
   //====================================================
   // INVENTORY CORRECTION BUY
   //====================================================
   if (inventoryCorrectionBuy)
   {
      signal.buy = true;
      signal.inventoryCorrection = true;
      signal.priority =
          PRIORITY_STRONG;
      signal.score += 7.0;
      signal.reasons +=
          "SHORT_COVERING|";
      signal.type =
          "INVENTORY BUY";
      signal.entry = ask;
      signal.sl =
          val - step * 3.0;
      signal.tp = vah;
   }
   //====================================================
   // INVENTORY CORRECTION SELL
   //====================================================
   if (inventoryCorrectionSell)
   {
      signal.sell = true;
      signal.inventoryCorrection = true;
      signal.priority =
          PRIORITY_STRONG;
      signal.score += 7.0;
      signal.reasons +=
          "LONG_LIQUIDATION|";
      signal.type =
          "INVENTORY SELL";
      signal.entry = bid;
      signal.sl =
          vah + step * 3.0;
      signal.tp = val;
   }
   //====================================================
   // CONFLICT FILTER
   // If both directions fired, the signal is ambiguous — reject.
   //====================================================
   if (signal.buy && signal.sell)
      return;
   //====================================================
   // FINAL SCORE FILTER
   //====================================================
   if (signal.score < 5.0)
      return;
   //====================================================
   // R:R FILTER
   //====================================================
   double rrBuy =
       (signal.tp -
        signal.entry) /
       MathMax(
           signal.entry -
               signal.sl,
           _Point);
   double rrSell =
       (signal.entry -
        signal.tp) /
       MathMax(
           signal.sl -
               signal.entry,
           _Point);
   if (signal.buy && rrBuy < 1.5)
      return;
   if (signal.sell && rrSell < 1.5)
      return;
   //====================================================
   // HTF BIAS FILTER
   // htfBias == 1  → bullish: block sell signals
   // htfBias == -1 → bearish: block buy signals
   // htfBias == 0  → neutral: allow both directions
   //====================================================
   if (htfBias == 1 && signal.sell)
      return;
   if (htfBias == -1 && signal.buy)
      return;
   //====================================================
   // FINAL VALIDATION
   //====================================================
   if (signal.buy)
   {
      if (signal.sl >= signal.entry)
         return;
      if (signal.tp <= signal.entry)
         return;
      buySignal = true;
   }
   if (signal.sell)
   {
      if (signal.sl <= signal.entry)
         return;
      if (signal.tp >= signal.entry)
         return;
      sellSignal = true;
   }
   //====================================================
   // SIGNAL TYPE
   //====================================================
   signalType =
       signal.type +
       " | " +
       signal.reasons;
   if (signal.type == "")
      return;
   //====================================================
   // CLEAN OBJECTS
   //====================================================
   ObjectDelete(0, Name("STRAT_ENTRY"));
   ObjectDelete(0, Name("STRAT_ENTRY_LINE"));
   ObjectDelete(0, Name("STRAT_SL"));
   ObjectDelete(0, Name("STRAT_TP"));
   ObjectDelete(0, Name("STRAT_TEXT"));
   //====================================================
   // VISUALIZATION
   //====================================================
   datetime t1 =
       TimeCurrent();
   datetime t2 =
       t1 +
       (PeriodSeconds(LTF) * 20);
   //====================================================
   // ENTRY ZONE
   //====================================================
   ObjectCreate(
       0,
       Name("STRAT_ENTRY"),
       OBJ_RECTANGLE,
       0,
       t1,
       signal.entry + spread,
       t2,
       signal.entry - spread);
   ObjectSetInteger(
       0,
       Name("STRAT_ENTRY"),
       OBJPROP_FILL,
       true);
   ObjectSetInteger(
       0,
       Name("STRAT_ENTRY"),
       OBJPROP_BACK,
       true);
   ObjectSetInteger(
       0,
       Name("STRAT_ENTRY"),
       OBJPROP_COLOR,
       signal.buy
           ? clrLime
           : clrTomato);
   //====================================================
   // ENTRY LINE
   //====================================================
   ObjectCreate(
       0,
       Name("STRAT_ENTRY_LINE"),
       OBJ_HLINE,
       0,
       0,
       signal.entry);
   //====================================================
   // STOP LOSS
   //====================================================
   ObjectCreate(
       0,
       Name("STRAT_SL"),
       OBJ_HLINE,
       0,
       0,
       signal.sl);
   ObjectSetInteger(
       0,
       Name("STRAT_SL"),
       OBJPROP_COLOR,
       clrRed);
   //====================================================
   // TAKE PROFIT
   //====================================================
   ObjectCreate(
       0,
       Name("STRAT_TP"),
       OBJ_HLINE,
       0,
       0,
       signal.tp);
   ObjectSetInteger(
       0,
       Name("STRAT_TP"),
       OBJPROP_COLOR,
       clrDeepSkyBlue);
   //====================================================
   // SIGNAL TEXT
   //====================================================
   string txt =
       signal.type +
       "\nSTATE : " +
       IntegerToString(signal.state) +
       "\nSCORE : " +
       DoubleToString(signal.score, 1) +
       "\nENTRY : " +
       DoubleToString(signal.entry, _Digits) +
       "\nSL : " +
       DoubleToString(signal.sl, _Digits) +
       "\nTP : " +
       DoubleToString(signal.tp, _Digits) +
       "\nREASONS : " +
       signal.reasons;
   //====================================================
   // TEXT
   //====================================================
   ObjectCreate(
       0,
       "STRAT_TEXT",
       OBJ_TEXT,
       0,
       t2,
       signal.entry);
   ObjectSetString(
       0,
       "STRAT_TEXT",
       OBJPROP_TEXT,
       txt);
   ObjectSetInteger(
       0,
       "STRAT_TEXT",
       OBJPROP_COLOR,
       clrWhite);
   //====================================================
   // DEBUG
   //====================================================
   DrawSMCDebug(
       signal.type,
       TimeCurrent(),
       signal.entry,
       signal.buy
           ? ColorToPRGB(clrLime, 65)
           : ColorToPRGB(clrTomato, 65),
       signal.buy
           ? 233
           : 234);
   //====================================================
   // REFRESH
   //====================================================
   ChartRedraw();
}

void OnTick()
{
   if (!IsNewLTFBar())
      return;
   double atr = GetATR(1);
   double autoSL = 0.0;
   double entry = iClose(_Symbol, LTF, 3);
   int bias = GetHTFBias();
   if (bias == 1)
   {
      autoSL =
          entry -
          (atr * 2.0);
   }
   else
   {
      autoSL =
          entry +
          (atr * 2.0);
   }
   autoSL =
       NormalizeDouble(
           autoSL,
           _Digits);
   DrawLineCu(Name("SL Info"), autoSL, clrSilver, STYLE_DASH, iTime(_Symbol, LTF, 3), iTime(_Symbol, LTF, 0) + PeriodSeconds(LTF) * 20);
   // Tick-level: entry engine + exit management (must run every tick)
  
   Init();
   TradeEngine(CachedBullEvent, CachedBearEvent);
 
   ManageScalingExit();
   ManageDynamicTPScaling();
}

void CloseAllOrders()
{
   // Loop semua posisi terbuka
   for (int i = PositionsTotal() - 1; i >= 0; i--)
   {
      string symbol;
      ulong ticket;
      if ((ticket = PositionGetTicket(i)) > 0)
      {
         symbol = PositionGetString(POSITION_SYMBOL);
         if (symbol != _Symbol)
            continue;
         double volume = PositionGetDouble(POSITION_VOLUME);
         // Tutup posisi
         if (!trade.PositionClose(ticket))
         {
            Print("Gagal menutup posisi: ", symbol, " Ticket: ", ticket, " Error: ", GetLastError());
         }
         else
         {
            Print("Berhasil menutup posisi: ", symbol, " Ticket: ", ticket);
         }
      }
   }
}

{

   double Ask = SymbolInfoDouble(_Symbol, SYMBOL_ASK);
   double price = (Bid + Ask) * 0.5;
   double atr = GetATR(1);
   int htfBias = GetHTFBias();

   string biasStr = (htfBias == 1) ? "▲ BULLISH" : (htfBias == -1) ? "▼ BEARISH"
                                                                   : "◆ NEUTRAL";
   color biasColor = (htfBias == 1) ? clrLimeGreen : (htfBias == -1) ? clrOrangeRed
                                                                     : clrGray;

   string regimeStr = "";
   if (regime == REGIME_TREND_BULL)
      regimeStr = "TREND BULL";
   else if (regime == REGIME_TREND_BEAR)
      regimeStr = "TREND BEAR";
   else if (regime == REGIME_EXPANSION)
      regimeStr = "EXPANSION";
   else if (regime == REGIME_COMPRESSION)
      regimeStr = "COMPRESSION";
   else if (regime == REGIME_RANGE)
      regimeStr = "RANGE";
   else
      regimeStr = "UNKNOWN";

   SMCEvent event = (htfBias == 1 ? CachedBullEvent : CachedBearEvent);
   string eventType = (htfBias == 1 ? "BUY" : "SELL");
   string zoneType = event.hasFVG ? "FVG" : event.hasOB ? "OB"
                                                        : "NONE";
   string entryState = event.entryReady ? "✔ READY" : event.valid ? "✓ VALID"
                                                                  : "✗ WAIT";

   bool safe = IsSafeToTrade();
   int totalPos = TotalTrade();
   double totalProfit = 0;

   for (int i = 0; i < PositionsTotal(); i++)
   {
      ulong ticket = PositionGetTicket(i);
      if (!PositionSelectByTicket(ticket))
         continue;
      if (PositionGetString(POSITION_SYMBOL) != _Symbol)
         continue;
      totalProfit += PositionGetDouble(POSITION_PROFIT);
   }

   MqlDateTime dt;
   TimeToStruct(GetUTCTime(), dt);
   int utcHour = dt.hour;
   int utcMin = dt.min;

   string sessionActive = "";
   if (asiaKillzone)
      sessionActive = "⚡ ASIA KZ";
   else if (londonKillzone)
      sessionActive = "⚡ LONDON KZ";
   else if (newYorkKillzone)
      sessionActive = "⚡ NEW YORK KZ";
   else if (asiaSession)
      sessionActive = "ASIA";
   else if (londonSession)
      sessionActive = "LONDON";
   else if (newYorkSession)
      sessionActive = "NEW YORK";
   else
      sessionActive = "OFF";

   int x = 12;
   int y = 10;
   int panelWidth = 380;
   int lineHeight = 20;

   //====================================================
   // HEADER PANEL (Symbol + HTF Bias)
   //====================================================
   DrawDashboardHeader(x, y, panelWidth, biasStr, biasColor);
   y += 35;

   //====================================================
   // MARKET DATA SECTION
   //====================================================
   DrawDashboardSection(x, y, panelWidth, "MARKET", clrDarkGray);
   y += 20;

   DrawDashboardRow(x + 8, y, "Price", DoubleToString(price, _Digits), clrWhite, panelWidth - 16);
   y += lineHeight;
   DrawDashboardRow(x + 8, y, "Bid/Ask", DoubleToString(Bid, _Digits) + " / " + DoubleToString(Ask, _Digits), clrLightGray, panelWidth - 16);
   y += lineHeight;
   DrawDashboardRow(x + 8, y, "ATR", DoubleToString(atr, _Digits), clrWhite, panelWidth - 16);
   y += lineHeight;

   string vpPos = (price > vah ? "ABOVE (Premium)" : price < val ? "BELOW (Discount)"
                                                                 : "INSIDE (Fair)");
   DrawDashboardRow(x + 8, y, "VA", vpPos, clrYellow, panelWidth - 16);
   y += lineHeight + 5;

   //====================================================
   // HTF BIAS SECTION
   //====================================================
   DrawDashboardSection(x, y, panelWidth, "HTF BIAS", biasColor);
   y += 20;

   DrawDashboardRow(x + 8, y, "Bias", biasStr, biasColor, panelWidth - 16);
   y += lineHeight;
   DrawDashboardRow(x + 8, y, "Regime", regimeStr + StringFormat(" (%.2f)", regimeStrength), clrWhite, panelWidth - 16);
   y += lineHeight + 5;

   //====================================================

//+------------------------------------------------------------------+
datetime g_lastBuyEntry = 0;
datetime g_lastSellEntry = 0;

//+------------------------------------------------------------------+
//| CHECK ENTRY FREEZE                                               |
//+------------------------------------------------------------------+
bool IsEntryFrozen(bool isBuy, int freezeSeconds = 90)
{
   datetime now = TimeCurrent();
   if (isBuy)
   {
      if ((now - g_lastBuyEntry) < freezeSeconds)
         return true;
   }
   else
   {
      if ((now - g_lastSellEntry) < freezeSeconds)
         return true;
   }
   return false;
}

//+------------------------------------------------------------------+
//| CHECK SAME DIRECTION POSITION                                    |
//+------------------------------------------------------------------+
bool HasPositionDirection(bool isBuy)
{
   for (int i = PositionsTotal() - 1; i >= 0; i--)
   {
      ulong ticket = PositionGetTicket(i);
      if (!PositionSelectByTicket(ticket))
         continue;
      if (PositionGetString(POSITION_SYMBOL) != _Symbol)
         continue;
      ENUM_POSITION_TYPE type =
          (ENUM_POSITION_TYPE)
              PositionGetInteger(POSITION_TYPE);
      if (isBuy && type == POSITION_TYPE_BUY)
         return true;
      if (!isBuy && type == POSITION_TYPE_SELL)
         return true;
   }
   return false;
}

//+------------------------------------------------------------------+
//| CHECK DUPLICATE ZONE ENTRY                                       |
//+------------------------------------------------------------------+
bool HasNearbyPosition(
    bool isBuy,
    double zoneLow,
    double zoneHigh,
    double atr)
{
   double mid =
       (zoneLow +
        zoneHigh) *
       0.5;
   for (int i = PositionsTotal() - 1; i >= 0; i--)
   {
      ulong ticket = PositionGetTicket(i);
      if (!PositionSelectByTicket(ticket))
         continue;
      if (PositionGetString(POSITION_SYMBOL) != _Symbol)
         continue;
      ENUM_POSITION_TYPE type =
          (ENUM_POSITION_TYPE)
              PositionGetInteger(POSITION_TYPE);
      bool sameDirection =
          ((isBuy && type == POSITION_TYPE_BUY) ||
           (!isBuy && type == POSITION_TYPE_SELL));
      if (!sameDirection)
         continue;
      double entry =
          PositionGetDouble(
              POSITION_PRICE_OPEN);
      double distance =
          MathAbs(entry - mid);
      //------------------------------------------------
      // POSITION ALREADY INSIDE SAME DELIVERY ZONE
      //------------------------------------------------
      if (distance < atr * 0.50)
         return true;
   }
   return false;
}

//+------------------------------------------------------------------+
//| TRADE FLOW (shared buy/bear pipeline)                            |
//+------------------------------------------------------------------+
void ExecuteTradeFlow(bool isBuy, SMCEvent &e, double ask, double bid, double atrFast)
{
   //----------------------------------------------------
   // ENTRY FREEZE + DIRECTION FILTER (symmetric)
   //----------------------------------------------------
   if (IsEntryFrozen(isBuy))
      return;
   if (HasPositionDirection(isBuy))
      return;
   //----------------------------------------------------
   // DELIVERY ZONE
   //----------------------------------------------------
   double zLow = e.hasFVG ? e.fvgBottom : e.obLow;
   double zHigh = e.hasFVG ? e.fvgTop : e.obHigh;
   if (zLow <= 0 || zHigh <= zLow)
      return;
   if (HasNearbyPosition(isBuy, zLow, zHigh, atrFast))
      return;
   //----------------------------------------------------
   // ZONE VISUAL
   //----------------------------------------------------
   color zoneClr = isBuy ? clrLime : clrRed;
   DrawLine(isBuy ? "BUY_ZONE_LOW" : "SELL_ZONE_LOW", zLow, zoneClr);
   DrawLine(isBuy ? "BUY_ZONE_HIGH" : "SELL_ZONE_HIGH", zHigh, zoneClr);
   //----------------------------------------------------
   // MITIGATION: buy checks bid (fill side), sell checks ask
   //----------------------------------------------------
   double checkPrice = isBuy ? bid : ask;
   bool mitigation = (checkPrice >= zLow && checkPrice <= zHigh);
   if (mitigation)
      DrawSMCDebug(isBuy ? "BUY_MITIGATION" : "SELL_MITIGATION",
                   TimeCurrent(), checkPrice, zoneClr, 221);
   //----------------------------------------------------
   // REJECTION: wick must be real (guard flat candles)
   //----------------------------------------------------
   double candleRange = ratesLTF[0].high - ratesLTF[0].low;
   bool rejection = false;
   if (candleRange > atrFast * 0.10)
   {
      if (isBuy)
      {
         double lowerWick = MathMin(ratesLTF[0].open, ratesLTF[0].close) - ratesLTF[0].low;
         rejection = (ratesLTF[0].close > ratesLTF[0].open) && lowerWick > candleRange * 0.30;
      }
      else
      {
         double upperWick = ratesLTF[0].high - MathMax(ratesLTF[0].open, ratesLTF[0].close);
         rejection = (ratesLTF[0].close < ratesLTF[0].open) && upperWick > candleRange * 0.30;
      }
   }
   if (rejection)
      DrawSMCDebug(isBuy ? "BUY_REJECTION" : "SELL_REJECTION",
                   TimeCurrent(), checkPrice, isBuy ? clrAqua : clrOrange, 222);
   //----------------------------------------------------
   // MOMENTUM: price breaks prior candle's extreme
   //----------------------------------------------------
   bool momentum = isBuy
                       ? (ratesLTF[0].close > ratesLTF[1].high)
                       : (ratesLTF[0].close < ratesLTF[1].low);
   //----------------------------------------------------
   // ENTRY SIGNAL: mitigation + rejection + momentum
   //----------------------------------------------------
   if (!(mitigation && rejection && momentum))
      return;
   DrawSMCDebug(isBuy ? "BUY_ENTRY" : "SELL_ENTRY",
                TimeCurrent(), isBuy ? ask : bid, clrWhite, isBuy ? 241 : 242);
   //----------------------------------------------------
   // BUILD LAYERS + EXECUTE
   //----------------------------------------------------
   double sl = isBuy ? zLow - atrFast * 0.20 : zHigh + atrFast * 0.20;
   EntryLayer layers[];
   int count = BuildEntryLayers(isBuy, zLow, zHigh, sl, layers);
   if (count <= 0)
      return;
   TPLadder tp = GetSmartTP(isBuy, layers[0].price, sl);
   bool entrySuccess = false;
   for (int i = 0; i < ArraySize(layers); i++)
   {
      if (!layers[i].active || layers[i].lot <= 0)
         continue;
      bool result = isBuy
                        ? trade.Buy(layers[i].lot, _Symbol, ask, 0, 0)
                        : trade.Sell(layers[i].lot, _Symbol, bid, 0, 0);
      if (result)
         entrySuccess = true;
   }
   if (entrySuccess)
   {
      if (isBuy)
         g_lastBuyEntry = TimeCurrent();
      else
         g_lastSellEntry = TimeCurrent();
   }
}
//+------------------------------------------------------------------+
//| TRADE ENGINE                                                     |
//+------------------------------------------------------------------+
void TradeEngine(SMCEvent &eBuy, SMCEvent &eSell)
{
   if (TotalTrade() >= maxTrade)
      return;
   double ask = SymbolInfoDouble(_Symbol, SYMBOL_ASK);
   double bid = SymbolInfoDouble(_Symbol, SYMBOL_BID);
   double atrFast = GetATR(1);
   if (atrFast <= 0)
      return;
   if ((ask - bid) > atrFast * 0.15)
      return;
   int htfBias = GetHTFBias();
   if (eBuy.valid && htfBias == 1)
      ExecuteTradeFlow(true, eBuy, ask, bid, atrFast);
   if (eSell.valid && htfBias == -1)
      ExecuteTradeFlow(false, eSell, ask, bid, atrFast);
}
//+------------------------------------------------------------------+
//| MARKET REGIME ENGINE                                             |
//+------------------------------------------------------------------+
MarketRegime DetectMarketRegime(
    double &strength)
{
   strength = 0;
   //------------------------------------------------
   // VOLATILITY (HTF): current ATR vs 14 bars ago
   //------------------------------------------------
   double atrNow = GetATR_HTF(1);
   double atrPrev = GetATR(14);
   if (atrNow <= 0 || atrPrev <= 0)
      return REGIME_UNKNOWN;
   //------------------------------------------------
   // EMA STRUCTURE (HTF) — cached handles
   // Created once per EA lifetime, never released mid-run.
   //------------------------------------------------
   static int h20 = INVALID_HANDLE;
   static int h50 = INVALID_HANDLE;
   if (h20 == INVALID_HANDLE)
      h20 = iMA(_Symbol, HTF, 20, 0, MODE_EMA, PRICE_CLOSE);
   if (h50 == INVALID_HANDLE)
      h50 = iMA(_Symbol, HTF, 50, 0, MODE_EMA, PRICE_CLOSE);
   if (h20 == INVALID_HANDLE || h50 == INVALID_HANDLE)
      return REGIME_UNKNOWN;
   double buf20[1], buf50[1];
   ArraySetAsSeries(buf20, true);
   ArraySetAsSeries(buf50, true);
   if (CopyBuffer(h20, 0, 0, 1, buf20) < 1)
      return REGIME_UNKNOWN;
   if (CopyBuffer(h50, 0, 0, 1, buf50) < 1)
      return REGIME_UNKNOWN;
   double ema20 = buf20[0];
   double ema50 = buf50[0];
   if (ema20 <= 0 || ema50 <= 0)
      return REGIME_UNKNOWN;
   //------------------------------------------------
   // HTF ATR — used as threshold for HTF EMA slope.
   // Must match the timeframe of the EMAs being compared.
   //------------------------------------------------
   double atrHTF = GetATR_HTF(1);
   if (atrHTF <= 0)
      return REGIME_UNKNOWN;
   double slope = MathAbs(ema20 - ema50);
   double volRatio = atrNow / atrPrev;
   //------------------------------------------------
   // TREND: HTF EMA gap significant relative to HTF ATR
   //------------------------------------------------
   if (slope > atrHTF * 0.5)
   {
      strength = slope / atrHTF;
      if (ema20 > ema50)
         return REGIME_TREND_BULL;
      if (ema20 < ema50)
         return REGIME_TREND_BEAR;
   }
   //------------------------------------------------
   // EXPANSION: LTF volatility surging vs recent baseline
   //------------------------------------------------
   if (volRatio > 1.4)
   {
      strength = volRatio;
      return REGIME_EXPANSION;
   }
   //------------------------------------------------
   // COMPRESSION: LTF volatility contracting
   //------------------------------------------------
   if (volRatio < 0.7)
   {
      strength = 1.0 - volRatio;
      return REGIME_COMPRESSION;
   }
   //------------------------------------------------
   // RANGE
   //------------------------------------------------
   strength = 0.5;
   return REGIME_RANGE;
}
//+------------------------------------------------------------------+
//| BULLISH EVENT ENGINE                                             |
//| FULL PIPELINE REFACTOR                                           |
//+------------------------------------------------------------------+
//+------------------------------------------------------------------+
//| SMC EVENT CORE                                                   |
//| Shared pipeline for both bull and bear events.                   |
//| isBull=true → bullish setup, isBull=false → bearish setup        |
//+------------------------------------------------------------------+
bool BuildSMCEventCore(SMCEvent &s, bool isBull)
{
   ZeroMemory(s);
   EventScore score;
   ResetScore(score);
   //====================================================
   // VALIDATION
   //====================================================
   int bars = ArraySize(ratesLTF);
   if (bars < 150)
      return false;
   double atr = GetATR(1);
   if (atr <= 0)
      return false;
   double currentPrice = ratesLTF[0].close;
   double bufATR = atr * 0.05;
   //====================================================
   // HTF RANGE
   //====================================================
   SwingPoint htfHigh, htfLow;
   if (!GetLastSwings(ratesHTF, ArraySize(ratesHTF), htfHigh, htfLow))
      return false;
   s.dealingRangeHigh = htfHigh.price;
   s.dealingRangeLow = htfLow.price;
   double equilibrium = (s.dealingRangeHigh + s.dealingRangeLow) * 0.5;
   //====================================================
   // REGIME
   // bull: price in discount (below EQ) → confluence
   // bear: price in premium (above EQ)  → confluence
   //====================================================
   bool favorableSide = isBull ? (currentPrice < equilibrium)
                               : (currentPrice >= equilibrium);
   s.isReversal = favorableSide;
   if (favorableSide)
      score.confluence += 1.5;
   //====================================================
   // LIQUIDITY
   // bull: look for strongest swing LOW below price
   // bear: look for strongest swing HIGH above price
   //====================================================
   double liquidityLevel = isBull ? DBL_MAX : -DBL_MAX;
   int liquidityShift = -1;
   double bestStrength = 0.0;
   for (int i = 5; i < 60 && i + 1 < bars; i++)
   {
      double lvl = isBull ? ratesLTF[i].low : ratesLTF[i].high;
      bool isSwing = isBull
                         ? (ratesLTF[i].low < ratesLTF[i - 1].low && ratesLTF[i].low < ratesLTF[i + 1].low)
                         : (ratesLTF[i].high > ratesLTF[i - 1].high && ratesLTF[i].high > ratesLTF[i + 1].high);
      if (!isSwing)
         continue;
      double sRight = MathAbs(lvl - (isBull ? ratesLTF[i + 1].low : ratesLTF[i + 1].high));
      double sLeft = MathAbs(lvl - (isBull ? ratesLTF[i - 1].low : ratesLTF[i - 1].high));
      double strength = MathMin(sRight, sLeft);
      if (strength < bufATR)
         continue;
      if (strength > bestStrength)
      {
         bestStrength = strength;
         liquidityLevel = lvl;
         liquidityShift = i;
      }
   }
   if (liquidityShift == -1)
      return false;
   // pool must sit on the correct side of current price
   if (isBull ? liquidityLevel >= currentPrice : liquidityLevel <= currentPrice)
      return false;
   s.liquidity = liquidityLevel;
   s.liquidityShift = liquidityShift;
   score.liquidity += 2.0;
   //====================================================
   // SWEEP
   // bull: wick below liq.low, close back above
   // bear: wick above liq.high, close back below
   //====================================================
   bool sweepFound = false;
   int sweepLimit = MathMin(liquidityShift - 1, 30);
   for (int i = sweepLimit; i >= 1; i--)
   {
      bool swept = isBull
                       ? (ratesLTF[i].low < liquidityLevel && ratesLTF[i].close > liquidityLevel)
                       : (ratesLTF[i].high > liquidityLevel && ratesLTF[i].close < liquidityLevel);
      double rejection = isBull
                             ? (liquidityLevel - ratesLTF[i].low)
                             : (ratesLTF[i].high - liquidityLevel);
      if (swept && rejection > atr * 0.08)
      {
         sweepFound = true;
         s.sweepShift = i;
         score.sweep += 2.5;
         break;
      }
   }
   if (!sweepFound)
      return false;
   //====================================================
   // DISPLACEMENT
   // bull: impulsive bullish candle after sweep, closes above liq
   // bear: impulsive bearish candle after sweep, closes below liq
   //====================================================
   bool displacementFound = false;
   double bestImpulse = 0.0;
   for (int i = s.sweepShift - 1; i >= 1 && (s.sweepShift - i) <= 10; i--)
   {
      double range = ratesLTF[i].high - ratesLTF[i].low;
      double body = MathAbs(ratesLTF[i].close - ratesLTF[i].open);
      if (range <= 0)
         continue;
      double ratio = body / range;
      bool correctDir = isBull ? (ratesLTF[i].close > ratesLTF[i].open)
                               : (ratesLTF[i].close < ratesLTF[i].open);
      bool impulsive = (range > atr * 0.55 && ratio > 0.60 && correctDir);
      bool cleanSweep = isBull ? (ratesLTF[i].close > liquidityLevel + bufATR)
                               : (ratesLTF[i].close < liquidityLevel - bufATR);
      if (!impulsive || !cleanSweep)
         continue;
      double impulseScore = range * ratio;
      if (impulseScore > bestImpulse)
      {
         bestImpulse = impulseScore;
         displacementFound = true;
         s.displacementShift = i;
      }
   }
   if (!displacementFound)
      return false;
   score.displacement += 3.0;
   //====================================================
   // PROTECTED LEVEL
   // In as-series: "before sweep" = higher index (older).
   // bull: look for swing HIGH before the sweep
   // bear: look for swing LOW  before the sweep
   //====================================================
   double protectedLevel = isBull ? 0.0 : DBL_MAX;
   for (int i = s.sweepShift + 1; i <= s.sweepShift + 25 && i + 1 < bars; i++)
   {
      double lvl = isBull ? ratesLTF[i].high : ratesLTF[i].low;
      bool isSwing = isBull
                         ? (ratesLTF[i].high > ratesLTF[i - 1].high && ratesLTF[i].high > ratesLTF[i + 1].high)
                         : (ratesLTF[i].low < ratesLTF[i - 1].low && ratesLTF[i].low < ratesLTF[i + 1].low);
      if (!isSwing)
         continue;
      double lDiff = MathAbs(lvl - (isBull ? ratesLTF[i + 1].high : ratesLTF[i + 1].low));
      double rDiff = MathAbs(lvl - (isBull ? ratesLTF[i - 1].high : ratesLTF[i - 1].low));
      if (lDiff < bufATR && rDiff < bufATR)
         continue;
      protectedLevel = lvl;
      break;
   }
   bool protectedInvalid = isBull ? (protectedLevel <= 0)
                                  : (protectedLevel == DBL_MAX || protectedLevel <= 0);
   if (protectedInvalid)
      return false;
   if (isBull)
      s.protectedHigh = protectedLevel;
   else
      s.protectedLow = protectedLevel;
   //====================================================
   // CHOCH
   // bull: close breaks above protectedHigh
   // bear: close breaks below protectedLow
   //====================================================
   bool chochFound = false;
   for (int i = s.displacementShift; i >= 0; i--)
   {
      bool breakClose = isBull
                            ? (ratesLTF[i].close > protectedLevel + atr * 0.03)
                            : (ratesLTF[i].close < protectedLevel - atr * 0.03);
      if (breakClose)
      {
         chochFound = true;
         s.chochShift = i;
         score.choch += 3.0;
         break;
      }
   }
   if (!chochFound)
      return false;
   //====================================================
   // ORDER BLOCK (ICT standard)
   //
   // Definition: the LAST opposing candle immediately
   // before the displacement move begins.
   //   bull OB = last BEARISH candle closest to displacement
   //   bear OB = last BULLISH candle closest to displacement
   //
   // Scan direction: displacementShift+1 → sweepShift
   // (low index = newer in as-series, so we scan from
   //  newest to oldest and break on first match —
   //  guaranteeing we find the candle closest to displacement)
   //
   // Zone = candle BODY (open/close), not wicks.
   // Wicks are noise; institutions place orders at the
   // price the candle opened/closed, not at wick extremes.
   //====================================================
   for (int i = s.displacementShift + 1; i <= s.sweepShift && i > 0 && i < bars; i++)
   {
      if (i - 1 < 0)
         continue;
      bool correctOBDir = isBull ? (ratesLTF[i].close < ratesLTF[i].open)
                                 : (ratesLTF[i].close > ratesLTF[i].open);
      if (!correctOBDir)
         continue;
      // The NEXT (newer) candle must break out from this OB candle,
      // confirming it as a genuine institutional order zone.
      bool confirmOB = isBull
                           ? (ratesLTF[i - 1].close > ratesLTF[i].high || ratesLTF[i - 1].high > ratesLTF[i].high + bufATR)
                           : (ratesLTF[i - 1].close < ratesLTF[i].low || ratesLTF[i - 1].low < ratesLTF[i].low - bufATR);
      if (confirmOB)
      {
         s.hasOB = true;
         s.obLow = MathMin(ratesLTF[i].open, ratesLTF[i].close);
         s.obHigh = MathMax(ratesLTF[i].open, ratesLTF[i].close);
         s.obShift = i;
         score.delivery += 2.0;
         break;
      }
   }
   //====================================================
   // FVG (fresh-first, mitigated-fallback)
   //
   // Standard 3-candle imbalance:
   //   bull: candle[i].low > candle[i+2].high  → gap exists above
   //   bear: candle[i].high < candle[i+2].low  → gap exists below
   //
   // Restricted to post-sweep zone (i+2 < sweepShift) so
   // we only consider imbalances created by THIS setup's
   // displacement, not older unrelated moves.
   //====================================================
   double fallbackTop = 0.0, fallbackBottom = 0.0, fallbackDist = DBL_MAX;
   bool fvgFound = false;
   for (int i = 1; i + 2 < bars; i++)
   {
      // Only scan within the post-sweep displacement window
      if (i + 2 >= s.sweepShift)
         break;
      double fvgTop, fvgBottom;
      if (isBull)
      {
         fvgBottom = ratesLTF[i + 2].high;
         fvgTop = ratesLTF[i].low;
      }
      else
      {
         fvgTop = ratesLTF[i + 2].low;
         fvgBottom = ratesLTF[i].high;
      }
      double gap = fvgTop - fvgBottom;
      if (gap <= bufATR)
         continue;
      bool validSide = isBull ? (fvgBottom >= liquidityLevel) : (fvgTop <= liquidityLevel);
      if (!validSide)
         continue;
      bool mitigated = IsFVGMitigated(fvgBottom, fvgTop, atr, isBull);
      double dist = MathAbs(currentPrice - (fvgTop + fvgBottom) * 0.5);
      if (!mitigated)
      {
         s.hasFVG = true;
         s.fvgTop = fvgTop;
         s.fvgBottom = fvgBottom;
         fvgFound = true;
         break;
      }
      else if (dist < fallbackDist)
      {
         fallbackDist = dist;
         fallbackTop = fvgTop;
         fallbackBottom = fvgBottom;
      }
   }
   if (!fvgFound && fallbackDist < DBL_MAX)
   {
      s.hasFVG = true;
      s.fvgTop = fallbackTop;
      s.fvgBottom = fallbackBottom;
      fvgFound = true;
   }
   if (fvgFound)
      score.delivery += 2.0;
   //====================================================
   // MITIGATION
   //====================================================
   double zoneLow = s.hasFVG ? s.fvgBottom : s.obLow;
   double zoneHigh = s.hasFVG ? s.fvgTop : s.obHigh;
   bool hasZone = (zoneLow > 0 && zoneHigh > zoneLow);
   s.mitigated = hasZone && (currentPrice >= zoneLow && currentPrice <= zoneHigh);
   if (s.mitigated)
      score.mitigation += 2.0;
   //====================================================
   // MOMENTUM
   // bull: current close > previous high, body is bullish
   // bear: current close < previous low,  body is bearish
   //====================================================
   double bodySize = isBull ? (ratesLTF[0].close - ratesLTF[0].open)
                            : (ratesLTF[0].open - ratesLTF[0].close);
   bool momentum = isBull
                       ? (ratesLTF[0].close > ratesLTF[1].high && bodySize > 0 && bodySize > atr * 0.06)
                       : (ratesLTF[0].close < ratesLTF[1].low && bodySize > 0 && bodySize > atr * 0.06);
   s.reacceleration = momentum;
   if (momentum)
      score.momentum += 2.0;
   //====================================================
   // FINAL
   //====================================================
   s.strength = (int)MathRound(FinalizeScore(score));
   s.valid = (s.strength >= 10 && (s.hasOB || s.hasFVG));
   bool inZone = s.hasFVG
                     ? (currentPrice >= s.fvgBottom && currentPrice <= s.fvgTop)
                     : (currentPrice >= s.obLow && currentPrice <= s.obHigh);
   bool confirm = inZone && s.chochShift >= 0 &&
                  (isBull ? ConfirmBullishCandle(1) : ConfirmBearishCandle(1));
   s.entryReady = (s.valid && s.mitigated && s.reacceleration && inZone && confirm);
   s.eventShift = s.sweepShift;
   s.eventTime = ratesLTF[s.sweepShift].time;
   s.isBullish = isBull;
   return s.valid;
}

//| SESSION STRUCTURE ENGINE (UTC VERSION)                           |
//| ASIA - LONDON - NEW YORK                                         |
//| INSTITUTIONAL SESSION ENGINE                                     |
//+------------------------------------------------------------------+

//====================================================
// SESSION STATES
//====================================================
bool asiaSession = false;
bool londonSession = false;
bool newYorkSession = false;

bool asiaKillzone = false;
bool londonKillzone = false;
bool newYorkKillzone = false;

//====================================================
// SESSION LEVELS
//====================================================
double asiaHigh = 0.0;
double asiaLow = 0.0;

double londonHigh = 0.0;
double londonLow = 0.0;

double newYorkHigh = 0.0;
double newYorkLow = 0.0;
//====================================================
// UTC SESSION DEFINITIONS
//====================================================
//
// ASIA SESSION
// 00:00 → 05:30 UTC
//
// LONDON SESSION
// 07:30 → 15:00 UTC
//
// NEW YORK SESSION
// 13:30 → 19:30 UTC
//
//====================================================

//+------------------------------------------------------------------+
//| DETECT ACTIVE SESSION UTC                                        |
//+------------------------------------------------------------------+
void DetectTradingSessionUTC()
{
   //====================================================
   // RESET
   //====================================================
   asiaSession = false;
   londonSession = false;
   newYorkSession = false;
   asiaKillzone = false;
   londonKillzone = false;
   newYorkKillzone = false;
   //====================================================
   // UTC TIME
   //====================================================
   datetime utcNow =
       TimeGMT();
   MqlDateTime tm;
   TimeToStruct(
       utcNow,
       tm);
   int minutes =
       (tm.hour * 60) +
       tm.min;
   //====================================================
   // ASIA SESSION
   // 00:00 → 05:30 UTC
   //====================================================
   if (
       minutes >= 0 &&
       minutes < 330)
   {
      asiaSession = true;
   }
   //====================================================
   // LONDON SESSION
   // 07:30 → 15:00 UTC
   //====================================================
   if (
       minutes >= 450 &&
       minutes < 900)
   {
      londonSession = true;
   }
   //====================================================
   // NEW YORK SESSION
   // 13:30 → 19:30 UTC
   //====================================================
   if (
       minutes >= 810 &&
       minutes < 1170)
   {
      newYorkSession = true;
   }
   //====================================================
   // KILLZONES
   //====================================================
   //
   // ASIA KZ
   // 00:00 → 02:00 UTC
   //
   // LONDON KZ
   // 07:30 → 10:00 UTC
   //
   // NEW YORK KZ
   // 13:30 → 16:00 UTC
   //
   //====================================================
   //----------------------------------------------------
   // ASIA KILLZONE
   //----------------------------------------------------
   if (
       minutes >= 0 &&
       minutes < 120)
   {
      asiaKillzone = true;
   }
   //----------------------------------------------------
   // LONDON KILLZONE
   //----------------------------------------------------
   if (
       minutes >= 450 &&
       minutes < 600)
   {
      londonKillzone = true;
   }
   //----------------------------------------------------
   // NEW YORK KILLZONE
   //----------------------------------------------------
   if (
       minutes >= 810 &&
       minutes < 960)
   {
      newYorkKillzone = true;
   }
}

//+------------------------------------------------------------------+
//| BUILD SESSION RANGES UTC                                         |
//+------------------------------------------------------------------+
void BuildSessionRangesUTC()
{
   //====================================================
   // RESET
   //====================================================
   asiaHigh = -DBL_MAX;
   asiaLow = DBL_MAX;
   londonHigh = -DBL_MAX;
   londonLow = DBL_MAX;
   newYorkHigh = -DBL_MAX;
   newYorkLow = DBL_MAX;
   //====================================================
   // UTC DAY START
   //====================================================
   datetime utcNow =
       TimeGMT();
   string utcDate =
       TimeToString(
           utcNow,
           TIME_DATE);
   datetime utcDayStart =
       StringToTime(
           utcDate);
   //====================================================
   // SESSION WINDOWS UTC
   //====================================================
   //----------------------------------------------------
   // ASIA
   // 00:00 → 05:30
   //----------------------------------------------------
   asiaStart =
       utcDayStart + (0 * 3600);
   asiaEnd =
       utcDayStart + (5 * 3600) + (30 * 60);
   //----------------------------------------------------
   // LONDON
   // 07:30 → 15:00
   //----------------------------------------------------
   londonStart =
       utcDayStart + (7 * 3600) + (30 * 60);
   londonEnd =
       utcDayStart + (15 * 3600);
   //----------------------------------------------------
   // NEW YORK
   // 13:30 → 19:30
   //----------------------------------------------------
   nyStart =
       utcDayStart + (13 * 3600) + (30 * 60);
   nyEnd =
       utcDayStart + (19 * 3600) + (30 * 60);
   //====================================================
   // SERVER OFFSET
   //====================================================
   int offsetSeconds =
       (int)(TimeCurrent() -
             TimeGMT());
   //====================================================
   // CONVERT TO SERVER TIME
   //====================================================
   asiaStart += offsetSeconds;
   asiaEnd += offsetSeconds;
   londonStart += offsetSeconds;
   londonEnd += offsetSeconds;
   nyStart += offsetSeconds;
   nyEnd += offsetSeconds;
   //====================================================
   // BUILD SESSION RANGE
   //====================================================
   int bars =
       ArraySize(ratesORB);
   for (int i = 0; i < bars; i++)
   {
      datetime t =
          ratesORB[i].time;
      //----------------------------------------------------
      // ASIA
      //----------------------------------------------------
      if (
          t >= asiaStart &&
          t <= asiaEnd)
      {
         if (ratesORB[i].high > asiaHigh)
            asiaHigh = ratesORB[i].high;
         if (ratesORB[i].low < asiaLow)
            asiaLow = ratesORB[i].low;
      }
      //----------------------------------------------------
      // LONDON
      //----------------------------------------------------
      if (
          t >= londonStart &&
          t <= londonEnd)
      {
         if (ratesORB[i].high > londonHigh)
            londonHigh = ratesORB[i].high;
         if (ratesORB[i].low < londonLow)
            londonLow = ratesORB[i].low;
      }
      //----------------------------------------------------
      // NEW YORK
      //----------------------------------------------------
      if (
          t >= nyStart &&
          t <= nyEnd)
      {
         if (ratesORB[i].high > newYorkHigh)
            newYorkHigh = ratesORB[i].high;
         if (ratesORB[i].low < newYorkLow)
            newYorkLow = ratesORB[i].low;
      }
   }
}

//+------------------------------------------------------------------+
//| DRAW SESSION ZONES UTC                                           |
//+------------------------------------------------------------------+
void DrawSessionZonesUTC()
{
   //====================================================
   // CLEAN OBJECTS
   //====================================================
   ObjectsDeleteAll(

//+------------------------------------------------------------------+
//| BUILD OPENING RANGE                                              |
//| FIRST 30 MINUTES OF SESSION                                      |
//+------------------------------------------------------------------+
void BuildOpeningRangeUTC()
{
   //====================================================
   // RESET
   //====================================================
   asiaORHigh = -DBL_MAX;
   asiaORLow = DBL_MAX;
   londonORHigh = -DBL_MAX;
   londonORLow = DBL_MAX;
   nyORHigh = -DBL_MAX;
   nyORLow = DBL_MAX;
   //====================================================
   // UTC DAY START
   //====================================================
   datetime utcNow =
       TimeGMT();
   string utcDate =
       TimeToString(
           utcNow,
           TIME_DATE);
   datetime utcDayStart =
       StringToTime(
           utcDate);
   //====================================================
   // SESSION OPENING RANGE WINDOWS
   //====================================================
   //----------------------------------------------------
   // ASIA
   // 00:00 → 00:30 UTC
   //----------------------------------------------------
   asiaORStart =
       utcDayStart;
   asiaOREnd =
       utcDayStart + PeriodSeconds(ORBTF);
   //----------------------------------------------------
   // LONDON
   // 07:30 → 07:30+ORBTF UTC
   //----------------------------------------------------
   londonORStart =
       utcDayStart +
       (7 * 3600) +
       (30 * 60);
   londonOREnd =
       londonORStart +
       PeriodSeconds(ORBTF);
   //----------------------------------------------------
   // NEW YORK
   // 13:30 → 13:30+ORBTF UTC
   //----------------------------------------------------
   nyORStart =
       utcDayStart +
       (13 * 3600) +
       (30 * 60);
   nyOREnd =
       nyORStart +
       PeriodSeconds(ORBTF);
   //====================================================
   // SERVER OFFSET
   //====================================================
   int offsetSeconds =
       (int)(TimeCurrent() -
             TimeGMT());
   //====================================================
   // CONVERT TO SERVER TIME
   //====================================================
   asiaORStart += offsetSeconds;
   asiaOREnd += offsetSeconds;
   londonORStart += offsetSeconds;
   londonOREnd += offsetSeconds;
   nyORStart += offsetSeconds;
   nyOREnd += offsetSeconds;
   //====================================================
   // BUILD RANGE
   //====================================================
   int bars =
       ArraySize(ratesORB);
   for (int i = 0; i < bars; i++)
   {
      datetime t =
          ratesORB[i].time;
      //----------------------------------------------------
      // ASIA OR
      //----------------------------------------------------
      if (
          t >= asiaORStart &&
          t < asiaOREnd)
      {
         if (ratesORB[i].high > asiaORHigh)
            asiaORHigh = ratesORB[i].high;
         if (ratesORB[i].low < asiaORLow)
            asiaORLow = ratesORB[i].low;
      }
      //----------------------------------------------------
      // LONDON OR
      //----------------------------------------------------
      if (
          t >= londonORStart &&
          t < londonOREnd)
      {
         if (ratesORB[i].high > londonORHigh)
            londonORHigh = ratesORB[i].high;
         if (ratesORB[i].low < londonORLow)
            londonORLow = ratesORB[i].low;
      }
      //----------------------------------------------------
      // NY OR
      //----------------------------------------------------
      if (
          t >= nyORStart &&
          t < nyOREnd)
      {
         if (ratesORB[i].high > nyORHigh)
            nyORHigh = ratesORB[i].high;
         if (ratesORB[i].low < nyORLow)
            nyORLow = ratesORB[i].low;
      }
   }
   //====================================================
   // CONFIRM: mark range valid only if it was built
   //====================================================
   asiaORConfirmed = (asiaORHigh > asiaORLow && asiaORHigh > 0);
   londonORConfirmed = (londonORHigh > londonORLow && londonORHigh > 0);
   nyORConfirmed = (nyORHigh > nyORLow && nyORHigh > 0);
}

//+------------------------------------------------------------------+
//| DETECT OPENING RANGE BREAKOUT                                    |
//+------------------------------------------------------------------+
void DetectOpeningRangeBreakoutUTC()
{
   //====================================================
   // RESET ALL STATES
   //====================================================
   asiaBreakoutUp = false;
   asiaBreakoutDown = false;
   londonBreakoutUp = false;
   londonBreakoutDown = false;
   nyBreakoutUp = false;
   nyBreakoutDown = false;
   asiaRejectHigh = false;
   asiaRejectLow = false;
   londonRejectHigh = false;
   londonRejectLow = false;
   nyRejectHigh = false;
   nyRejectLow = false;
   asiaFalseBreakHigh = false;
   asiaFalseBreakLow = false;
   londonFalseBreakHigh = false;
   londonFalseBreakLow = false;
   nyFalseBreakHigh = false;
   nyFalseBreakLow = false;
   asiaBreakoutReentryHigh = false;
   asiaBreakoutReentryLow = false;
   londonBreakoutReentryHigh = false;
   londonBreakoutReentryLow = false;
   nyBreakoutReentryHigh = false;
   nyBreakoutReentryLow = false;
   //====================================================
   // VALIDATION
   //====================================================
   double atr = GetATR(1);
   if (atr <= 0)
      return;
   int totalBars = ArraySize(ratesORB);
   if (totalBars < 12)
      return;
   //====================================================
   // CURRENT CANDLE
   //====================================================
   double closePrice = ratesORB[0].close;
   double openPrice = ratesORB[0].open;
   double highPrice = ratesORB[0].high;
   double lowPrice = ratesORB[0].low;
   double upperWick = highPrice - MathMax(openPrice, closePrice);
   double lowerWick = MathMin(openPrice, closePrice) - lowPrice;
   double body = MathAbs(closePrice - openPrice);
   //----------------------------------------------------
   // strongUpperReject: wick is dominant AND large enough to matter
   //----------------------------------------------------
   bool strongUpperReject =
       (upperWick > body * 1.2 && upperWick > atr * 0.10);
   bool strongLowerReject =
       (lowerWick > body * 1.2 && lowerWick > atr * 0.10);
   bool impulsive = (body > atr * 0.35);
   double bufATR = atr * 0.03;
   datetime now = TimeCurrent();
   //====================================================
   // ASIA — all checks after OR window closes
   //====================================================
   if (asiaORConfirmed && now >= asiaOREnd)
   {
      //--------------------------------------------------
      // BREAKOUT
      //--------------------------------------------------
      bool aBoUp = (closePrice > asiaORHigh + bufATR);
      bool aBoDown = (closePrice < asiaORLow - bufATR);
      if (aBoUp && impulsive)
         asiaBreakoutUp = true;
      if (aBoDown && impulsive)
         asiaBreakoutDown = true;
      //--------------------------------------------------
      // REJECTION (wick-through, close back inside)
      //--------------------------------------------------
      if (highPrice > asiaORHigh && closePrice < asiaORHigh && strongUpperReject)
         asiaRejectHigh = true;
      if (lowPrice < asiaORLow && closePrice > asiaORLow && strongLowerReject)
         asiaRejectLow = true;
      //--------------------------------------------------
      // FALSE BREAKOUT (1-3 bars outside → current back inside)
      //--------------------------------------------------
      bool prevFarAboveA = false, prevFarBelowA = false;
      for (int k = 1; k <= 3; k++)
      {
         if (ratesORB[k].close > asiaORHigh + bufATR)
            prevFarAboveA = true;
         if (ratesORB[k].close < asiaORLow - bufATR)
            prevFarBelowA = true;
      }
      if (prevFarAboveA && closePrice < asiaORHigh && closePrice < openPrice && impulsive)
         asiaFalseBreakHigh = true;
      if (prevFarBelowA && closePrice > asiaORLow && closePrice > openPrice && impulsive)
         asiaFalseBreakLow = true;
      //--------------------------------------------------
      // RE-ENTRY (4-10 bars outside → current back inside range)
      //--------------------------------------------------
      bool prevSustainedAboveA = false, prevSustainedBelowA = false;
      for (int k = 4; k <= 10; k++)
      {
         if (ratesORB[k].close > asiaORHigh + bufATR)
            prevSustainedAboveA = true;
         if (ratesORB[k].close < asiaORLow - bufATR)
            prevSustainedBelowA = true;
      }
      if (prevSustainedAboveA && closePrice < asiaORHigh && closePrice > asiaORLow)
         asiaBreakoutReentryHigh = true;
      if (prevSustainedBelowA && closePrice > asiaORLow && closePrice < asiaORHigh)
         asiaBreakoutReentryLow = true;
   }
   //====================================================
   // LONDON — all checks after OR window closes
   //====================================================
   if (londonORConfirmed && now >= londonOREnd)
   {
      //--------------------------------------------------
      // BREAKOUT
      //--------------------------------------------------
      bool lBoUp = (closePrice > londonORHigh + bufATR);
      bool lBoDown = (closePrice < londonORLow - bufATR);
      if (lBoUp && impulsive)
         londonBreakoutUp = true;
      if (lBoDown && impulsive)
         londonBreakoutDown = true;
      //--------------------------------------------------
      // REJECTION
      //--------------------------------------------------
      if (highPrice > londonORHigh && closePrice < londonORHigh && strongUpperReject)
         londonRejectHigh = true;
      if (lowPrice < londonORLow && closePrice > londonORLow && strongLowerReject)
         londonRejectLow = true;
      //--------------------------------------------------
      // FALSE BREAKOUT (1-3 bars outside → current back inside)
      //--------------------------------------------------
      bool prevFarAboveL = false, prevFarBelowL = false;
      for (int k = 1; k <= 3; k++)
      {
         if (ratesORB[k].close > londonORHigh + bufATR)
            prevFarAboveL = true;
         if (ratesORB[k].close < londonORLow - bufATR)
            prevFarBelowL = true;
      }
      if (prevFarAboveL && closePrice < londonORHigh && closePrice < openPrice && impulsive)
         londonFalseBreakHigh = true;
      if (prevFarBelowL && closePrice > londonORLow && closePrice > openPrice && impulsive)
         londonFalseBreakLow = true;
      //--------------------------------------------------
      // RE-ENTRY (4-10 bars outside → current back inside range)
      //--------------------------------------------------
      bool prevSustainedAboveL = false, prevSustainedBelowL = false;
      for (int k = 4; k <= 10; k++)
      {
         if (ratesORB[k].close > londonORHigh + bufATR)
            prevSustainedAboveL = true;
         if (ratesORB[k].close < londonORLow - bufATR)
            prevSustainedBelowL = true;
      }
      if (prevSustainedAboveL && closePrice < londonORHigh && closePrice > londonORLow)
         londonBreakoutReentryHigh = true;
      if (prevSustainedBelowL && closePrice > londonORLow && closePrice < londonORHigh)
         londonBreakoutReentryLow = true;
   }
   //====================================================
   // NEW YORK — all checks after OR window closes
   //====================================================
   if (nyORConfirmed && now >= nyOREnd)
   {
      //--------------------------------------------------
      // BREAKOUT
      //--------------------------------------------------
      bool nBoUp = (closePrice > nyORHigh + bufATR);
      bool nBoDown = (closePrice < nyORLow - bufATR);
      if (nBoUp && impulsive)
         nyBreakoutUp = true;
      if (nBoDown && impulsive)
         nyBreakoutDown = true;
      //--------------------------------------------------
      // REJECTION
      //--------------------------------------------------
      if (highPrice > nyORHigh && closePrice < nyORHigh && strongUpperReject)
         nyRejectHigh = true;
      if (lowPrice < nyORLow && closePrice > nyORLow && strongLowerReject)
         nyRejectLow = true;
      //--------------------------------------------------
      // FALSE BREAKOUT (1-3 bars outside → current back inside)
      //--------------------------------------------------
      bool prevFarAboveN = false, prevFarBelowN = false;
      for (int k = 1; k <= 3; k++)
      {
         if (ratesORB[k].close > nyORHigh + bufATR)
            prevFarAboveN = true;
         if (ratesORB[k].close < nyORLow - bufATR)
            prevFarBelowN = true;
      }
      if (prevFarAboveN && closePrice < nyORHigh && closePrice < openPrice && impulsive)
         nyFalseBreakHigh = true;
      if (prevFarBelowN && closePrice > nyORLow && closePrice > openPrice && impulsive)
         nyFalseBreakLow = true;
      //--------------------------------------------------
      // RE-ENTRY (4-10 bars outside → current back inside range)
      //--------------------------------------------------
      bool prevSustainedAboveN = false, prevSustainedBelowN = false;
      for (int k = 4; k <= 10; k++)
      {
         if (ratesORB[k].close > nyORHigh + bufATR)
            prevSustainedAboveN = true;
         if (ratesORB[k].close < nyORLow - bufATR)
            prevSustainedBelowN = true;
      }
      if (prevSustainedAboveN && closePrice < nyORHigh && closePrice > nyORLow)
         nyBreakoutReentryHigh = true;
      if (prevSustainedBelowN && closePrice > nyORLow && closePrice < nyORHigh)
         nyBreakoutReentryLow = true;
   }
}

//| ORB ALERT ENGINE                                                 |
//+------------------------------------------------------------------+
void ORBAlertEngine()
{
   //====================================================
   // ASIA BUY ALERT
   //====================================================
   if (
       asiaBreakoutUp &&
       !asiaBuyAlertSent)
   {
      string msg =
          _Symbol +
          " | ASIA ORB BUY BREAKOUT";
      SendDashboardNotification(msg);
      asiaBuyAlertSent =
          true;
      asiaSellAlertSent =
          false;
   }
   //====================================================
   // ASIA SELL ALERT
   //====================================================
   if (
       asiaBreakoutDown &&
       !asiaSellAlertSent)
   {
      string msg =
          _Symbol +
          " | ASIA ORB SELL BREAKOUT";
      SendDashboardNotification(msg);
      asiaSellAlertSent =
          true;
      asiaBuyAlertSent =
          false;
   }
   //====================================================
   // LONDON BUY ALERT
   //====================================================
   if (
       londonBreakoutUp &&
       !londonBuyAlertSent)
   {
      string msg =
          _Symbol +
          " | LONDON ORB BUY BREAKOUT";
      SendDashboardNotification(msg);
      londonBuyAlertSent =
          true;
      londonSellAlertSent =
          false;
   }
   //====================================================
   // LONDON SELL ALERT
   //====================================================
   if (
       londonBreakoutDown &&
       !londonSellAlertSent)
   {
      string msg =
          _Symbol +
          " | LONDON ORB SELL BREAKOUT";
      SendDashboardNotification(msg);
      londonSellAlertSent =
          true;
      londonBuyAlertSent =
          false;
   }
   //====================================================
   // NY BUY ALERT
   //====================================================
   if (
       nyBreakoutUp &&
       !nyBuyAlertSent)
   {
      string msg =
          _Symbol +
          " | NEW YORK ORB BUY BREAKOUT";
      SendDashboardNotification(msg);
      nyBuyAlertSent =
          true;
      nySellAlertSent =
          false;
   }
   //====================================================
   // NY SELL ALERT
   //====================================================
   if (
       nyBreakoutDown &&
       !nySellAlertSent)
   {
      string msg =
          _Symbol +
          " | NEW YORK ORB SELL BREAKOUT";
      SendDashboardNotification(msg);
      nySellAlertSent =
          true;
      nyBuyAlertSent =
          false;
   }
   //====================================================
   // REJECTION ALERTS (independent high / low checks)
   //====================================================
   if (asiaRejectHigh && !asiaRejectHighAlert)
   {
      SendDashboardNotification(_Symbol + " | ASIA ORB REJECT HIGH");
      asiaRejectHighAlert = true;
   }
   if (asiaRejectLow && !asiaRejectLowAlert)
   {
      SendDashboardNotification(_Symbol + " | ASIA ORB REJECT LOW");
      asiaRejectLowAlert = true;
   }
   if (londonRejectHigh && !londonRejectHighAlert)
   {
      SendDashboardNotification(_Symbol + " | LONDON ORB REJECT HIGH");
      londonRejectHighAlert = true;
   }
   if (londonRejectLow && !londonRejectLowAlert)
   {
      SendDashboardNotification(_Symbol + " | LONDON ORB REJECT LOW");
      londonRejectLowAlert = true;
   }
   if (nyRejectHigh && !nyRejectHighAlert)
   {
      SendDashboardNotification(_Symbol + " | NY ORB REJECT HIGH");
      nyRejectHighAlert = true;
   }
   if (nyRejectLow && !nyRejectLowAlert)
   {
      SendDashboardNotification(_Symbol + " | NY ORB REJECT LOW");
      nyRejectLowAlert = true;
   }
   //====================================================
   // FALSE BREAKOUT ALERTS
   //====================================================
   if (asiaFalseBreakHigh && !asiaFalseBreakHighAlert)
   {
      SendDashboardNotification(_Symbol + " | ASIA FALSE BREAK HIGH");
      asiaFalseBreakHighAlert = true;
   }
   if (asiaFalseBreakLow && !asiaFalseBreakLowAlert)
   {
      SendDashboardNotification(_Symbol + " | ASIA FALSE BREAK LOW");
      asiaFalseBreakLowAlert = true;
   }
   if (londonFalseBreakHigh && !londonFalseBreakHighAlert)
   {
      SendDashboardNotification(_Symbol + " | LONDON FALSE BREAK HIGH");
      londonFalseBreakHighAlert = true;
   }
   if (londonFalseBreakLow && !londonFalseBreakLowAlert)
   {
      SendDashboardNotification(_Symbol + " | LONDON FALSE BREAK LOW");
      londonFalseBreakLowAlert = true;
   }
   if (nyFalseBreakHigh && !nyFalseBreakHighAlert)
   {
      SendDashboardNotification(_Symbol + " | NY FALSE BREAK HIGH");
      nyFalseBreakHighAlert = true;
   }
   if (nyFalseBreakLow && !nyFalseBreakLowAlert)
   {
      SendDashboardNotification(_Symbol + " | NY FALSE BREAK LOW");
      nyFalseBreakLowAlert = true;
   }
   //====================================================
   // RE-ENTRY ALERTS (breakout failed, price back inside)
   //====================================================
   if (asiaBreakoutReentryHigh && !asiaReentryHighAlert)
   {
      SendDashboardNotification(_Symbol + " | ASIA ORB RE-ENTRY FROM ABOVE");
      asiaReentryHighAlert = true;
   }
   if (asiaBreakoutReentryLow && !asiaReentryLowAlert)
   {
      SendDashboardNotification(_Symbol + " | ASIA ORB RE-ENTRY FROM BELOW");
      asiaReentryLowAlert = true;
   }
   if (londonBreakoutReentryHigh && !londonReentryHighAlert)
   {
      SendDashboardNotification(_Symbol + " | LONDON ORB RE-ENTRY FROM ABOVE");
      londonReentryHighAlert = true;
   }
   if (londonBreakoutReentryLow && !londonReentryLowAlert)
   {
      SendDashboardNotification(_Symbol + " | LONDON ORB RE-ENTRY FROM BELOW");
      londonReentryLowAlert = true;
   }
   if (nyBreakoutReentryHigh && !nyReentryHighAlert)
   {
      SendDashboardNotification(_Symbol + " | NY ORB RE-ENTRY FROM ABOVE");
      nyReentryHighAlert = true;
   }
   if (nyBreakoutReentryLow && !nyReentryLowAlert)
   {
      SendDashboardNotification(_Symbol + " | NY ORB RE-ENTRY FROM BELOW");
      nyReentryLowAlert = true;
   }
